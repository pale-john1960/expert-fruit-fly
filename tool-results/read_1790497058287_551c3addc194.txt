   132→export function taskMeta(task: string) {
   133→  return (
   134→    TASK_META[task] ?? {
   135→      label: task ? task.charAt(0).toUpperCase() + task.slice(1) : "Unknown",
   136→      icon: FlaskConical,
   137→      badge: "border-border bg-muted text-muted-foreground",
   138→      iconColor: "text-muted-foreground",
   139→    }
   140→  );
   141→}
   142→
   143→export function formatScore(task: string, score: number): string {
   144→  const num =
   145→    Math.abs(score) >= 1000 ? `${(score / 1000).toFixed(1)}k` : `${Math.round(score * 10) / 10}`;
   146→  if (task === "dino") return `${num} pts`;
   147→  if (task === "bicycle") return `${num} m`;
   148→  return num;
   149→}
   150→
   151→export function formatBytes(n: number): string {
   152→  if (n >= 1024 * 1024) return `${(n / (1024 * 1024)).toFixed(1)} MB`;
   153→  if (n >= 1024) return `${(n / 1024).toFixed(1)} KB`;
   154→  return `${n} B`;
   155→}
   156→
   157→export function relativeDate(iso: string): string {
   158→  try {
   159→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
   160→  } catch {
   161→    return "unknown";
   162→  }
   163→}
   164→
   165→function sanitizeFileName(name: string): string {
   166→  const clean = name
   167→    .toLowerCase()
   168→    .replace(/[^a-z0-9-_]+/g, "-")
   169→    .replace(/^-+|-+$/g, "");
   170→  return clean || "brain";
   171→}
   172→
   173→export async function fetchBrain(id: string): Promise<{ brain: BrainRow; snapshot: BrainSnapshot }> {
   174→  const res = await fetch(`/api/brains/${id}`, { cache: "no-store" });
   175→  const data = await res.json().catch(() => ({}));
   176→  if (!res.ok) {
   177→    throw new Error(data?.error ?? `Could not fetch brain (${res.status})`);
   178→  }
   179→  if (!data?.snapshot || typeof data.snapshot !== "object" || !data.snapshot.weights) {
   180→    throw new Error("Stored snapshot is malformed");
   181→  }
   182→  return data;
   183→}

   207→export function BrainLibrary(_props: BrainLibraryProps = {}) {
   208→  const [brains, setBrains] = useState<BrainRow[]>([]);
   209→  const [loading, setLoading] = useState(true);
   210→  const [error, setError] = useState<string | null>(null);
   211→  const [taskFilter, setTaskFilter] = useState<TaskFilter>("all");
   212→  const [query, setQuery] = useState("");
   213→  const [busyId, setBusyId] = useState<string | null>(null);
   214→  const [deleteTarget, setDeleteTarget] = useState<BrainRow | null>(null);
   215→  const [deleteBusy, setDeleteBusy] = useState(false);
   216→  const [importOpen, setImportOpen] = useState(false);
   217→  const [importing, setImporting] = useState(false);
   218→  const [dragOver, setDragOver] = useState(false);
   219→  /** the brain whose report-card dialog is open (Task 10-b) */
   220→  const [reportRow, setReportRow] = useState<BrainRow | null>(null);
   221→  /** ordered ids of the brains picked for the genome diff (max 2, Task 11-a) */
   222→  const [diffIds, setDiffIds] = useState<string[]>([]);
   223→  const [diffOpen, setDiffOpen] = useState(false);
   224→
   225→  const fileInputRef = useRef<HTMLInputElement>(null);
   226→
   227→  const loadList = useCallback(async (task: TaskFilter) => {
   228→    setLoading(true);
   229→    setError(null);
   230→    try {
   231→      const qs = task !== "all" ? `?task=${encodeURIComponent(task)}` : "";
   232→      const res = await fetch(`/api/brains${qs}`, { cache: "no-store" });
   233→      const data = await res.json().catch(() => ({}));
   234→      if (!res.ok) throw new Error(data?.error ?? `Server responded ${res.status}`);
   235→      setBrains(Array.isArray(data.brains) ? (data.brains as BrainRow[]) : []);
   236→    } catch (err) {
   237→      const msg = err instanceof Error ? err.message : "Failed to load the library";
   238→      setError(msg);
   239→      if (brains.length > 0) toast.error(`Couldn't refresh the library: ${msg}`);
   240→    } finally {
   241→      setLoading(false);
   242→    }
   243→  }, [brains.length]);
   244→
   245→  useEffect(() => {
   246→    void loadList("all");
   247→  }, []);
   248→
   249→  // keep the diff selection valid when the list changes (deletes / refresh)
   250→  useEffect(() => {
   251→    setDiffIds((ids) => ids.filter((id) => brains.some((b) => b.id === id)));
   252→  }, [brains]);
   253→
   254→  /** toggle a row into the genome-diff selection (max 2, FIFO replacement) */
   255→  const toggleDiff = useCallback((id: string) => {
   256→    setDiffIds((prev) => {
   257→      if (prev.includes(id)) return prev.filter((x) => x !== id);
   258→      if (prev.length >= 2) return [prev[1], id];
   259→      return [...prev, id];
   260→    });
   261→  }, []);
   262→
   263→  const handleFilterChange = (task: TaskFilter) => {
   264→    setTaskFilter(task);
   265→    void loadList(task);
   266→  };
   267→
   268→  const visible = useMemo(() => {
   269→    const q = query.trim().toLowerCase();
   270→    return brains.filter((b) => {
   271→      if (q) {
   272→        const haystack = `${b.name} ${b.task} ${b.note ?? ""}`.toLowerCase();
   273→        if (!haystack.includes(q)) return false;
   274→      }
   275→      return true;
   276→    });
   277→  }, [brains, query]);
   278→
   279→  const totalBytes = useMemo(() => brains.reduce((acc, b) => acc + (b.snapshotBytes ?? 0), 0), [brains]);
   280→
   281→  /** the [A, B] pair once exactly two brains are selected for comparison */
   282→  const diffRows = useMemo(() => {
   283→    if (diffIds.length !== 2) return null;
   284→    const a = brains.find((b) => b.id === diffIds[0]);
   285→    const b = brains.find((b) => b.id === diffIds[1]);
   286→    return a && b ? ([a, b] as const) : null;
   287→  }, [diffIds, brains]);
   288→
   289→  // ----- actions ----------------------------------------------------------
   290→
   291→  const handleLoad = async (row: BrainRow) => {
   292→    setBusyId(row.id);
   293→    try {
   294→      const { snapshot } = await fetchBrain(row.id);
   295→      useBrainStore.getState().requestLoad(row.task, snapshot, row.name);
   296→      const meta = taskMeta(row.task);
   297→      toast.success(`Loaded "${row.name}" — open the ${meta.label} tab to continue training`);
   298→    } catch (err) {
   299→      toast.error(err instanceof Error ? err.message : "Load failed");
   300→    } finally {
   301→      setBusyId(null);
   302→    }
   303→  };
   304→
   305→  const handleExport = async (row: BrainRow) => {
   306→    setBusyId(row.id);
   307→    try {
   308→      const { snapshot } = await fetchBrain(row.id);
   309→      const blob = new Blob([JSON.stringify(snapshot, null, 2)], { type: "application/json" });
   310→      const url = URL.createObjectURL(blob);
   311→      const a = document.createElement("a");
   312→      a.href = url;
   313→      a.download = `fly-brain-${sanitizeFileName(row.name)}-gen${row.generation}.json`;
   314→      document.body.appendChild(a);
   315→      a.click();
   316→      a.remove();
   317→      URL.revokeObjectURL(url);
   318→      toast.success(`Exported "${row.name}" (gen ${row.generation}) as JSON`);
   319→    } catch (err) {
   320→      toast.error(err instanceof Error ? err.message : "Export failed");
   321→    } finally {
   322→      setBusyId(null);
   323→    }
   324→  };
   325→
   326→  const handleDelete = async () => {
   327→    if (!deleteTarget) return;
   328→    setDeleteBusy(true);
   329→    const row = deleteTarget;
   330→    try {
   331→      const res = await fetch(`/api/brains/${row.id}`, { method: "DELETE" });
   332→      const data = await res.json().catch(() => ({}));
   333→      if (!res.ok) throw new Error(data?.error ?? `Delete failed (${res.status})`);
   334→      setBrains((bs) => bs.filter((b) => b.id !== row.id));
   335→      toast.success(`Deleted "${row.name}" — that connectome is gone`);
   336→      setDeleteTarget(null);
   337→    } catch (err) {
   338→      toast.error(err instanceof Error ? err.message : "Delete failed");
   339→    } finally {
   340→      setDeleteBusy(false);
   341→    }
   342→  };
   343→
   344→  const handleImportFile = async (file: File) => {
   345→    setImporting(true);
   346→    try {
   347→      const text = await file.text();
   348→      let parsed: unknown;
   349→      try {
   350→        parsed = JSON.parse(text);
   351→      } catch {
   352→        throw new Error("That file isn't valid JSON");
   353→      }
   354→      const snap = parsed as Partial<BrainSnapshot> | null;
   355→      if (!snap || typeof snap !== "object" || Array.isArray(snap)) {
   356→        throw new Error("Not a fly-brain file — expected a JSON object");
   357→      }
   358→      if (snap.version !== 1) throw new Error(`Unsupported brain version: ${String(snap.version)}`);
   359→      if (typeof snap.task !== "string" || !snap.task.trim()) throw new Error("Snapshot is missing its task");
   360→      if (!snap.arch || typeof snap.arch !== "object") throw new Error("Snapshot is missing its architecture");
   361→      if (!snap.weights || typeof snap.weights !== "object") throw new Error("Snapshot is missing its weights");
   362→      for (const key of ["kenyonToMbon", "mbonToMotor", "mbonBias", "motorBias"] as const) {
   363→        if (!Array.isArray(snap.weights[key])) throw new Error(`Snapshot weights.${key} is missing`);
   364→      }
   365→
   366→      const fallbackName = file.name
   367→        .replace(/\.json$/i, "")
   368→        .replace(/[_-]+/g, " ")
   369→        .trim();
   370→      const name =
   371→        typeof snap.name === "string" && snap.name.trim() ? snap.name.trim() : fallbackName || "Imported brain";
   372→
   373→      const res = await fetch("/api/brains", {
   374→        method: "POST",
   375→        headers: { "Content-Type": "application/json" },
   376→        body: JSON.stringify({
   377→          name,
   378→          task: snap.task,
   379→          generation: typeof snap.generation === "number" ? snap.generation : 0,
   380→          score: typeof snap.score === "number" ? snap.score : 0,
   381→          note: typeof snap.meta?.note === "string" ? snap.meta.note : undefined,
   382→          snapshot: snap,
   383→        }),
   384→      });
   385→      const data = await res.json().catch(() => ({}));
   386→      if (!res.ok) throw new Error(data?.error ?? `Import failed (${res.status})`);
   387→
   388→      toast.success(`Imported "${data.brain?.name ?? name}" (gen ${data.brain?.generation ?? 0}) into the library`);
   389→      setImportOpen(false);
   390→      await loadList(taskFilter);
   391→    } catch (err) {
   392→      toast.error(err instanceof Error ? err.message : "Import failed");
   393→    } finally {
   394→      setImporting(false);
   395→      if (fileInputRef.current) fileInputRef.current.value = "";
   396→    }
   397→  };
   398→
   399→  // ----- report card (Task 10-b) ------------------------------------------
   400→
   401→  const handleOpenReport = (row: BrainRow) => {
   402→    playSound("click");
   403→    setReportRow(row);
   404→  };
   405→
   406→  // ----- genome diff selection (Task 11-a) ---------------------------------
   407→
   408→  /** per-row compare toggle — 44px touch target, aria-pressed, amber when on */
   409→  const diffToggleButton = (row: BrainRow) => {
   410→    const selected = diffIds.includes(row.id);
   411→    const position = selected ? diffIds.indexOf(row.id) : -1;
   412→    return (
   413→      <Button
   414→        variant="outline"
   415→        size="icon"
   416→        data-testid="diff-toggle"
   417→        data-selected={selected}
   418→        aria-pressed={selected}
   419→        aria-label={
   420→          selected
   421→            ? `Remove ${row.name} from the genome comparison`
   422→            : `Select ${row.name} for genome comparison`
   423→        }
   424→        title={
   425→          selected
   426→            ? `In comparison as ${position === 0 ? "A" : "B"} — click to remove`
   427→            : "Compare genomes"
   428→        }
   429→        className={cn(
   430→          "h-11 w-11 shrink-0 transition-colors",
   431→          selected
   432→            ? "border-amber-500/60 bg-amber-500/15 text-amber-500 hover:bg-amber-500/25 hover:text-amber-500 dark:text-amber-400 dark:hover:text-amber-300"
   433→            : "text-muted-foreground hover:border-amber-500/40 hover:text-amber-500 dark:hover:text-amber-400",
   434→        )}
   435→        onClick={() => toggleDiff(row.id)}
   436→      >
   437→        <GitCompareArrows className="h-4 w-4" aria-hidden />
   438→      </Button>
   439→    );
   440→  };
   441→
   442→  // ----- render helpers ---------------------------------------------------
   443→
   444→  const showEmptyState = !loading && !error && visible.length === 0;
   445→  const showFilteredEmpty = showEmptyState && (brains.length > 0 || query.trim() !== "");
   446→
   447→  const rowActions = (row: BrainRow, full = false) => (
   448→    <div className={cn("flex items-center gap-2", full && "w-full")}>
   449→      <Button
   450→        size="sm"
   451→        className={cn("h-11 gap-1.5 bg-emerald-600 px-4 text-white hover:bg-emerald-500", full && "flex-1")}
   452→        disabled={busyId === row.id}
   453→        onClick={() => void handleLoad(row)}
   454→        aria-label={`Load ${row.name} into the trainer`}
   455→      >
   456→        {busyId === row.id ? <Loader2 className="h-4 w-4 animate-spin" /> : <Play className="h-4 w-4" />}
   457→        Load
   458→      </Button>
   459→      <Button
   460→        variant="outline"
   461→        size={full ? "sm" : "icon"}
   462→        className={cn(
   463→          "h-11 shrink-0 border-teal-500/30 hover:border-teal-500/50 hover:bg-teal-500/10 hover:text-teal-600 dark:hover:text-teal-300",
   464→          full ? "flex-1 gap-1.5 px-4" : "w-11",
   465→        )}
   466→        disabled={busyId === row.id}
   467→        onClick={() => handleOpenReport(row)}
   468→        aria-label={`Open report card for ${row.name}`}
   469→        title="Report card"
   470→      >
   471→        <GraduationCap className="h-4 w-4" aria-hidden />
   472→        {full ? "Report card" : null}
   473→      </Button>
   474→      <DropdownMenu>
   475→        <DropdownMenuTrigger asChild>
   476→          <Button
   477→            variant="outline"
   478→            size="icon"
   479→            className="h-11 w-11 shrink-0"
   480→            disabled={busyId === row.id}
   481→            aria-label={`More actions for ${row.name}`}
   482→          >
   483→            <MoreHorizontal className="h-4 w-4" />
   484→          </Button>
   485→        </DropdownMenuTrigger>
   486→        <DropdownMenuContent align="end" className="w-44">
   487→          <DropdownMenuItem
   488→            className="min-h-11 cursor-pointer"
   489→            onSelect={() => void handleExport(row)}
   490→          >
   491→            <Download className="h-4 w-4" /> Export JSON
   492→          </DropdownMenuItem>
   493→          <DropdownMenuItem
   494→            className="min-h-11 cursor-pointer text-rose-400 focus:text-rose-300"
   495→            onSelect={() => setDeleteTarget(row)}
   496→          >
   497→            <Trash2 className="h-4 w-4" /> Delete
   498→          </DropdownMenuItem>
   499→        </DropdownMenuContent>
   500→      </DropdownMenu>
   501→    </div>
   502→  );
   503→
   504→  // ----- render -----------------------------------------------------------
   505→
   506→  return (
   507→    <div data-testid="brain-library" className="w-full">
   508→      {/* sonner toaster — layout only mounts the radix one, so mount it here */}
   509→      <Toaster position="bottom-right" />
   510→
   511→      <Card className="gap-4 border-border/60 bg-card/70 py-5 backdrop-blur-sm sm:py-6">
   512→        <CardHeader className="border-b border-border/60 px-4 pb-4 sm:px-6">
   513→          <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
   514→            <div className="flex items-start gap-3">
   515→              <div className="mt-0.5 flex h-10 w-10 shrink-0 items-center justify-center rounded-xl border border-emerald-500/30 bg-emerald-500/10">
   516→                <Brain className="h-5 w-5 text-emerald-400" aria-hidden />
   517→              </div>
   518→              <div>
   519→                <CardTitle className="text-lg">Brain Library</CardTitle>
   520→                <CardDescription className="mt-1 max-w-xl text-sm">
   521→                  Saved connectomes. Load one and the fly remembers everything it learned —
   522→                  weights, generation, scars.
   523→                </CardDescription>
   524→              </div>
   525→            </div>
   526→            <div className="flex shrink-0 items-center gap-2">
   527→              <Button
   528→                variant="outline"
   529→                size="icon"
   530→                className="h-11 w-11"
   531→                onClick={() => void loadList(taskFilter)}
   532→                disabled={loading}
   533→                aria-label="Refresh the brain library"

   535→              >
   536→                <RefreshCw className={cn("h-4 w-4", loading && "animate-spin")} />
   537→              </Button>
   538→              <Button

   541→                onClick={() => setImportOpen(true)}
