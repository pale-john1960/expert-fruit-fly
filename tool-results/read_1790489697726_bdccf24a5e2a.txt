    96→function eventIcon(kind: TrainerEvent["kind"]) {
    97→  switch (kind) {
    98→    case "fall":
    99→      return <Zap className="h-3.5 w-3.5 text-rose-400" />;
   100→    case "offroad":
   101→      return <TriangleAlert className="h-3.5 w-3.5 text-amber-400" />;
   102→    case "milestone":
   103→      return <Sparkles className="h-3.5 w-3.5 text-amber-300" />;
   104→    case "gen":
   105→      return <Flag className="h-3.5 w-3.5 text-rose-300" />;
   106→    case "best":
   107→      return <Trophy className="h-3.5 w-3.5 text-emerald-400" />;
   108→    case "load":
   109→      return <Eye className="h-3.5 w-3.5 text-emerald-400" />;
   110→    default:
   111→      return <Info className="h-3.5 w-3.5 text-muted-foreground" />;
   112→  }

   116→  turbo,

   170→export function BicycleTrainer() {
   171→  const [core] = useState(() => new BicycleTrainerCore(5));
   172→  const retinaCanvas = useRef<HTMLCanvasElement | null>(null);
   173→  const [hud, setHud] = useState<HudSnapshot | null>(null);
   174→  const [leaderBrain, setLeaderBrain] = useState<FlyBrain | null>(null);
   175→  const [running, setRunning] = useState(true);
   176→  const [turbo, setTurbo] = useState(1);
   177→  const [preset, setPresetState] = useState<"steady" | "standard" | "frisky">("standard");
   178→  const [popQueued, setPopQueued] = useState(5);
   179→  const [mutation, setMutation] = useState(0.3);
   180→  const [watchBest, setWatchBest] = useState(false);
   181→  const [brainName, setBrainName] = useState("Dusk Rider");
   182→  const [saving, setSaving] = useState(false);
   183→  const [session, setSession] = useState<{
   184→    generation: number;
   185→    bestDistance: number;
   186→    history: { gen: number; best: number; avg: number }[];
   187→    snapshot: BrainSnapshot;
   188→  } | null>(null);
   189→
   190→  // --- "You vs the fly" challenge (sim lives on the core; React mirrors) ---
   191→  const [challengeActive, setChallengeActive] = useState(false);
   192→  const [challengeHud, setChallengeHud] = useState<ChallengeHud | null>(null);
   193→  /** session W/L tally — persists across challenges AND population resets */
   194→  const tallyRef = useRef({ wins: 0, losses: 0, best: 0 });
   195→  const tallyCountedRef = useRef(false);
   196→
   197→  // ---- HUD polling (keeps React renders at ~6 Hz, the scene stays 60 fps)
   198→  useEffect(() => {
   199→    const iv = setInterval(() => {
   200→      setHud(core.hudSnapshot());
   201→      const ch = core.challenge;
   202→      if (ch) {
   203→        // count the tally exactly once per finished challenge
   204→        if (ch.result && !tallyCountedRef.current) {
   205→          tallyCountedRef.current = true;
   206→          if (ch.result.winner === "you") tallyRef.current.wins += 1;
   207→          else if (ch.result.winner === "fly") tallyRef.current.losses += 1;
   208→          if (ch.result.youS > tallyRef.current.best) {
   209→            tallyRef.current.best = ch.result.youS;
   210→            playSound("milestone"); // new challenge best distance
   211→          }
   212→        }
   213→        const t = tallyRef.current;
   214→        setChallengeHud({
   215→          you: ch.human.st.alive ? ch.human.st.s : ch.human.st.finalS,
   216→          fly: ch.fly.st.alive ? ch.fly.st.s : ch.fly.st.finalS,
   217→          youAlive: ch.human.st.alive,
   218→          flyAlive: ch.fly.st.alive,
   219→          phi: ch.human.st.phi,
   220→          steer: ch.humanSteer,
   221→          flyDop: ch.fly.brain.getDopamine(),
   222→          ended: ch.ended,
   223→          result: ch.result,
   224→          wins: t.wins,
   225→          losses: t.losses,
   226→          best: t.best,
   227→        });
   228→        // the brain panel shows the champion you're racing, not the human
   229→        setLeaderBrain((prev) => (prev === ch.fly.brain ? prev : ch.fly.brain));
   230→      } else {
   231→        setChallengeHud(null);
   232→        const lead = core.leader;
   233→        if (lead) {
   234→          setLeaderBrain((prev) => (prev === lead.brain ? prev : lead.brain));
   235→        }
   236→      }
   237→    }, 160);
   238→    return () => clearInterval(iv);
   239→  }, [core]);
   240→
   241→  // ---- brain-library load integration + session resume banner
   242→  useEffect(() => {
   243→    const tryConsume = () => {
   244→      const snap = useBrainStore.getState().pending.bicycle;
   245→      if (snap) {
   246→        useBrainStore.getState().consumeLoad("bicycle");
   247→        if (core.challenge) {
   248→          // adopting replaces the population — leave the challenge first
   249→          core.endChallenge();
   250→          setChallengeActive(false);
   251→          setChallengeHud(null);
   252→        }
   253→        core.adoptSnapshot(snap);
   254→        setSession(null);
   255→        toast.success(`Loaded "${snap.name}"`, {
   256→          description: `Gen ${snap.generation} champion (${snap.score.toFixed(0)} m) now seeds the whole population.`,
   257→        });
   258→      }
   259→    };
   260→    tryConsume();
   261→    const unsub = useBrainStore.subscribe(tryConsume);
   262→    const sess = BicycleTrainerCore.loadSession();
   263→    if (sess && sess.bestDistance > 1) setSession(sess);
   264→    return () => unsub();
   265→  }, [core]);
   266→
   267→  // ---- keep the R3F subtree out of the HUD re-render loop
   268→  const canvasElement = useMemo(
   269→    () => (
   270→      <Canvas
   271→        dpr={[1, 1.75]}
   272→        camera={{ fov: 55, near: 0.1, far: 2200, position: [0, 2.8, -8] }}
   273→        gl={{ antialias: true, powerPreference: "high-performance" }}
   274→        style={{ position: "absolute", inset: 0 }}
   275→      >
   276→        <DuskScene core={core} retinaCanvas={retinaCanvas} maxRiders={MAX_RIDERS} />
   277→      </Canvas>
   278→    ),
   279→    [core]
   280→  );
   281→
   282→  // ------------------------------------------------------------- controls
   283→  const toggleRun = () => {
   284→    const next = !running;
   285→    setRunning(next);
   286→    core.setRunning(next);
   287→    playSound("click");
   288→  };
   289→
   290→  const doReset = () => {
   291→    core.reset();
   292→    setWatchBest(false);
   293→    setHud(core.hudSnapshot());
   294→    toast.info("Fresh start", { description: "New random brains — Generation 1 rolls out." });
   295→  };
   296→
   297→  const changeTurbo = (t: number) => {
   298→    setTurbo(t);
   299→    core.setTurbo(t);
   300→  };
   301→
   302→  const changePreset = (p: "steady" | "standard" | "frisky") => {
   303→    if (p === preset) return;
   304→    setPresetState(p);
   305→    core.setPreset(p);
   306→    playSound("click");
   307→  };
   308→
   309→  const changePopulation = (v: number[]) => {
   310→    const n = v[0];
   311→    setPopQueued(n);
   312→    core.setQueuedPopulation(n);
   313→  };
   314→
   315→  const changeMutation = (v: number[]) => {
   316→    const s = v[0];
   317→    setMutation(s);
   318→    core.setMutationStrength(s);
   319→  };
   320→
   321→  const toggleWatchBest = (on: boolean) => {
   322→    setWatchBest(on);
   323→    core.setWatchBest(on);
   324→  };
   325→
   326→  // ---------------------------------------------- "You vs the fly" challenge
   327→
   328→  const enterChallenge = () => {
   329→    if (!core.bestBrain) return;
   330→    playSound("click");
   331→    if (watchBest) {
   332→      // leave watch-best first so the parked population comes back intact
   333→      setWatchBest(false);
   334→      core.setWatchBest(false);
   335→    }
   336→    core.startChallenge();
   337→    tallyCountedRef.current = false;
   338→    setChallengeActive(true);
   339→    toast.info("You vs the fly — balance!", {
   340→      description:
   341→        "← / → (or A / D) steer · pedal is automatic. Last one rolling wins.",
   342→    });
   343→  };
   344→
   345→  const rematchChallenge = () => {
   346→    if (!core.challenge) return;
   347→    playSound("click");
   348→    core.startChallenge(); // keeps the parked training population
   349→    tallyCountedRef.current = false;
   350→  };
   351→
   352→  const exitChallenge = () => {
   353→    if (!core.challenge) return;
   354→    playSound("click");
   355→    core.endChallenge(); // folds the champion's learning back in
   356→    setChallengeActive(false);
   357→    setChallengeHud(null);
   358→  };
   359→
   360→  // --- challenge keyboard: window listeners active ONLY while challenging ---
   361→  useEffect(() => {
   362→    if (!challengeActive) return;
   363→    const isEditable = (t: EventTarget | null): boolean => {
   364→      const el = t as HTMLElement | null;
   365→      return (
   366→        !!el &&
   367→        typeof el.tagName === "string" &&
   368→        (el.tagName === "INPUT" ||
   369→          el.tagName === "TEXTAREA" ||
   370→          el.isContentEditable === true)
   371→      );
   372→    };
   373→    const apply = (key: string, down: boolean) => {
   374→      const ch = core.challenge;
   375→      if (!ch || ch.ended) return;
   376→      if (key === "ArrowLeft" || key === "a" || key === "A") ch.keys.left = down;
   377→      else if (key === "ArrowRight" || key === "d" || key === "D")
   378→        ch.keys.right = down;
   379→    };
   380→    const down = (e: KeyboardEvent) => {
   381→      if (isEditable(e.target)) return; // never steal typing
   382→      if (e.key === "ArrowLeft" || e.key === "ArrowRight") {
   383→        e.preventDefault(); // stop page scroll (held keys → repeat is fine)
   384→      }
   385→      apply(e.key, true);
   386→    };
   387→    const up = (e: KeyboardEvent) => apply(e.key, false);
   388→    const blur = () => {
   389→      // alt-tab mid-steer would otherwise stick the key down forever
   390→      const ch = core.challenge;
   391→      if (ch) {
   392→        ch.keys.left = false;
   393→        ch.keys.right = false;
   394→      }
   395→    };
   396→    window.addEventListener("keydown", down);
   397→    window.addEventListener("keyup", up);
   398→    window.addEventListener("blur", blur);
   399→    return () => {
   400→      window.removeEventListener("keydown", down);
   401→      window.removeEventListener("keyup", up);
   402→      window.removeEventListener("blur", blur);
   403→    };
   404→  }, [challengeActive, core]);
   405→
   406→  const resumeSession = () => {
   407→    if (!session) return;
   408→    playSound("click");
   409→    if (core.challenge) {
   410→      // resuming replaces the population — leave the challenge first
   411→      core.endChallenge();
   412→      setChallengeActive(false);
   413→      setChallengeHud(null);
   414→    }
   415→    core.adoptSnapshot(session.snapshot);
   416→    core.history = session.history;
   417→    core.generation = session.generation;
   418→    setSession(null);
   419→    setHud(core.hudSnapshot());
   420→    toast.success("Session resumed", {
   421→      description: `Continuing from generation ${session.generation} (best ${session.bestDistance.toFixed(0)} m).`,
   422→    });
   423→  };
   424→
   425→  const saveBest = async () => {
   426→    const brain = core.bestBrain ?? core.leader?.brain;
   427→    if (!brain) {
   428→      toast.error("Nothing to save yet", { description: "Let at least one generation finish." });
   429→      return;
   430→    }
   431→    const name = brainName.trim() || "Dusk Rider";
   432→    const score = core.bestEverDistance;
   433→    setSaving(true);
   434→    try {
   435→      const snapshot = brain.toJSON("bicycle", name, core.generation, score);
   436→      const res = await fetch("/api/brains", {
   437→        method: "POST",
   438→        headers: { "Content-Type": "application/json" },
   439→        body: JSON.stringify({
   440→          name,
   441→          task: "bicycle",
   442→          generation: core.generation,
   443→          score,
   444→          snapshot,
   445→        }),
   446→      });
   447→      if (!res.ok) {
   448→        const detail = await res.text().catch(() => "");
   449→        throw new Error(detail || `HTTP ${res.status}`);
   450→      }
   451→      toast.success(`Saved "${name}"`, {
   452→        description: `Generation ${core.generation} champion · ${score.toFixed(0)} m — find it in the Brain Library.`,
   453→      });
   454→      playSound("ding");
   455→    } catch (err) {
   456→      const msg = err instanceof Error ? err.message : String(err);
   457→      toast.error("Save failed", {
   458→        description: msg.includes("404")
   459→          ? "The brain library API is not available yet — try again in a moment."
   460→          : msg,
   461→      });
   462→    } finally {
   463→      setSaving(false);
   464→    }
   465→  };
   466→
   467→  // ----------------------------------------------------------------- HUD
   468→  const alive = hud?.alive ?? 0;
   469→  const total = hud?.total ?? 5;
   470→  const gen = hud?.gen ?? 1;
   471→  const bestEver = hud?.bestEver ?? 0;
   472→  const leaderS = hud?.leaderS ?? 0;
   473→  const leaderV = hud?.leaderV ?? 0;
   474→  const leaderPhi = hud?.leaderPhi ?? 0;
   475→  const dop = hud?.leaderDop ?? 0;
   476→  const dopPct = Math.min(1, Math.abs(dop));
   477→  const events = hud?.events ?? [];
   478→  const history = hud?.history ?? [];
   479→  const popApplied = (hud?.popSize ?? popQueued) === popQueued;
   480→
   481→  return (
   482→    <section className="flex flex-col gap-4">
   483→      <Toaster theme="dark" position="bottom-right" closeButton />
   484→
   485→      {/* ------------------------------------------------------- header */}
   486→      <Card className="border-border/70 bg-gradient-to-br from-[#2a1224] via-card to-[#241a20]">
   487→        <CardHeader className="pb-3">
   488→          <div className="flex flex-wrap items-center gap-3">
   489→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-rose-500 to-amber-500 text-white shadow-lg shadow-rose-900/40">
   490→              <Bike className="h-5 w-5" />
   491→            </div>
   492→            <div className="min-w-0 flex-1">
   493→              <CardTitle className="text-lg">Bicycle Training</CardTitle>
   494→              <CardDescription className="mt-0.5 text-sm">
   495→                Each fly rides the same dusk road. Distance = sugar. Falling or
   496→                leaving the road = shock. The best riders breed, generation by
   497→                generation.
   498→              </CardDescription>
   499→            </div>
   500→            <div className="flex items-center gap-2">
   501→              <Badge variant="secondary" className="gap-1.5 border-amber-500/30 bg-amber-500/10 text-amber-300">
   502→                <Sparkles className="h-3 w-3" /> reward +0.03 / m
   503→              </Badge>
   504→              <Badge variant="secondary" className="gap-1.5 border-rose-500/30 bg-rose-500/10 text-rose-300">
   505→                <Zap className="h-3 w-3" /> shock −0.5 on fall
   506→              </Badge>
   507→            </div>
   508→          </div>
   509→        </CardHeader>
   510→      </Card>
   511→
   512→      {/* ------------------------------------------------ resume banner */}
   513→      {session && (
   514→        <div className="flex flex-wrap items-center gap-3 rounded-xl border border-amber-500/40 bg-amber-500/10 px-4 py-3 text-sm text-amber-200">
   515→          <History className="h-4 w-4 shrink-0" />
   516→          <span className="min-w-0 flex-1">
   517→            Previous dusk ride found — generation {session.generation}, best{" "}
   518→            {session.bestDistance.toFixed(0)} m. Continue breeding from that champion?
   519→          </span>
   520→          <Button
   521→            size="sm"
   522→            onClick={resumeSession}
   523→            className="h-9 bg-amber-500 text-black hover:bg-amber-400"
   524→          >
   525→            Resume last session
   526→          </Button>
   527→          <button
   528→            onClick={() => setSession(null)}
   529→            className="rounded-md p-1.5 opacity-60 hover:opacity-100"
   530→            aria-label="Dismiss session banner"
   531→          >
   532→            <X className="h-4 w-4" />
   533→          </button>
   534→        </div>
   535→      )}
   536→
   537→      {/* ------------------------------------------------- main + side */}
   538→      <div className="flex flex-col gap-4 lg:flex-row">
   539→        {/* 3D canvas card */}
   540→        <Card className="min-w-0 flex-1 overflow-hidden p-0">
   541→          <div className="relative aspect-video min-h-[420px] w-full bg-[#150a1e]">
   542→            {canvasElement}
   543→
   544→            {/* HUD overlay */}
   545→            <div className="pointer-events-none absolute inset-0 select-none">
   546→              <div
   547→                className="absolute left-3 top-3 flex flex-wrap items-center gap-2"
