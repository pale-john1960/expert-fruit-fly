     3→/**
     4→ * DinoTrainer — a population of FlyBrains learns to play the Chrome Dino
     5→ * runner, generation by generation.
     6→ *
     7→ * How it works:
     8→ *  - every 60Hz sim step, the shared world (obstacles, ground, sky — NO
     9→ *    flies) is drawn to an offscreen canvas, downscaled to 240×90 and
    10→ *    encoded into a 24×9 retina frame shared by ALL flies (they can't see
    11→ *    themselves — pure reactive vision)
    12→ *  - each alive fly steps its brain: motor 0 > 0.12 → jump, motor 1 > 0.15 → duck
    13→ *  - clearing an obstacle injects +0.4 dopamine (sugar); crashing injects
    14→ *    −0.3 (shock) and kills the fly
    15→ *  - when every fly is dead (or the 3-minute cap hits), the generation
    16→ *    ends: top-2 brains are cloned (elitism), the rest are bred via
    17→ *    tournament-of-5 crossover + mutation → next generation, automatically
    18→ *  - cross-component brain loading via useBrainStore, session autosave via
    19→ *    localStorage, best-brain export to the library via POST /api/brains
    20→ */

   108→interface FeedEvent {

   116→interface SimRunner {
   117→  brain: FlyBrain;
   118→  fly: ReturnType<typeof freshFly>;

   124→interface Sim {
   125→  world: ReturnType<typeof createWorld>;
   126→  runners: SimRunner[];

   140→interface Art {

   147→const STEP_DT = 1 / 60;

   151→// ---------------------------------------------------------------------------
   152→// Simulation internals (module level — zero React re-renders in the loop)
   153→// ---------------------------------------------------------------------------
   154→function createArt(): Art {
   155→  const game = document.createElement("canvas");
   156→  game.width = GAME_W;
   157→  game.height = GAME_H;
   158→  const gameCtx = game.getContext("2d");
   159→  const retina = document.createElement("canvas");
   160→  retina.width = RETINA_W;
   161→  retina.height = RETINA_H;
   162→  const retinaCtx = retina.getContext("2d", { willReadFrequently: true });
   163→  if (!gameCtx || !retinaCtx) {
   164→    throw new Error("2D canvas unavailable");
   165→  }
   166→  return { game, gameCtx, retina, retinaCtx };
   167→}
   168→
   169→function makeRunner(brain: FlyBrain, idx: number): SimRunner {
   170→  return { brain, fly: freshFly(), alive: true, deathScore: 0, idx };
   171→}
   172→
   173→function pushEvent(
   174→  sim: Sim,
   175→  label: string,
   176→  value: number,
   177→  kind: FeedEvent["kind"],
   178→  t: number
   179→): void {
   180→  sim.eventId += 1;
   181→  sim.events.push({ id: sim.eventId, label, value, kind, t });
   182→  if (sim.events.length > 40) sim.events.splice(0, sim.events.length - 40);
   183→}
   184→
   185→function bestAliveBrain(sim: Sim): FlyBrain | null {
   186→  let best: FlyBrain | null = null;
   187→  let bestR = -Infinity;
   188→  for (const r of sim.runners) {
   189→    if (r.alive && r.brain.rewards > bestR) {
   190→      bestR = r.brain.rewards;
   191→      best = r.brain;
   192→    }
   193→  }
   194→  return best ?? sim.runners[0]?.brain ?? null;
   195→}
   196→
   197→function autosave(sim: Sim): void {
   198→  try {
   199→    // snapshot.generation = session's latest gen, so "Resume" continues
   200→    // numbering where the session left off (population = clones of best brain)
   201→    const snap = sim.bestBrain
   202→      ? sim.bestBrain.toJSON("dino", "session best", sim.generation, sim.bestEver)
   203→      : null;
   204→    localStorage.setItem(
   205→      AUTOSAVE_KEY,
   206→      JSON.stringify({
   207→        v: 1,
   208→        gen: sim.generation,
   209→        bestScore: sim.bestEver,
   210→        history: sim.history.slice(-200),
   211→        snapshot: snap,
   212→      })
   213→    );
   214→  } catch {
   215→    // storage blocked/full — autosave is best-effort
   216→  }
   217→}
   218→
   219→function endGeneration(sim: Sim, popSize: number, mutStrength: number): void {
   220→  for (const r of sim.runners) if (r.alive) r.deathScore = sim.world.score;
   221→  const scores = sim.runners.map((r) => r.deathScore);
   222→  const best = Math.max(...scores);
   223→  const avg = scores.reduce((a, b) => a + b, 0) / Math.max(1, scores.length);
   224→  sim.history.push({ gen: sim.generation, best, avg });
   225→
   226→  let bi = 0;
   227→  for (let i = 1; i < scores.length; i++) if (scores[i] > scores[bi]) bi = i;
   228→  if (best > sim.bestEver) {
   229→    sim.bestEver = best;
   230→    sim.bestGen = sim.generation;
   231→    sim.bestBrain = sim.runners[bi].brain.clone();
   232→  }
   233→  pushEvent(
   234→    sim,
   235→    `Generation ${sim.generation} ended — best ${best}, avg ${avg.toFixed(1)}`,
   236→    0,
   237→    "info",
   238→    sim.world.t
   239→  );
   240→  autosave(sim);
   241→
   242→  const brains = evolvePopulation(
   243→    sim.runners.map((r) => r.brain),
   244→    scores,
   245→    popSize,
   246→    mutStrength
   247→  );
   248→  sim.generation += 1;
   249→  sim.runners = brains.map((b, i) => makeRunner(b, i));
   250→  sim.world = createWorld();
   251→}

   253→/** One 60Hz simulation step: world → vision → brains → action → collisions. */
   254→function stepSim(
   255→  sim: Sim,
   256→  art: Art,
   257→  dt: number,
   258→  popSize: number,
   259→  mutStrength: number
   260→): void {
   261→  const w = sim.world;
   262→  const cleared = stepWorld(w, dt);
   263→
   264→  // fly avatar physics
   265→  for (const r of sim.runners) {
   266→    const f = r.fly;
   267→    f.wingPhase += dt * 36;
   268→    if (f.jumpCooldownMs > 0) {
   269→      f.jumpCooldownMs = Math.max(0, f.jumpCooldownMs - dt * 1000);
   270→    }
   271→    if (f.airborne) {
   272→      // duck input while airborne = fast fall, like the real game
   273→      f.vy += GRAVITY * (f.ducking ? 2.6 : 1) * dt;
   274→      f.y += f.vy * dt;
   275→      if (f.y >= GROUND_Y) {
   276→        f.y = GROUND_Y;
   277→        f.vy = 0;
   278→        f.airborne = false;
   279→      }
   280→    }
   281→    f.ducking = false; // re-decided by the brain every step
   282→  }
   283→
   284→  // vision: world without flies → 240×90 → 24×9 retina (shared input)
   285→  drawWorld(art.gameCtx, w);
   286→  art.retinaCtx.drawImage(art.game, 0, 0, RETINA_W, RETINA_H);
   287→  const img = art.retinaCtx.getImageData(0, 0, RETINA_W, RETINA_H);
   288→  const input = retinaFromImageData(
   289→    img.data,
   290→    RETINA_W,
   291→    RETINA_H,
   292→    RETINA_COLS,
   293→    RETINA_ROWS,
   294→    false
   295→  );
   296→
   297→  // brains: dopamine rides along with the regular step
   298→  for (const r of sim.runners) {
   299→    if (!r.alive) continue;
   300→    const delta = cleared.length > 0 ? REWARD_CLEAR : 0;
   301→    const m = r.brain.step(input, delta);
   302→    const f = r.fly;
   303→    if (!f.airborne && f.jumpCooldownMs <= 0 && m[0] > JUMP_THRESHOLD) {
   304→      f.vy = -JUMP_V;
   305→      f.airborne = true;
   306→      f.jumpCooldownMs = JUMP_COOLDOWN_MS;
   307→    }
   308→    if (m[1] > DUCK_THRESHOLD) f.ducking = true;
   309→  }
   310→  if (cleared.length > 0) {
   311→    pushEvent(sim, "Obstacle cleared", REWARD_CLEAR, "reward", w.t);
   312→  }
   313→
   314→  // collisions → death shock
   315→  for (const r of sim.runners) {
   316→    if (!r.alive) continue;
   317→    if (collides(w, r.fly)) {
   318→      r.alive = false;
   319→      r.deathScore = w.score;
   320→      r.brain.step(input, -PUNISH_CRASH);
   321→      pushEvent(sim, `Fly #${r.idx + 1} crashed`, -PUNISH_CRASH, "punish", w.t);
   322→    }
   323→  }
   324→
   325→  // generation end (or watch-mode replay restart)
   326→  const anyAlive = sim.runners.some((r) => r.alive);
   327→  if (!anyAlive || (!sim.watchMode && w.t >= GEN_TIME_CAP_S)) {
   328→    if (sim.watchMode) {
   329→      sim.watchCooldown = (sim.watchCooldown ?? 45) - 1;
   330→      if (sim.watchCooldown <= 0) {
   331→        sim.watchCooldown = null;
   332→        if (sim.bestBrain) {
   333→          sim.runners = [makeRunner(sim.bestBrain.clone(), 0)];
   334→        }
   335→        sim.world = createWorld();
   336→      }
   337→    } else {
   338→      endGeneration(sim, popSize, mutStrength);
   339→    }
   340→  }
   341→}
   342→
   343→function renderVisible(
   344→  canvas: HTMLCanvasElement,
   345→  game: HTMLCanvasElement,
   346→  sim: Sim,
   347→  tNow: number
   348→): void {
   349→  const ctx = canvas.getContext("2d");
   350→  if (!ctx) return;
   351→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   352→  ctx.clearRect(0, 0, canvas.width, canvas.height);
   353→  if (canvas.width > 0 && canvas.height > 0) {
   354→    ctx.imageSmoothingEnabled = true;
   355→    ctx.drawImage(game, 0, 0, GAME_W, GAME_H, 0, 0, canvas.width, canvas.height);
   356→  }
   357→  const sx = canvas.width / GAME_W;
   358→  const sy = canvas.height / GAME_H;
   359→  ctx.setTransform(sx, 0, 0, sy, 0, 0);
   360→
   361→  const leader = bestAliveBrain(sim);
   362→  for (const r of sim.runners) {
   363→    if (!r.alive) continue;
   364→    const isBest = sim.watchMode ? true : leader === r.brain;
   365→    drawFly(ctx, {
   366→      x: FLY_X,
   367→      feetY: r.fly.y,
   368→      airborne: r.fly.airborne,
   369→      ducking: r.fly.ducking,
   370→      wingPhase: r.fly.wingPhase,
   371→      alpha: isBest ? 1 : 0.38,
   372→      isBest,
   373→      label: sim.watchMode ? undefined : "BEST",
   374→      t: tNow,
   375→    });
   376→  }
   377→}

   382→export function DinoTrainer() {
   383→  // --- UI state -------------------------------------------------------------
   384→  const [running, setRunning] = useState(false);
   385→  const [turbo, setTurbo] = useState(1);
   386→  const [popSize, setPopSize] = useState(6);
   387→  const [mutStrength, setMutStrength] = useState(0.3);
   388→  const [watchBest, setWatchBest] = useState(false);
   389→  const [brainName, setBrainName] = useState("");
   390→  const [saving, setSaving] = useState(false);
   391→  const [hud, setHud] = useState({
   392→    gen: 1,
   393→    alive: 6,
   394→    total: 6,
   395→    score: 0,
   396→    best: 0,
   397→    speed: BASE_SPEED,
   398→    watch: false,
   399→  });
   400→  const [feed, setFeed] = useState<FeedEvent[]>([]);
   401→  const [history, setHistory] = useState<GenStat[]>([]);
   402→  const [panelBrain, setPanelBrain] = useState<FlyBrain | null>(null);
   403→  const [hasBest, setHasBest] = useState(false);
   404→  const [resume, setResume] = useState<{
   405→    gen: number;
   406→    best: number;
   407→    snap: BrainSnapshot;
   408→    history: GenStat[];
   409→  } | null>(null);
   410→
   411→  // --- refs (read inside the rAF loop without restarting it) ----------------
   412→  const runningRef = useRef(running);
   413→  const turboRef = useRef(turbo);
   414→  const popSizeRef = useRef(popSize);
   415→  const mutStrengthRef = useRef(mutStrength);
   416→  const adoptedRef = useRef(false);
   417→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   418→  const simRef = useRef<Sim | null>(null);
   419→  const artRef = useRef<Art | null>(null);
   420→
   421→  useEffect(() => {
   422→    runningRef.current = running;
   423→  }, [running]);
   424→  useEffect(() => {
   425→    turboRef.current = turbo;
   426→  }, [turbo]);
   427→  useEffect(() => {
   428→    popSizeRef.current = popSize;
   429→  }, [popSize]);
   430→  useEffect(() => {
   431→    mutStrengthRef.current = mutStrength;
   432→  }, [mutStrength]);
   433→
   434→  /** Push sim state into React at ~4 Hz (keeps 60fps rendering pure canvas). */
   435→  const flush = useCallback((s: Sim) => {
   436→    setHud({
   437→      gen: s.generation,
   438→      alive: s.runners.filter((r) => r.alive).length,
   439→      total: s.runners.length,
   440→      score: s.world.score,
   441→      best: s.bestEver,
   442→      speed: s.world.speed,
   443→      watch: s.watchMode,
   444→    });
   445→    setFeed(s.events.slice(-8).reverse());
   446→    setHistory(s.history.slice());
   447→    setHasBest(Boolean(s.bestBrain));
   448→    setPanelBrain((prev) => {
   449→      const next = s.watchMode
   450→        ? (s.runners[0]?.brain ?? null)
   451→        : bestAliveBrain(s);
   452→      return next === prev ? prev : next;
   453→    });
   454→  }, []);
   455→
   456→  // --- mount: build the sim + the animation loop ----------------------------
   457→  useEffect(() => {
   458→    const art = createArt();
   459→    artRef.current = art;
   460→    const sim: Sim = {
   461→      world: createWorld(),
   462→      runners: Array.from({ length: popSizeRef.current }, (_, i) =>
   463→        makeRunner(
   464→          new FlyBrain({ ...DEFAULT_ARCH_DINO, seed: (Math.random() * 0x7fffffff) | 0 }),
   465→          i
   466→        )
   467→      ),
   468→      savedRunners: null,
   469→      watchMode: false,
   470→      watchCooldown: null,
   471→      generation: 1,
   472→      bestEver: 0,
   473→      bestGen: 0,
   474→      bestBrain: null,
   475→      history: [],
   476→      events: [],
   477→      eventId: 0,
   478→    };
   479→    simRef.current = sim;
   480→    drawWorld(art.gameCtx, sim.world); // first paint so pause view isn't blank
   481→    setPanelBrain(sim.runners[0]?.brain ?? null);
   482→
   483→    let raf = 0;
   484→    let last = performance.now();
   485→    let acc = 0;
   486→    let lastFlush = 0;
   487→    const frame = (ts: number) => {
   488→      raf = requestAnimationFrame(frame);
   489→      const dt = Math.min((ts - last) / 1000, 0.25);
   490→      last = ts;
   491→      const s = simRef.current;
   492→      if (!s) return;
   493→      if (runningRef.current) {
   494→        acc += dt * turboRef.current;
   495→        let n = 0;
   496→        while (acc >= STEP_DT && n < MAX_STEPS_PER_FRAME) {
   497→          stepSim(s, art, STEP_DT, popSizeRef.current, mutStrengthRef.current);
   498→          acc -= STEP_DT;
   499→          n += 1;
   500→        }
   501→        if (acc > 0.2) acc = 0.2; // never spiral after a long stall
   502→      }
   503→      const canvas = canvasRef.current;
   504→      if (canvas) renderVisible(canvas, art.game, s, ts / 1000);
   505→      if (ts - lastFlush >= 250) {
   506→        lastFlush = ts;
   507→        flush(s);
   508→      }
   509→    };
   510→    raf = requestAnimationFrame(frame);
   511→    return () => cancelAnimationFrame(raf);
   512→  }, [flush]);
   513→
   514→  // --- responsive canvas backing store --------------------------------------
   515→  useEffect(() => {
   516→    const canvas = canvasRef.current;
   517→    const wrap = canvas?.parentElement;
   518→    if (!canvas || !wrap) return;
   519→    const apply = () => {
   520→      const w = wrap.clientWidth;
   521→      if (w <= 0) return;
   522→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   523→      const cw = Math.round(w * dpr);
   524→      const ch = Math.round((w * dpr * GAME_H) / GAME_W);
   525→      if (canvas.width !== cw || canvas.height !== ch) {
   526→        canvas.width = cw;
   527→        canvas.height = ch;
   528→      }
   529→    };
   530→    apply();
   531→    const ro = new ResizeObserver(apply);
   532→    ro.observe(wrap);
   533→    return () => ro.disconnect();
   534→  }, []);
   535→
   536→  // --- adopt a snapshot (library load / session resume) ---------------------
   537→  const adoptSnapshot = useCallback(
   538→    (snap: BrainSnapshot, restoreHistory?: GenStat[]) => {
   539→      adoptedRef.current = true;
   540→      setResume(null);
   541→      const s = simRef.current;
   542→      if (!s) return;
   543→      if (!snap.arch || snap.arch.motorCount < 2) {
   544→        toast.error("This snapshot is not compatible with the Dino task.");
   545→        return;
   546→      }
   547→      const base = FlyBrain.fromJSON(snap);
   548→      s.watchMode = false;
   549→      s.savedRunners = null;
   550→      s.watchCooldown = null;
   551→      s.generation = Math.max(1, snap.generation || 1);
   552→      s.bestEver = snap.score ?? 0;
   553→      s.bestGen = snap.generation || 1;
   554→      s.bestBrain = base.clone();
   555→      s.history = restoreHistory ?? [];
   556→      s.events = [];
   557→      s.eventId = 0;
   558→      const n = popSizeRef.current;
   559→      s.runners = Array.from({ length: n }, (_, i) => makeRunner(base.clone(), i));
   560→      s.world = createWorld();
   561→      setWatchBest(false);
   562→      setRunning(true);
   563→      runningRef.current = true;
   564→      flush(s);
   565→      toast.success(
   566→        `Loaded "${snap.name}" (gen ${snap.generation}, score ${snap.score}) — population replaced with its clones`
   567→      );
   568→    },
   569→    [flush]
   570→  );
   571→
   572→  // --- cross-tab brain loads (Brain Library "Load" button) ------------------
   573→  const pending = useBrainStore((s) => s.pending);
   574→  useEffect(() => {
   575→    const snap = useBrainStore.getState().consumeLoad("dino");
   576→    if (snap) adoptSnapshot(snap);
   577→  }, [pending, adoptSnapshot]);
   578→
   579→  // --- offer to resume the last session -------------------------------------
   580→  useEffect(() => {
   581→    if (adoptedRef.current) return;
   582→    try {
   583→      const raw = localStorage.getItem(AUTOSAVE_KEY);
   584→      if (!raw) return;
   585→      const parsed = JSON.parse(raw) as {
   586→        gen: number;
