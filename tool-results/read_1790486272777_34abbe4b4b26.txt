     3→/**
     4→ * DinoTrainer — a population of FlyBrains learns to play the Chrome Dino
     5→ * runner, generation by generation.
     6→ *
     7→ * How it works:
     8→ *  - every 60Hz sim step, the shared world (obstacles, ground, sky — NO
     9→ *    flies) is drawn to an offscreen canvas, downscaled to 240×90 and
    10→ *    encoded into a 24×9 retina frame shared by ALL flies (they can't see
    11→ *    themselves — pure reactive vision)
    12→ *  - each alive fly steps its brain: motor 0 > 0.12 → jump, motor 1 > 0.15 → duck
    13→ *  - clearing an obstacle injects +0.4 dopamine (sugar); crashing injects
    14→ *    −0.3 (shock) and kills the fly
    15→ *  - when every fly is dead (or the 3-minute cap hits), the generation
    16→ *    ends: top-2 brains are cloned (elitism), the rest are bred via
    17→ *    tournament-of-5 crossover + mutation → next generation, automatically
    18→ *  - cross-component brain loading via useBrainStore, session autosave via
    19→ *    localStorage, best-brain export to the library via POST /api/brains
    20→ *  - duck-defense metrics ("Pterodactyl report"): every run tracks birds
    21→ *    seen vs cleared + a death cause (bird/cactus/timeout), aggregated into
    22→ *    a session progress bar; "bird practice" spawns pterodactyls from
    23→ *    score 0 so ducking can be trained deliberately
    24→ *  - sound effects via @/lib/sound (milestone on new HI, gen on evolve,
    25→ *    crash/jump at real-time speed, ding on save, click on start/resume)
    26→ *  - "You vs the fly" duel mode: the human plays a dino avatar (arrow keys)
    27→ *    head-to-head against the champion brain in the SAME world — identical
    28→ *    obstacles for both, so it's a fair race. Each avatar collides
    29→ *    independently; once one dies the survivor gets 30s to run up its
    30→ *    score. The champion keeps receiving its usual sugar (+0.4 per clear)
    31→ *    and shock (−0.3 on crash) during duels — extra lifetime learning —
    32→ *    and its trained weights are written back into the trainer's champion
    33→ *    when the duel ends, so rematches compound. Training is only PARKED
    34→ *    while dueling (same generation, same world on return).
    35→ */

   266→function createArt(): Art {

   268→  game.width = GAME_W;

   270→  const gameCtx = game.getContext("2d");

   273→  retina.height = RETINA_H;

   276→    throw new Error("2D canvas unavailable");

   278→  return { game, gameCtx, retina, retinaCtx };

   312→function bestAliveBrain(sim: Sim): FlyBrain | null {
   313→  let best: FlyBrain | null = null;
   314→  let bestR = -Infinity;
   315→  for (const r of sim.runners) {
   316→    if (r.alive && r.brain.rewards > bestR) {
   317→      bestR = r.brain.rewards;
   318→      best = r.brain;
   319→    }
   320→  }
   321→  return best ?? sim.runners[0]?.brain ?? null;
   322→}

   324→function autosave(sim: Sim): void {
   325→  try {
   326→    // snapshot.generation = session's latest gen, so "Resume" continues
   327→    // numbering where the session left off (population = clones of best brain)
   328→    const snap = sim.bestBrain
   329→      ? sim.bestBrain.toJSON("dino", "session best", sim.generation, sim.bestEver)
   330→      : null;
   331→    localStorage.setItem(
   332→      AUTOSAVE_KEY,
   333→      JSON.stringify({
   334→        v: 1,
   335→        gen: sim.generation,
   336→        bestScore: sim.bestEver,
   337→        history: sim.history.slice(-200),
   338→        snapshot: snap,
   339→      })
   340→    );
   341→  } catch {
   342→    // storage blocked/full — autosave is best-effort
   343→  }
   344→}

   346→function endGeneration(sim: Sim, popSize: number, mutStrength: number): void {
   347→  for (const r of sim.runners) {
   348→    if (r.alive) {
   349→      r.deathScore = sim.world.score;
   350→      r.deathCause = "timeout";
   351→      sim.genTimeoutDeaths += 1;
   352→    }
   353→  }
   354→  const scores = sim.runners.map((r) => r.deathScore);
   355→  const best = Math.max(...scores);
   356→  const avg = scores.reduce((a, b) => a + b, 0) / Math.max(1, scores.length);
   357→  sim.history.push({ gen: sim.generation, best, avg });
   358→
   359→  // fold this generation's per-run bird encounters into the session totals
   360→  const genBirdsSeen = sim.runners.reduce((a, r) => a + birdsSeenCount(r.duck), 0);
   361→  const genBirdsCleared = sim.runners.reduce((a, r) => a + birdsClearedCount(r.duck), 0);
   362→  sim.sessionBirdsSeen += genBirdsSeen;
   363→  sim.sessionBirdsCleared += genBirdsCleared;
   364→
   365→  let bi = 0;
   366→  for (let i = 1; i < scores.length; i++) if (scores[i] > scores[bi]) bi = i;
   367→  if (best > sim.bestEver) {
   368→    sim.bestEver = best;
   369→    sim.bestGen = sim.generation;
   370→    sim.bestBrain = sim.runners[bi].brain.clone();
   371→    playSound("milestone"); // new all-time HI score
   372→  }
   373→  const duckNote =
   374→    genBirdsSeen > 0
   375→      ? ` · duck ${Math.round((100 * genBirdsCleared) / genBirdsSeen)}%`
   376→      : "";
   377→  pushEvent(
   378→    sim,
   379→    `Generation ${sim.generation} ended — best ${best}, avg ${avg.toFixed(1)}${duckNote}`,
   380→    0,
   381→    "info",
   382→    sim.world.t
   383→  );
   384→  autosave(sim);
   385→  playSound("gen"); // generation complete → evolve
   386→
   387→  const brains = evolvePopulation(
   388→    sim.runners.map((r) => r.brain),
   389→    scores,
   390→    popSize,
   391→    mutStrength
   392→  );
   393→  sim.generation += 1;
   394→  sim.runners = brains.map((b, i) => makeRunner(b, i));
   395→  sim.genBirdDeaths = 0;
   396→  sim.genCactusDeaths = 0;
   397→  sim.genTimeoutDeaths = 0;
   398→  sim.world = freshWorld(sim);
   399→}

   404→function createDuel(sim: Sim): DuelSim {
   405→  const w = createWorld();
   406→  w.birdPractice = sim.birdPractice; // duel world behaves like the trainer's
   407→  return {
   408→    world: w,
   409→    human: { fly: freshFly(), alive: true, deathScore: 0 },
   410→    ai: {
   411→      brain: sim.bestBrain ? sim.bestBrain.clone() : new FlyBrain(DEFAULT_ARCH_DINO),
   412→      fly: freshFly(),
   413→      alive: true,
   414→      deathScore: 0,
   415→    },
   416→    duckHeld: false,
   417→    jumpQueued: false,
   418→    firstDeathT: null,
   419→    ended: false,
   420→    result: null,
   421→  };
   422→}

   424→/** Both avatars are done — tally the session, fold the champion's lifetime
   425→ *  dopamine back into the trainer's best brain, freeze the result. */
   426→function finishDuel(d: DuelSim, s: Sim): void {
   427→  if (d.result) return;
   428→  const youScore = d.human.deathScore;
   429→  const flyScore = d.ai.deathScore;
   430→  const winner: DuelResult["winner"] =
   431→    youScore > flyScore ? "you" : youScore < flyScore ? "fly" : "tie";
   432→  if (winner === "you") s.duelWins += 1;
   433→  else if (winner === "fly") s.duelLosses += 1;
   434→  const newBest = youScore > s.duelBest;
   435→  if (newBest) s.duelBest = youScore;
   436→  // duels double as extra lifetime learning — keep the trained weights
   437→  s.bestBrain = d.ai.brain.clone();
   438→  d.result = { winner, youScore, flyScore };
   439→  d.ended = true;
   440→  if (newBest) playSound("milestone"); // new best duel score by the human
   441→}

   443→/** One 60Hz duel step: shared world → vision → human keys + champion brain →
   444→ *  physics → per-avatar collisions → end conditions. */
   445→function stepDuel(d: DuelSim, art: Art, s: Sim, dt: number): void {
   446→  const w = d.world;
   447→  const cleared = stepWorld(w, dt);
   448→
   449→  // shared avatar physics (same helper as the training runners)
   450→  stepAvatarPhysics(d.human.fly, dt);
   451→  stepAvatarPhysics(d.ai.fly, dt);
   452→
   453→  // human input: duck is held, jump is edge-triggered (ignores repeats)
   454→  d.human.fly.ducking = d.duckHeld;
   455→  if (d.jumpQueued) {
   456→    d.jumpQueued = false;
   457→    const f = d.human.fly;
   458→    if (!f.airborne && f.jumpCooldownMs <= 0) {
   459→      f.vy = -JUMP_V;
   460→      f.airborne = true;
   461→      f.jumpCooldownMs = JUMP_COOLDOWN_MS;
   462→      playSound("jump", 0.15);
   463→    }
   464→  }
   465→  // the champion re-decides duck every step, exactly like a training runner
   466→  d.ai.fly.ducking = false;
   467→
   468→  // vision: world WITHOUT avatars → shared retina (the fly can't see you)
   469→  drawWorld(art.gameCtx, w);
   470→  art.retinaCtx.drawImage(art.game, 0, 0, RETINA_W, RETINA_H);
   471→  const img = art.retinaCtx.getImageData(0, 0, RETINA_W, RETINA_H);
   472→  const input = retinaFromImageData(
   473→    img.data,
   474→    RETINA_W,
   475→    RETINA_H,
   476→    RETINA_COLS,
   477→    RETINA_ROWS,
   478→    false
   479→  );
   480→
   481→  // champion brain — usual dopamine during duels (obstacle clear +0.4,
   482→  // crash −0.3 further down), so every duel is extra lifetime learning
   483→  if (d.ai.alive) {
   484→    const delta = cleared.length > 0 ? REWARD_CLEAR : 0;
   485→    const m = d.ai.brain.step(input, delta);
   486→    const f = d.ai.fly;
   487→    if (!f.airborne && f.jumpCooldownMs <= 0 && m[0] > JUMP_THRESHOLD) {
   488→      f.vy = -JUMP_V;
   489→      f.airborne = true;
   490→      f.jumpCooldownMs = JUMP_COOLDOWN_MS;
   491→    }
   492→    if (m[1] > DUCK_THRESHOLD) f.ducking = true;
   493→  }
   494→
   495→  // independent collisions — each avatar dies on its own
   496→  if (d.human.alive && hitObstacle(w, d.human.fly)) {
   497→    d.human.alive = false;
   498→    d.human.deathScore = w.score;
   499→    if (d.firstDeathT === null) d.firstDeathT = w.t;
   500→    playSound("crash", 0.25);
   501→  }
   502→  if (d.ai.alive && hitObstacle(w, d.ai.fly)) {
   503→    d.ai.alive = false;
   504→    d.ai.deathScore = w.score;
   505→    if (d.firstDeathT === null) d.firstDeathT = w.t;
   506→    d.ai.brain.step(input, -PUNISH_CRASH); // death shock, same as training
   507→    playSound("crash", 0.25);
   508→  }
   509→
   510→  // end conditions: both dead, or the survivor's 30s cap is up
   511→  if (!d.human.alive && !d.ai.alive) {
   512→    finishDuel(d, s);
   513→  } else if (
   514→    d.firstDeathT !== null &&
   515→    w.t - d.firstDeathT >= DUEL_SURVIVOR_CAP_S
   516→  ) {
   517→    if (d.human.alive) {
   518→      d.human.alive = false;
   519→      d.human.deathScore = w.score;
   520→    }
   521→    if (d.ai.alive) {
   522→      d.ai.alive = false;
   523→      d.ai.deathScore = w.score;
   524→    }
   525→    finishDuel(d, s);
   526→  }
   527→}

   540→function renderDuel(
   541→  canvas: HTMLCanvasElement,
   542→  game: HTMLCanvasElement,
   543→  d: DuelSim,
   544→  tNow: number
   545→): void {
   546→  const ctx = canvas.getContext("2d");
   547→  if (!ctx) return;
   548→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   549→  ctx.clearRect(0, 0, canvas.width, canvas.height);
   550→  if (canvas.width > 0 && canvas.height > 0) {
   551→    ctx.imageSmoothingEnabled = true;
   552→    ctx.drawImage(game, 0, 0, GAME_W, GAME_H, 0, 0, canvas.width, canvas.height);
   553→  }
   554→  const sx = canvas.width / GAME_W;
   555→  const sy = canvas.height / GAME_H;
   556→  ctx.setTransform(sx, 0, 0, sy, 0, 0);
   557→
   558→  // champion FLY — emerald ring + halo, drawn first, slightly ghosted
   559→  if (d.ai.alive) {
   560→    drawDuelMarker(ctx, DUEL_FLY_X, "rgba(52,211,153,0.7)");
   561→    drawFly(ctx, {
   562→      x: DUEL_FLY_X,
   563→      feetY: d.ai.fly.y,
   564→      airborne: d.ai.fly.airborne,
   565→      ducking: d.ai.fly.ducking,
   566→      wingPhase: d.ai.fly.wingPhase,

   594→/** One 60Hz simulation step: world → vision → brains → action → collisions. */
   595→function stepSim(
   596→  sim: Sim,
   597→  art: Art,
   598→  dt: number,
   599→  popSize: number,
   600→  mutStrength: number,
   601→  realTime: boolean
   602→): void {
   603→  const w = sim.world;
   604→  const cleared = stepWorld(w, dt);
   605→
   606→  // fly avatar physics (shared with the duel — same helper, no drift)
   607→  for (const r of sim.runners) {
   608→    stepAvatarPhysics(r.fly, dt);
   609→    r.fly.ducking = false; // re-decided by the brain every step
   610→  }
   611→
   612→  // vision: world without flies → 240×90 → 24×9 retina (shared input)
   613→  drawWorld(art.gameCtx, w);
   614→  art.retinaCtx.drawImage(art.game, 0, 0, RETINA_W, RETINA_H);
   615→  const img = art.retinaCtx.getImageData(0, 0, RETINA_W, RETINA_H);
   616→  const input = retinaFromImageData(
   617→    img.data,
   618→    RETINA_W,
   619→    RETINA_H,
   620→    RETINA_COLS,
   621→    RETINA_ROWS,
   622→    false
   623→  );
   624→
   625→  // brains: dopamine rides along with the regular step
   626→  for (const r of sim.runners) {
   627→    if (!r.alive) continue;
   628→    trackBirdEncounters(w, r.duck); // duck-defense bookkeeping (alive flies only)
   629→    const delta = cleared.length > 0 ? REWARD_CLEAR : 0;
   630→    const m = r.brain.step(input, delta);
   631→    const f = r.fly;
   632→    if (!f.airborne && f.jumpCooldownMs <= 0 && m[0] > JUMP_THRESHOLD) {
   633→      f.vy = -JUMP_V;
   634→      f.airborne = true;
   635→      f.jumpCooldownMs = JUMP_COOLDOWN_MS;
   636→      if (realTime) playSound("jump", 0.15);
   637→    }
   638→    if (m[1] > DUCK_THRESHOLD) f.ducking = true;
   639→  }
   640→  if (cleared.length > 0) {
   641→    const dodgedBird = cleared.some((o) => o.type === "bird");
   642→    pushEvent(
   643→      sim,
   644→      dodgedBird ? "Pterodactyl dodged (ducked under)" : "Obstacle cleared",
   645→      REWARD_CLEAR,
   646→      "reward",
   647→      w.t
   648→    );
   649→  }
   650→
   651→  // collisions → death shock (+ pterodactyl vs cactus report)
   652→  for (const r of sim.runners) {
   653→    if (!r.alive) continue;
   654→    const hit = hitObstacle(w, r.fly);
   655→    if (hit) {
   656→      r.alive = false;
   657→      r.deathScore = w.score;
   658→      r.deathCause = hit.type;
   659→      if (hit.type === "bird") sim.genBirdDeaths += 1;
   660→      else sim.genCactusDeaths += 1;
   661→      r.brain.step(input, -PUNISH_CRASH);
   662→      pushEvent(
   663→        sim,
   664→        `Fly #${r.idx + 1} ${hit.type === "bird" ? "hit the bird" : "hit a cactus"}`,
   665→        -PUNISH_CRASH,
   666→        "punish",
   667→        w.t
   668→      );
   669→      if (realTime) playSound("crash", 0.25);
   670→    }
   671→  }
   672→
   673→  // generation end (or watch-mode replay restart)
   674→  const anyAlive = sim.runners.some((r) => r.alive);
   675→  if (!anyAlive || (!sim.watchMode && w.t >= GEN_TIME_CAP_S)) {
   676→    if (sim.watchMode) {
   677→      sim.watchCooldown = (sim.watchCooldown ?? 45) - 1;
   678→      if (sim.watchCooldown <= 0) {
   679→        sim.watchCooldown = null;
   680→        if (sim.bestBrain) {
   681→          sim.runners = [makeRunner(sim.bestBrain.clone(), 0)];
   682→        }
   683→        sim.world = freshWorld(sim);
   684→      }
   685→    } else {
   686→      endGeneration(sim, popSize, mutStrength);
   687→    }
   688→  }
   689→}

   691→function renderVisible(
   692→  canvas: HTMLCanvasElement,
   693→  game: HTMLCanvasElement,
   694→  sim: Sim,
   695→  tNow: number
   696→): void {
   697→  const ctx = canvas.getContext("2d");
   698→  if (!ctx) return;
   699→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   700→  ctx.clearRect(0, 0, canvas.width, canvas.height);
   701→  if (canvas.width > 0 && canvas.height > 0) {
   702→    ctx.imageSmoothingEnabled = true;
   703→    ctx.drawImage(game, 0, 0, GAME_W, GAME_H, 0, 0, canvas.width, canvas.height);
   704→  }
   705→  const sx = canvas.width / GAME_W;
   706→  const sy = canvas.height / GAME_H;
   707→  ctx.setTransform(sx, 0, 0, sy, 0, 0);
   708→
   709→  const leader = bestAliveBrain(sim);
   710→  for (const r of sim.runners) {
   711→    if (!r.alive) continue;
   712→    const isBest = sim.watchMode ? true : leader === r.brain;
   713→    drawFly(ctx, {
   714→      x: FLY_X,
   715→      feetY: r.fly.y,
   716→      airborne: r.fly.airborne,
   717→      ducking: r.fly.ducking,
   718→      wingPhase: r.fly.wingPhase,
   719→      alpha: isBest ? 1 : 0.38,
   720→      isBest,
   721→      label: sim.watchMode ? undefined : "BEST",
   722→      t: tNow,
   723→    });
   724→  }
   725→}

  1046→      s.world = freshWorld(s);

  1105→
  1106→  const toggleWatchMode = (on: boolean) => {
