     3→/**
     4→ * DinoTrainer — a population of FlyBrains learns to play the Chrome Dino
     5→ * runner, generation by generation.
     6→ *
     7→ * How it works:
     8→ *  - every 60Hz sim step, the shared world (obstacles, ground, sky — NO
     9→ *    flies) is drawn to an offscreen canvas, downscaled to 240×90 and
    10→ *    encoded into a 24×9 retina frame shared by ALL flies (they can't see
    11→ *    themselves — pure reactive vision)
    12→ *  - each alive fly steps its brain: motor 0 > 0.12 → jump, motor 1 > 0.15 → duck
    13→ *  - clearing an obstacle injects +0.4 dopamine (sugar); crashing injects
    14→ *    −0.3 (shock) and kills the fly
    15→ *  - when every fly is dead (or the 3-minute cap hits), the generation
    16→ *    ends: top-2 brains are cloned (elitism), the rest are bred via
    17→ *    tournament-of-5 crossover + mutation → next generation, automatically
    18→ *  - cross-component brain loading via useBrainStore, session autosave via
    19→ *    localStorage, best-brain export to the library via POST /api/brains
    20→ *  - duck-defense metrics ("Pterodactyl report"): every run tracks birds
    21→ *    seen vs cleared + a death cause (bird/cactus/timeout), aggregated into
    22→ *    a session progress bar; "bird practice" spawns pterodactyls from
    23→ *    score 0 so ducking can be trained deliberately
    24→ *  - sound effects via @/lib/sound (milestone on new HI, gen on evolve,
    25→ *    crash/jump at real-time speed, ding on save, click on start/resume)
    26→ */

   181→function createArt(): Art {
   182→  const game = document.createElement("canvas");
   183→  game.width = GAME_W;
   184→  game.height = GAME_H;
   185→  const gameCtx = game.getContext("2d");
   186→  const retina = document.createElement("canvas");
   187→  retina.width = RETINA_W;
   188→  retina.height = RETINA_H;
   189→  const retinaCtx = retina.getContext("2d", { willReadFrequently: true });
   190→  if (!gameCtx || !retinaCtx) {
   191→    throw new Error("2D canvas unavailable");
   192→  }
   193→  return { game, gameCtx, retina, retinaCtx };
   194→}

   215→function pushEvent(
   216→  sim: Sim,
   217→  label: string,
   218→  value: number,
   219→  kind: FeedEvent["kind"],
   220→  t: number
   221→): void {
   222→  sim.eventId += 1;
   223→  sim.events.push({ id: sim.eventId, label, value, kind, t });
   224→  if (sim.events.length > 40) sim.events.splice(0, sim.events.length - 40);
   225→}

   227→function bestAliveBrain(sim: Sim): FlyBrain | null {
   228→  let best: FlyBrain | null = null;
   229→  let bestR = -Infinity;
   230→  for (const r of sim.runners) {
   231→    if (r.alive && r.brain.rewards > bestR) {
   232→      bestR = r.brain.rewards;
   233→      best = r.brain;
   234→    }
   235→  }
   236→  return best ?? sim.runners[0]?.brain ?? null;
   237→}

   239→function autosave(sim: Sim): void {
   240→  try {
   241→    // snapshot.generation = session's latest gen, so "Resume" continues
   242→    // numbering where the session left off (population = clones of best brain)
   243→    const snap = sim.bestBrain
   244→      ? sim.bestBrain.toJSON("dino", "session best", sim.generation, sim.bestEver)
   245→      : null;
   246→    localStorage.setItem(
   247→      AUTOSAVE_KEY,
   248→      JSON.stringify({
   249→        v: 1,
   250→        gen: sim.generation,
   251→        bestScore: sim.bestEver,
   252→        history: sim.history.slice(-200),
   253→        snapshot: snap,
   254→      })
   255→    );
   256→  } catch {
   257→    // storage blocked/full — autosave is best-effort
   258→  }
   259→}

   261→function endGeneration(sim: Sim, popSize: number, mutStrength: number): void {
   262→  for (const r of sim.runners) {
   263→    if (r.alive) {
   264→      r.deathScore = sim.world.score;
   265→      r.deathCause = "timeout";
   266→      sim.genTimeoutDeaths += 1;
   267→    }
   268→  }
   269→  const scores = sim.runners.map((r) => r.deathScore);
   270→  const best = Math.max(...scores);
   271→  const avg = scores.reduce((a, b) => a + b, 0) / Math.max(1, scores.length);
   272→  sim.history.push({ gen: sim.generation, best, avg });
   273→
   274→  // fold this generation's per-run bird encounters into the session totals
   275→  const genBirdsSeen = sim.runners.reduce((a, r) => a + birdsSeenCount(r.duck), 0);
   276→  const genBirdsCleared = sim.runners.reduce((a, r) => a + birdsClearedCount(r.duck), 0);
   277→  sim.sessionBirdsSeen += genBirdsSeen;
   278→  sim.sessionBirdsCleared += genBirdsCleared;
   279→
   280→  let bi = 0;
   281→  for (let i = 1; i < scores.length; i++) if (scores[i] > scores[bi]) bi = i;
   282→  if (best > sim.bestEver) {
   283→    sim.bestEver = best;
   284→    sim.bestGen = sim.generation;
   285→    sim.bestBrain = sim.runners[bi].brain.clone();
   286→    playSound("milestone"); // new all-time HI score
   287→  }
   288→  const duckNote =
   289→    genBirdsSeen > 0
   290→      ? ` · duck ${Math.round((100 * genBirdsCleared) / genBirdsSeen)}%`
   291→      : "";
   292→  pushEvent(
   293→    sim,
   294→    `Generation ${sim.generation} ended — best ${best}, avg ${avg.toFixed(1)}${duckNote}`,
   295→    0,
   296→    "info",
   297→    sim.world.t
   298→  );
   299→  autosave(sim);
   300→  playSound("gen"); // generation complete → evolve
   301→
   302→  const brains = evolvePopulation(
   303→    sim.runners.map((r) => r.brain),
   304→    scores,
   305→    popSize,
   306→    mutStrength
   307→  );
   308→  sim.generation += 1;
   309→  sim.runners = brains.map((b, i) => makeRunner(b, i));
   310→  sim.genBirdDeaths = 0;
   311→  sim.genCactusDeaths = 0;
   312→  sim.genTimeoutDeaths = 0;
   313→  sim.world = freshWorld(sim);
   314→}

   316→/** One 60Hz simulation step: world → vision → brains → action → collisions. */
   317→function stepSim(
   318→  sim: Sim,
   319→  art: Art,
   320→  dt: number,
   321→  popSize: number,
   322→  mutStrength: number,
   323→  realTime: boolean
   324→): void {
   325→  const w = sim.world;
   326→  const cleared = stepWorld(w, dt);
   327→
   328→  // fly avatar physics
   329→  for (const r of sim.runners) {
   330→    const f = r.fly;
   331→    f.wingPhase += dt * 36;
   332→    if (f.jumpCooldownMs > 0) {
   333→      f.jumpCooldownMs = Math.max(0, f.jumpCooldownMs - dt * 1000);
   334→    }
   335→    if (f.airborne) {
   336→      // duck input while airborne = fast fall, like the real game
   337→      f.vy += GRAVITY * (f.ducking ? 2.6 : 1) * dt;
   338→      f.y += f.vy * dt;
   339→      if (f.y >= GROUND_Y) {
   340→        f.y = GROUND_Y;
   341→        f.vy = 0;
   342→        f.airborne = false;
   343→      }
   344→    }
   345→    f.ducking = false; // re-decided by the brain every step
   346→  }
   347→
   348→  // vision: world without flies → 240×90 → 24×9 retina (shared input)
   349→  drawWorld(art.gameCtx, w);
   350→  art.retinaCtx.drawImage(art.game, 0, 0, RETINA_W, RETINA_H);
   351→  const img = art.retinaCtx.getImageData(0, 0, RETINA_W, RETINA_H);
   352→  const input = retinaFromImageData(
   353→    img.data,
   354→    RETINA_W,
   355→    RETINA_H,
   356→    RETINA_COLS,
   357→    RETINA_ROWS,
   358→    false
   359→  );
   360→
   361→  // brains: dopamine rides along with the regular step
   362→  for (const r of sim.runners) {
   363→    if (!r.alive) continue;
   364→    trackBirdEncounters(w, r.duck); // duck-defense bookkeeping (alive flies only)
   365→    const delta = cleared.length > 0 ? REWARD_CLEAR : 0;
   366→    const m = r.brain.step(input, delta);
   367→    const f = r.fly;
   368→    if (!f.airborne && f.jumpCooldownMs <= 0 && m[0] > JUMP_THRESHOLD) {
   369→      f.vy = -JUMP_V;
   370→      f.airborne = true;
   371→      f.jumpCooldownMs = JUMP_COOLDOWN_MS;
   372→      if (realTime) playSound("jump", 0.15);
   373→    }
   374→    if (m[1] > DUCK_THRESHOLD) f.ducking = true;
   375→  }
   376→  if (cleared.length > 0) {
   377→    const dodgedBird = cleared.some((o) => o.type === "bird");
   378→    pushEvent(
   379→      sim,
   380→      dodgedBird ? "Pterodactyl dodged (ducked under)" : "Obstacle cleared",
   381→      REWARD_CLEAR,
   382→      "reward",
   383→      w.t
   384→    );
   385→  }
   386→
   387→  // collisions → death shock (+ pterodactyl vs cactus report)
   388→  for (const r of sim.runners) {
   389→    if (!r.alive) continue;
   390→    const hit = hitObstacle(w, r.fly);
   391→    if (hit) {
   392→      r.alive = false;
   393→      r.deathScore = w.score;
   394→      r.deathCause = hit.type;
   395→      if (hit.type === "bird") sim.genBirdDeaths += 1;
   396→      else sim.genCactusDeaths += 1;
   397→      r.brain.step(input, -PUNISH_CRASH);
   398→      pushEvent(
   399→        sim,
   400→        `Fly #${r.idx + 1} ${hit.type === "bird" ? "hit the bird" : "hit a cactus"}`,
   401→        -PUNISH_CRASH,
   402→        "punish",
   403→        w.t
   404→      );
   405→      if (realTime) playSound("crash", 0.25);
   406→    }
   407→  }
   408→
   409→  // generation end (or watch-mode replay restart)
   410→  const anyAlive = sim.runners.some((r) => r.alive);
   411→  if (!anyAlive || (!sim.watchMode && w.t >= GEN_TIME_CAP_S)) {
   412→    if (sim.watchMode) {
   413→      sim.watchCooldown = (sim.watchCooldown ?? 45) - 1;
   414→      if (sim.watchCooldown <= 0) {
   415→        sim.watchCooldown = null;
   416→        if (sim.bestBrain) {
   417→          sim.runners = [makeRunner(sim.bestBrain.clone(), 0)];
   418→        }
   419→        sim.world = freshWorld(sim);
   420→      }
   421→    } else {
   422→      endGeneration(sim, popSize, mutStrength);
   423→    }
   424→  }
   425→}

   427→function renderVisible(
   428→  canvas: HTMLCanvasElement,
   429→  game: HTMLCanvasElement,
   430→  sim: Sim,
   431→  tNow: number
   432→): void {
   433→  const ctx = canvas.getContext("2d");
   434→  if (!ctx) return;
   435→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   436→  ctx.clearRect(0, 0, canvas.width, canvas.height);
   437→  if (canvas.width > 0 && canvas.height > 0) {
   438→    ctx.imageSmoothingEnabled = true;
   439→    ctx.drawImage(game, 0, 0, GAME_W, GAME_H, 0, 0, canvas.width, canvas.height);
   440→  }
   441→  const sx = canvas.width / GAME_W;
   442→  const sy = canvas.height / GAME_H;
   443→  ctx.setTransform(sx, 0, 0, sy, 0, 0);
   444→
   445→  const leader = bestAliveBrain(sim);
   446→  for (const r of sim.runners) {
   447→    if (!r.alive) continue;
   448→    const isBest = sim.watchMode ? true : leader === r.brain;
   449→    drawFly(ctx, {
   450→      x: FLY_X,
   451→      feetY: r.fly.y,
   452→      airborne: r.fly.airborne,
   453→      ducking: r.fly.ducking,
   454→      wingPhase: r.fly.wingPhase,
   455→      alpha: isBest ? 1 : 0.38,
   456→      isBest,
   457→      label: sim.watchMode ? undefined : "BEST",
   458→      t: tNow,
   459→    });
   460→  }
   461→}
