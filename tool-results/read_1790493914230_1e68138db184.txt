   129→export function taskMeta(task: string) {
   130→  return (
   131→    TASK_META[task] ?? {
   132→      label: task ? task.charAt(0).toUpperCase() + task.slice(1) : "Unknown",
   133→      icon: FlaskConical,
   134→      badge: "border-border bg-muted text-muted-foreground",
   135→      iconColor: "text-muted-foreground",
   136→    }
   137→  );
   138→}
   139→
   140→export function formatScore(task: string, score: number): string {
   141→  const num =
   142→    Math.abs(score) >= 1000 ? `${(score / 1000).toFixed(1)}k` : `${Math.round(score * 10) / 10}`;
   143→  if (task === "dino") return `${num} pts`;
   144→  if (task === "bicycle") return `${num} m`;
   145→  return num;
   146→}
   147→
   148→export function formatBytes(n: number): string {
   149→  if (n >= 1024 * 1024) return `${(n / (1024 * 1024)).toFixed(1)} MB`;
   150→  if (n >= 1024) return `${(n / 1024).toFixed(1)} KB`;
   151→  return `${n} B`;
   152→}
   153→
   154→export function relativeDate(iso: string): string {
   155→  try {
   156→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
   157→  } catch {
   158→    return "unknown";
   159→  }
   160→}
   161→
   162→function sanitizeFileName(name: string): string {
   163→  const clean = name
   164→    .toLowerCase()
   165→    .replace(/[^a-z0-9-_]+/g, "-")
   166→    .replace(/^-+|-+$/g, "");
   167→  return clean || "brain";
   168→}
   169→
   170→export async function fetchBrain(id: string): Promise<{ brain: BrainRow; snapshot: BrainSnapshot }> {
   171→  const res = await fetch(`/api/brains/${id}`, { cache: "no-store" });
   172→  const data = await res.json().catch(() => ({}));
   173→  if (!res.ok) {
   174→    throw new Error(data?.error ?? `Could not fetch brain (${res.status})`);
   175→  }
   176→  if (!data?.snapshot || typeof data.snapshot !== "object" || !data.snapshot.weights) {
   177→    throw new Error("Stored snapshot is malformed");
   178→  }
   179→  return data;
   180→}

   204→export function BrainLibrary(_props: BrainLibraryProps = {}) {
   205→  const [brains, setBrains] = useState<BrainRow[]>([]);
   206→  const [loading, setLoading] = useState(true);
   207→  const [error, setError] = useState<string | null>(null);
   208→  const [taskFilter, setTaskFilter] = useState<TaskFilter>("all");
   209→  const [query, setQuery] = useState("");
   210→  const [busyId, setBusyId] = useState<string | null>(null);
   211→  const [deleteTarget, setDeleteTarget] = useState<BrainRow | null>(null);
   212→  const [deleteBusy, setDeleteBusy] = useState(false);
   213→  const [importOpen, setImportOpen] = useState(false);
   214→  const [importing, setImporting] = useState(false);
   215→  const [dragOver, setDragOver] = useState(false);
   216→  /** the brain whose report-card dialog is open (Task 10-b) */
   217→  const [reportRow, setReportRow] = useState<BrainRow | null>(null);
   218→
   219→  const fileInputRef = useRef<HTMLInputElement>(null);
   220→
   221→  const loadList = useCallback(async (task: TaskFilter) => {
   222→    setLoading(true);
   223→    setError(null);
   224→    try {
   225→      const qs = task !== "all" ? `?task=${encodeURIComponent(task)}` : "";
   226→      const res = await fetch(`/api/brains${qs}`, { cache: "no-store" });
   227→      const data = await res.json().catch(() => ({}));
   228→      if (!res.ok) throw new Error(data?.error ?? `Server responded ${res.status}`);
   229→      setBrains(Array.isArray(data.brains) ? (data.brains as BrainRow[]) : []);
   230→    } catch (err) {
   231→      const msg = err instanceof Error ? err.message : "Failed to load the library";
   232→      setError(msg);
   233→      if (brains.length > 0) toast.error(`Couldn't refresh the library: ${msg}`);
   234→    } finally {
   235→      setLoading(false);
   236→    }
   237→  }, [brains.length]);
   238→
   239→  useEffect(() => {
   240→    void loadList("all");
   241→  }, []);
   242→
   243→  const handleFilterChange = (task: TaskFilter) => {
   244→    setTaskFilter(task);
   245→    void loadList(task);
   246→  };
   247→
   248→  const visible = useMemo(() => {
   249→    const q = query.trim().toLowerCase();
   250→    return brains.filter((b) => {
   251→      if (q) {
   252→        const haystack = `${b.name} ${b.task} ${b.note ?? ""}`.toLowerCase();
   253→        if (!haystack.includes(q)) return false;
   254→      }
   255→      return true;
   256→    });
   257→  }, [brains, query]);
   258→
   259→  const totalBytes = useMemo(() => brains.reduce((acc, b) => acc + (b.snapshotBytes ?? 0), 0), [brains]);
   260→
   261→  // ----- actions ----------------------------------------------------------
   262→
   263→  const handleLoad = async (row: BrainRow) => {
   264→    setBusyId(row.id);
   265→    try {
   266→      const { snapshot } = await fetchBrain(row.id);
   267→      useBrainStore.getState().requestLoad(row.task, snapshot, row.name);
   268→      const meta = taskMeta(row.task);
   269→      toast.success(`Loaded "${row.name}" — open the ${meta.label} tab to continue training`);
   270→    } catch (err) {
   271→      toast.error(err instanceof Error ? err.message : "Load failed");
   272→    } finally {
   273→      setBusyId(null);
   274→    }
   275→  };
   276→
   277→  const handleExport = async (row: BrainRow) => {
   278→    setBusyId(row.id);
   279→    try {
   280→      const { snapshot } = await fetchBrain(row.id);
   281→      const blob = new Blob([JSON.stringify(snapshot, null, 2)], { type: "application/json" });
   282→      const url = URL.createObjectURL(blob);
   283→      const a = document.createElement("a");
   284→      a.href = url;
   285→      a.download = `fly-brain-${sanitizeFileName(row.name)}-gen${row.generation}.json`;
   286→      document.body.appendChild(a);
   287→      a.click();
   288→      a.remove();
   289→      URL.revokeObjectURL(url);
   290→      toast.success(`Exported "${row.name}" (gen ${row.generation}) as JSON`);
   291→    } catch (err) {
   292→      toast.error(err instanceof Error ? err.message : "Export failed");
   293→    } finally {
   294→      setBusyId(null);
   295→    }
   296→  };
   297→
   298→  const handleDelete = async () => {
   299→    if (!deleteTarget) return;
   300→    setDeleteBusy(true);
   301→    const row = deleteTarget;
   302→    try {
   303→      const res = await fetch(`/api/brains/${row.id}`, { method: "DELETE" });
   304→      const data = await res.json().catch(() => ({}));
   305→      if (!res.ok) throw new Error(data?.error ?? `Delete failed (${res.status})`);
   306→      setBrains((bs) => bs.filter((b) => b.id !== row.id));
   307→      toast.success(`Deleted "${row.name}" — that connectome is gone`);
   308→      setDeleteTarget(null);
   309→    } catch (err) {
   310→      toast.error(err instanceof Error ? err.message : "Delete failed");
   311→    } finally {
   312→      setDeleteBusy(false);
   313→    }
   314→  };
   315→
   316→  const handleImportFile = async (file: File) => {
   317→    setImporting(true);
   318→    try {
   319→      const text = await file.text();
   320→      let parsed: unknown;
   321→      try {
   322→        parsed = JSON.parse(text);
   323→      } catch {
   324→        throw new Error("That file isn't valid JSON");
   325→      }
   326→      const snap = parsed as Partial<BrainSnapshot> | null;
   327→      if (!snap || typeof snap !== "object" || Array.isArray(snap)) {
   328→        throw new Error("Not a fly-brain file — expected a JSON object");
   329→      }
   330→      if (snap.version !== 1) throw new Error(`Unsupported brain version: ${String(snap.version)}`);
   331→      if (typeof snap.task !== "string" || !snap.task.trim()) throw new Error("Snapshot is missing its task");
   332→      if (!snap.arch || typeof snap.arch !== "object") throw new Error("Snapshot is missing its architecture");
   333→      if (!snap.weights || typeof snap.weights !== "object") throw new Error("Snapshot is missing its weights");
   334→      for (const key of ["kenyonToMbon", "mbonToMotor", "mbonBias", "motorBias"] as const) {
   335→        if (!Array.isArray(snap.weights[key])) throw new Error(`Snapshot weights.${key} is missing`);
   336→      }
   337→
   338→      const fallbackName = file.name
   339→        .replace(/\.json$/i, "")
   340→        .replace(/[_-]+/g, " ")
   341→        .trim();
   342→      const name =
   343→        typeof snap.name === "string" && snap.name.trim() ? snap.name.trim() : fallbackName || "Imported brain";
   344→
   345→      const res = await fetch("/api/brains", {
   346→        method: "POST",
   347→        headers: { "Content-Type": "application/json" },
   348→        body: JSON.stringify({
   349→          name,
   350→          task: snap.task,
   351→          generation: typeof snap.generation === "number" ? snap.generation : 0,
   352→          score: typeof snap.score === "number" ? snap.score : 0,
   353→          note: typeof snap.meta?.note === "string" ? snap.meta.note : undefined,
   354→          snapshot: snap,
   355→        }),
   356→      });
   357→      const data = await res.json().catch(() => ({}));
   358→      if (!res.ok) throw new Error(data?.error ?? `Import failed (${res.status})`);
   359→
   360→      toast.success(`Imported "${data.brain?.name ?? name}" (gen ${data.brain?.generation ?? 0}) into the library`);
   361→      setImportOpen(false);
   362→      await loadList(taskFilter);
   363→    } catch (err) {
   364→      toast.error(err instanceof Error ? err.message : "Import failed");
   365→    } finally {
   366→      setImporting(false);
   367→      if (fileInputRef.current) fileInputRef.current.value = "";
   368→    }
   369→  };
   370→
   371→  // ----- report card (Task 10-b) ------------------------------------------
   372→
   373→  const handleOpenReport = (row: BrainRow) => {
   374→    playSound("click");
   375→    setReportRow(row);
   376→  };
   377→
   378→  // ----- render helpers ---------------------------------------------------
   379→
   380→  const showEmptyState = !loading && !error && visible.length === 0;
   381→  const showFilteredEmpty = showEmptyState && (brains.length > 0 || query.trim() !== "");
   382→
   383→  const rowActions = (row: BrainRow, full = false) => (
   384→    <div className={cn("flex items-center gap-2", full && "w-full")}>
   385→      <Button
   386→        size="sm"
   387→        className={cn("h-11 gap-1.5 bg-emerald-600 px-4 text-white hover:bg-emerald-500", full && "flex-1")}
   388→        disabled={busyId === row.id}
   389→        onClick={() => void handleLoad(row)}
   390→        aria-label={`Load ${row.name} into the trainer`}
   391→      >
   392→        {busyId === row.id ? <Loader2 className="h-4 w-4 animate-spin" /> : <Play className="h-4 w-4" />}
   393→        Load
   394→      </Button>
   395→      <Button
   396→        variant="outline"
   397→        size={full ? "sm" : "icon"}
   398→        className={cn(
   399→          "h-11 shrink-0 border-teal-500/30 hover:border-teal-500/50 hover:bg-teal-500/10 hover:text-teal-600 dark:hover:text-teal-300",
   400→          full ? "flex-1 gap-1.5 px-4" : "w-11",
   401→        )}
   402→        disabled={busyId === row.id}
   403→        onClick={() => handleOpenReport(row)}
   404→        aria-label={`Open report card for ${row.name}`}
   405→        title="Report card"
   406→      >
   407→        <GraduationCap className="h-4 w-4" aria-hidden />
   408→        {full ? "Report card" : null}
   409→      </Button>
   410→      <DropdownMenu>
   411→        <DropdownMenuTrigger asChild>
   412→          <Button
   413→            variant="outline"
   414→            size="icon"
   415→            className="h-11 w-11 shrink-0"
   416→            disabled={busyId === row.id}
   417→            aria-label={`More actions for ${row.name}`}
   418→          >
   419→            <MoreHorizontal className="h-4 w-4" />
   420→          </Button>
   421→        </DropdownMenuTrigger>
   422→        <DropdownMenuContent align="end" className="w-44">
   423→          <DropdownMenuItem
   424→            className="min-h-11 cursor-pointer"
   425→            onSelect={() => void handleExport(row)}
   426→          >
   427→            <Download className="h-4 w-4" /> Export JSON
   428→          </DropdownMenuItem>
   429→          <DropdownMenuItem
   430→            className="min-h-11 cursor-pointer text-rose-400 focus:text-rose-300"
   431→            onSelect={() => setDeleteTarget(row)}
   432→          >
   433→            <Trash2 className="h-4 w-4" /> Delete
   434→          </DropdownMenuItem>
   435→        </DropdownMenuContent>
   436→      </DropdownMenu>
   437→    </div>
   438→  );
   439→
   440→  // ----- render -----------------------------------------------------------
   441→
   442→  return (
   443→    <div data-testid="brain-library" className="w-full">
   444→      {/* sonner toaster — layout only mounts the radix one, so mount it here */}
   445→      <Toaster position="bottom-right" />
   446→
   447→      <Card className="gap-4 border-border/60 bg-card/70 py-5 backdrop-blur-sm sm:py-6">
   448→        <CardHeader className="border-b border-border/60 px-4 pb-4 sm:px-6">
   449→          <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
   450→            <div className="flex items-start gap-3">
   451→              <div className="mt-0.5 flex h-10 w-10 shrink-0 items-center justify-center rounded-xl border border-emerald-500/30 bg-emerald-500/10">
   452→                <Brain className="h-5 w-5 text-emerald-400" aria-hidden />
   453→              </div>
   454→              <div>
   455→                <CardTitle className="text-lg">Brain Library</CardTitle>
   456→                <CardDescription className="mt-1 max-w-xl text-sm">
   457→                  Saved connectomes. Load one and the fly remembers everything it learned —
   458→                  weights, generation, scars.
   459→                </CardDescription>
   460→              </div>
   461→            </div>
   462→            <div className="flex shrink-0 items-center gap-2">
   463→              <Button
   464→                variant="outline"
   465→                size="icon"
   466→                className="h-11 w-11"
   467→                onClick={() => void loadList(taskFilter)}
   468→                disabled={loading}
   469→                aria-label="Refresh the brain library"
   470→                title="Refresh"
   471→              >
   472→                <RefreshCw className={cn("h-4 w-4", loading && "animate-spin")} />
   473→              </Button>
   474→              <Button
   475→                size="sm"
   476→                className="h-11 gap-1.5 bg-emerald-600 px-4 text-white hover:bg-emerald-500"
   477→                onClick={() => setImportOpen(true)}
   478→              >
   479→                <Upload className="h-4 w-4" />
   480→                Import
   481→              </Button>
   482→            </div>
   483→          </div>
   484→        </CardHeader>
   485→
   486→        <CardContent className="px-4 sm:px-6">
   487→          {/* filters + search */}
   488→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   489→            <div
   490→              role="tablist"
   491→              aria-label="Filter brains by task"
   492→              className="inline-flex max-w-full items-center gap-1 overflow-x-auto rounded-xl border border-border/60 bg-muted/40 p-1"
   493→            >
   494→              {FILTERS.map((f) => {
   495→                const active = taskFilter === f.value;
   496→                const Icon = f.icon;
   497→                return (
   498→                  <button
   499→                    key={f.value}
   500→                    role="tab"
   501→                    type="button"
   502→                    aria-selected={active}
   503→                    onClick={() => handleFilterChange(f.value)}

   505→                      "flex h-11 shrink-0 items-center gap-1.5 rounded-lg px-3 text-sm font-medium transition-colors",
   506→                      active
   507→                        ? "border border-border/60 bg-card text-foreground shadow-sm"
   508→                        : "text-muted-foreground hover:bg-card/60 hover:text-foreground"
   509→                    )}
   510→                  >
   511→                    <Icon
   512→                      className={cn(
   513→                        "h-4 w-4",
   514→                        f.value === "dino" && "text-amber-400",
   515→                        f.value === "bicycle" && "text-rose-400",

   525→            </div>

   529→                aria-hidden
   530→              />
   531→              <Input
   532→                value={query}
   533→                onChange={(e) => setQuery(e.target.value)}
   534→                placeholder="Search by name…"
   535→                className="h-11 pl-9"
   536→                aria-label="Search saved brains by name"
   537→              />
   538→            </div>
   539→          </div>
   540→
   541→          {/* content */}
   542→          <div className="mt-4">
   543→            {error && brains.length === 0 ? (
   544→              <div className="flex flex-col items-center gap-3 rounded-xl border border-rose-500/30 bg-rose-500/5 px-6 py-10 text-center">
   545→                <FlaskConical className="h-8 w-8 text-rose-400" aria-hidden />
   546→                <p className="text-sm text-rose-300">Couldn&apos;t reach the brain library: {error}</p>
   547→                <Button
   548→                  variant="outline"
