   227→   227→function bestAliveBrain(sim: Sim): FlyBrain | null {
   228→   228→  let best: FlyBrain | null = null;
   229→   229→  let bestR = -Infinity;
   230→   230→  for (const r of sim.runners) {
   231→   231→    if (r.alive && r.brain.rewards > bestR) {
   232→   232→      bestR = r.brain.rewards;
   233→   233→      best = r.brain;
   234→   234→    }
   235→   235→  }
   236→   236→  return best ?? sim.runners[0]?.brain ?? null;
   237→   237→}

   239→   239→function autosave(sim: Sim): void {
   240→   240→  try {
   241→   241→    // snapshot.generation = session's latest gen, so "Resume" continues
   242→   242→    // numbering where the session left off (population = clones of best brain)
   243→   243→    const snap = sim.bestBrain
   244→   244→      ? sim.bestBrain.toJSON("dino", "session best", sim.generation, sim.bestEver)
   245→   245→      : null;
   246→   246→    localStorage.setItem(
   247→   247→      AUTOSAVE_KEY,
   248→   248→      JSON.stringify({
   249→   249→        v: 1,
   250→   250→        gen: sim.generation,
   251→   251→        bestScore: sim.bestEver,
   252→   252→        history: sim.history.slice(-200),
   253→   253→        snapshot: snap,
   254→   254→      })
   255→   255→    );
   256→   256→  } catch {
   257→   257→    // storage blocked/full — autosave is best-effort
   258→   258→  }
   259→   259→}

   261→   261→function endGeneration(sim: Sim, popSize: number, mutStrength: number): void {
   262→   262→  for (const r of sim.runners) {
   263→   263→    if (r.alive) {
   264→   264→      r.deathScore = sim.world.score;
   265→   265→      r.deathCause = "timeout";
   266→   266→      sim.genTimeoutDeaths += 1;
   267→   267→    }
   268→   268→  }
   269→   269→  const scores = sim.runners.map((r) => r.deathScore);
   270→   270→  const best = Math.max(...scores);
   271→   271→  const avg = scores.reduce((a, b) => a + b, 0) / Math.max(1, scores.length);
   272→   272→  sim.history.push({ gen: sim.generation, best, avg });
   273→   273→
   274→   274→  // fold this generation's per-run bird encounters into the session totals
   275→   275→  const genBirdsSeen = sim.runners.reduce((a, r) => a + birdsSeenCount(r.duck), 0);
   276→   276→  const genBirdsCleared = sim.runners.reduce((a, r) => a + birdsClearedCount(r.duck), 0);
   277→   277→  sim.sessionBirdsSeen += genBirdsSeen;
   278→   278→  sim.sessionBirdsCleared += genBirdsCleared;
   279→   279→
   280→   280→  let bi = 0;
   281→   281→  for (let i = 1; i < scores.length; i++) if (scores[i] > scores[bi]) bi = i;
   282→   282→  if (best > sim.bestEver) {
   283→   283→    sim.bestEver = best;
   284→   284→    sim.bestGen = sim.generation;
   285→   285→    sim.bestBrain = sim.runners[bi].brain.clone();
   286→   286→    playSound("milestone"); // new all-time HI score
   287→   287→  }
   288→   288→  const duckNote =
   289→   289→    genBirdsSeen > 0
   290→   290→      ? ` · duck ${Math.round((100 * genBirdsCleared) / genBirdsSeen)}%`
   291→   291→      : "";
   292→   292→  pushEvent(
   293→   293→    sim,
   294→   294→    `Generation ${sim.generation} ended — best ${best}, avg ${avg.toFixed(1)}${duckNote}`,
   295→   295→    0,
   296→   296→    "info",
   297→   297→    sim.world.t
   298→   298→  );
   299→   299→  autosave(sim);
   300→   300→  playSound("gen"); // generation complete → evolve
   301→   301→
   302→   302→  const brains = evolvePopulation(
   303→   303→    sim.runners.map((r) => r.brain),
   304→   304→    scores,
   305→   305→    popSize,
   306→   306→    mutStrength
   307→   307→  );
   308→   308→  sim.generation += 1;
   309→   309→  sim.runners = brains.map((b, i) => makeRunner(b, i));
   310→   310→  sim.genBirdDeaths = 0;
   311→   311→  sim.genCactusDeaths = 0;
   312→   312→  sim.genTimeoutDeaths = 0;
   313→   313→  sim.world = freshWorld(sim);
   314→   314→}

   316→   316→/** One 60Hz simulation step: world → vision → brains → action → collisions. */
   317→   317→function stepSim(
   318→   318→  sim: Sim,
   319→   319→  art: Art,
   320→   320→  dt: number,
   321→   321→  popSize: number,
   322→   322→  mutStrength: number,
   323→   323→  realTime: boolean
   324→   324→): void {
   325→   325→  const w = sim.world;
   326→   326→  const cleared = stepWorld(w, dt);
   327→   327→
   328→   328→  // fly avatar physics
   329→   329→  for (const r of sim.runners) {
   330→   330→    const f = r.fly;
   331→   331→    f.wingPhase += dt * 36;
   332→   332→    if (f.jumpCooldownMs > 0) {
   333→   333→      f.jumpCooldownMs = Math.max(0, f.jumpCooldownMs - dt * 1000);
   334→   334→    }
   335→   335→    if (f.airborne) {
   336→   336→      // duck input while airborne = fast fall, like the real game
   337→   337→      f.vy += GRAVITY * (f.ducking ? 2.6 : 1) * dt;
   338→   338→      f.y += f.vy * dt;
   339→   339→      if (f.y >= GROUND_Y) {
   340→   340→        f.y = GROUND_Y;
   341→   341→        f.vy = 0;
   342→   342→        f.airborne = false;
   343→   343→      }
   344→   344→    }
   345→   345→    f.ducking = false; // re-decided by the brain every step
   346→   346→  }
   347→   347→
   348→   348→  // vision: world without flies → 240×90 → 24×9 retina (shared input)
   349→   349→  drawWorld(art.gameCtx, w);
   350→   350→  art.retinaCtx.drawImage(art.game, 0, 0, RETINA_W, RETINA_H);
   351→   351→  const img = art.retinaCtx.getImageData(0, 0, RETINA_W, RETINA_H);
   352→   352→  const input = retinaFromImageData(
   353→   353→    img.data,
   354→   354→    RETINA_W,
   355→   355→    RETINA_H,
   356→   356→    RETINA_COLS,
   357→   357→    RETINA_ROWS,
   358→   358→    false
   359→   359→  );
   360→   360→
   361→   361→  // brains: dopamine rides along with the regular step
   362→   362→  for (const r of sim.runners) {
   363→   363→    if (!r.alive) continue;
   364→   364→    trackBirdEncounters(w, r.duck); // duck-defense bookkeeping (alive flies only)
   365→   365→    const delta = cleared.length > 0 ? REWARD_CLEAR : 0;
   366→   366→    const m = r.brain.step(input, delta);
   367→   367→    const f = r.fly;
   368→   368→    if (!f.airborne && f.jumpCooldownMs <= 0 && m[0] > JUMP_THRESHOLD) {
   369→   369→      f.vy = -JUMP_V;
   370→   370→      f.airborne = true;
   371→   371→      f.jumpCooldownMs = JUMP_COOLDOWN_MS;
   372→   372→      if (realTime) playSound("jump", 0.15);
   373→   373→    }
   374→   374→    if (m[1] > DUCK_THRESHOLD) f.ducking = true;
   375→   375→  }
   376→   376→  if (cleared.length > 0) {
   377→   377→    const dodgedBird = cleared.some((o) => o.type === "bird");
   378→   378→    pushEvent(
   379→   379→      sim,
   380→   380→      dodgedBird ? "Pterodactyl dodged (ducked under)" : "Obstacle cleared",
   381→   381→      REWARD_CLEAR,
   382→   382→      "reward",
   383→   383→      w.t
   384→   384→    );
   385→   385→  }
   386→   386→
   387→   387→  // collisions → death shock (+ pterodactyl vs cactus report)
   388→   388→  for (const r of sim.runners) {
   389→   389→    if (!r.alive) continue;
   390→   390→    const hit = hitObstacle(w, r.fly);
   391→   391→    if (hit) {
   392→   392→      r.alive = false;
   393→   393→      r.deathScore = w.score;
   394→   394→      r.deathCause = hit.type;
   395→   395→      if (hit.type === "bird") sim.genBirdDeaths += 1;
   396→   396→      else sim.genCactusDeaths += 1;
   397→   397→      r.brain.step(input, -PUNISH_CRASH);
   398→   398→      pushEvent(
   399→   399→        sim,
   400→   400→        `Fly #${r.idx + 1} ${hit.type === "bird" ? "hit the bird" : "hit a cactus"}`,
   401→   401→        -PUNISH_CRASH,
   402→   402→        "punish",
   403→   403→        w.t
   404→   404→      );
   405→   405→      if (realTime) playSound("crash", 0.25);
   406→   406→    }
   407→   407→  }
   408→   408→
   409→   409→  // generation end (or watch-mode replay restart)
   410→   410→  const anyAlive = sim.runners.some((r) => r.alive);
   411→   411→  if (!anyAlive || (!sim.watchMode && w.t >= GEN_TIME_CAP_S)) {
   412→   412→    if (sim.watchMode) {
   413→   413→      sim.watchCooldown = (sim.watchCooldown ?? 45) - 1;
   414→   414→      if (sim.watchCooldown <= 0) {
   415→   415→        sim.watchCooldown = null;
   416→   416→        if (sim.bestBrain) {
   417→   417→          sim.runners = [makeRunner(sim.bestBrain.clone(), 0)];
   418→   418→        }
   419→   419→        sim.world = freshWorld(sim);
   420→   420→      }
   421→   421→    } else {
   422→   422→      endGeneration(sim, popSize, mutStrength);
   423→   423→    }
   424→   424→  }
   425→   425→}

   427→   427→function renderVisible(
   428→   428→  canvas: HTMLCanvasElement,
   429→   429→  game: HTMLCanvasElement,
   430→   430→  sim: Sim,
   431→   431→  tNow: number
   432→   432→): void {
   433→   433→  const ctx = canvas.getContext("2d");
   434→   434→  if (!ctx) return;
   435→   435→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   436→   436→  ctx.clearRect(0, 0, canvas.width, canvas.height);
   437→   437→  if (canvas.width > 0 && canvas.height > 0) {
   438→   438→    ctx.imageSmoothingEnabled = true;
   439→   439→    ctx.drawImage(game, 0, 0, GAME_W, GAME_H, 0, 0, canvas.width, canvas.height);
   440→   440→  }
   441→   441→  const sx = canvas.width / GAME_W;
   442→   442→  const sy = canvas.height / GAME_H;
   443→   443→  ctx.setTransform(sx, 0, 0, sy, 0, 0);
   444→   444→
   445→   445→  const leader = bestAliveBrain(sim);
   446→   446→  for (const r of sim.runners) {
   447→   447→    if (!r.alive) continue;
   448→   448→    const isBest = sim.watchMode ? true : leader === r.brain;
   449→   449→    drawFly(ctx, {
   450→   450→      x: FLY_X,
   451→   451→      feetY: r.fly.y,
   452→   452→      airborne: r.fly.airborne,
   453→   453→      ducking: r.fly.ducking,
   454→   454→      wingPhase: r.fly.wingPhase,
   455→   455→      alpha: isBest ? 1 : 0.38,
   456→   456→      isBest,
   457→   457→      label: sim.watchMode ? undefined : "BEST",
   458→   458→      t: tNow,
   459→   459→    });
   460→   460→  }
   461→   461→}
