    52→// ------------------------------------------------------------------- sky
    53→const SKY_VERT = /* glsl */ `
    54→  varying vec3 vDir;
    55→  void main() {
    56→    vDir = position;
    57→    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    58→  }
    59→`;
    60→const SKY_FRAG = /* glsl */ `
    61→  varying vec3 vDir;
    62→  uniform vec3 zenithColor;
    63→  uniform vec3 midColor;
    64→  uniform vec3 horizonColor;
    65→  uniform vec3 sunColor;
    66→  uniform vec3 sunDir;
    67→  void main() {
    68→    vec3 d = normalize(vDir);
    69→    float h = clamp(d.y, -0.08, 1.0);
    70→    vec3 col = mix(horizonColor, midColor, smoothstep(-0.02, 0.16, h));
    71→    col = mix(col, zenithColor, smoothstep(0.10, 0.52, h));
    72→    vec3 s = normalize(sunDir);
    73→    float sunDot = max(dot(d, s), 0.0);
    74→    float disc = pow(sunDot, 900.0);
    75→    float glow = pow(sunDot, 10.0);
    76→    float haze = pow(sunDot, 2.5);
    77→    col += sunColor * (disc * 1.35 + glow * 0.45 + haze * 0.12);
    78→    // faint early stars in the upper sky
    79→    vec3 cell = floor(d * 230.0);
    80→    float starHash = fract(sin(dot(cell, vec3(12.9898, 78.233, 37.719))) * 43758.5453);
    81→    float star = smoothstep(0.9975, 1.0, starHash) * smoothstep(0.28, 0.65, d.y);
    82→    col += vec3(1.0, 0.92, 0.82) * star * 0.55;
    83→    gl_FragColor = vec4(col, 1.0);
    84→  }
    85→`;
    86→
    87→const SUN_DIR = new THREE.Vector3(0.32, 0.1, 0.94).normalize();
    88→
    89→function DuskSky() {
    90→  const mat = useMemo(
    91→    () =>
    92→      new THREE.ShaderMaterial({
    93→        vertexShader: SKY_VERT,
    94→        fragmentShader: SKY_FRAG,
    95→        side: THREE.BackSide,
    96→        depthWrite: false,
    97→        uniforms: {
    98→          zenithColor: { value: new THREE.Color(C.zenith) },
    99→          midColor: { value: new THREE.Color(C.midSky) },
   100→          horizonColor: { value: new THREE.Color(C.horizon) },
   101→          sunColor: { value: new THREE.Color(C.sun) },
   102→          sunDir: { value: SUN_DIR },
   103→        },
   104→      }),
   105→    []
   106→  );
   107→  return (
   108→    <mesh material={mat} renderOrder={-10} frustumCulled={false}>
   109→      <sphereGeometry args={[950, 32, 20]} />
   110→    </mesh>
   111→  );
   112→}

   155→// ------------------------------------------------------------------ hills
   156→function Hills({ road }: { road: RoadData }) {
   157→  // built imperatively (instanced meshes + matrices) and mounted via
   158→  // <primitive> — plain three.js code, deterministic placement
   159→  const group = useMemo(() => {
   160→    const rand = mulberry32(20260207);
   161→    const m = new THREE.Matrix4();
   162→    const q = new THREE.Quaternion();
   163→    const col = new THREE.Color();
   164→
   165→    const nearMesh = new THREE.InstancedMesh(
   166→      new THREE.SphereGeometry(1, 9, 6),
   167→      new THREE.MeshStandardMaterial({ flatShading: true, roughness: 1, metalness: 0 }),
   168→      220
   169→    );
   170→    let n = 0;
   171→    // rolling hills every ~26 m on either side of the road
   172→    for (let s = 14; s < 2380; s += 24 + rand() * 16) {
   173→      for (const side of [-1, 1]) {
   174→        if (rand() < 0.35 || n >= 220) continue;
   175→        const { x, z, theta } = road.sample(s);
   176→        const lat = side * (13 + rand() * 65);
   177→        const r = 5 + rand() * 20;
   178→        const h = r * (0.28 + rand() * 0.3);
   179→        m.compose(
   180→          new THREE.Vector3(x - Math.sin(theta) * lat, h * 0.18, z + Math.cos(theta) * lat),
   181→          q,
   182→          new THREE.Vector3(r, h, r * (0.8 + rand() * 0.5))
   183→        );
   184→        nearMesh.setMatrixAt(n, m);
   185→        nearMesh.setColorAt(n, col.set(C.hills[Math.floor(rand() * C.hills.length)]));
   186→        n++;
   187→      }
   188→    }
   189→    nearMesh.count = n;
   190→    nearMesh.instanceMatrix.needsUpdate = true;
   191→    if (nearMesh.instanceColor) nearMesh.instanceColor.needsUpdate = true;
   192→
   193→    // distant ridge silhouettes
   194→    const farMesh = new THREE.InstancedMesh(
   195→      new THREE.SphereGeometry(1, 8, 5),
   196→      new THREE.MeshStandardMaterial({ flatShading: true, roughness: 1, metalness: 0 }),
   197→      14
   198→    );
   199→    for (let i = 0; i < 14; i++) {
   200→      const s = 100 + rand() * 2200;
   201→      const { x, z, theta } = road.sample(s);
   202→      const lat = (rand() < 0.5 ? -1 : 1) * (170 + rand() * 260);
   203→      const r = 90 + rand() * 140;
   204→      const h = r * (0.1 + rand() * 0.1);
   205→      m.compose(
   206→        new THREE.Vector3(x - Math.sin(theta) * lat, h * 0.15, z + Math.cos(theta) * lat),
   207→        q,
   208→        new THREE.Vector3(r, h, r)
   209→      );
   210→      farMesh.setMatrixAt(i, m);
   211→      farMesh.setColorAt(i, col.set(C.farHills));
   212→    }
   213→    farMesh.instanceMatrix.needsUpdate = true;
   214→    if (farMesh.instanceColor) farMesh.instanceColor.needsUpdate = true;
   215→
   216→    const g = new THREE.Group();
   217→    g.add(nearMesh, farMesh);
   218→    return g;
   219→  }, [road]);
   220→
   221→  return <primitive object={group} />;
   222→}

   224→// ------------------------------------------------------------------ trees
   225→function Trees({ road }: { road: RoadData }) {
   226→  const group = useMemo(() => {
   227→    const rand = mulberry32(424242);
   228→    const m = new THREE.Matrix4();
   229→    const q = new THREE.Quaternion();
   230→    const col = new THREE.Color();
   231→    const UP = new THREE.Vector3(0, 1, 0);
   232→    const COUNT = 170;
   233→    const trunkMesh = new THREE.InstancedMesh(
   234→      new THREE.CylinderGeometry(0.13, 0.18, 1.4, 5),
   235→      new THREE.MeshStandardMaterial({ color: C.trunk, flatShading: true, roughness: 1 }),
   236→      COUNT
   237→    );
   238→    const canopyMesh = new THREE.InstancedMesh(
   239→      new THREE.ConeGeometry(1.15, 3, 6),
   240→      new THREE.MeshStandardMaterial({ flatShading: true, roughness: 1, metalness: 0 }),
   241→      COUNT
   242→    );
   243→    for (let i = 0; i < COUNT; i++) {
   244→      const s = 8 + rand() * 2380;
   245→      const { x, z, theta } = road.sample(s);
   246→      const lat = (rand() < 0.5 ? -1 : 1) * (6.5 + rand() * 24);
   247→      const scale = 0.7 + rand() * 0.9;
   248→      const rot = rand() * Math.PI * 2;
   249→      const px = x - Math.sin(theta) * lat;
   250→      const pz = z + Math.cos(theta) * lat;
   251→      q.setFromAxisAngle(UP, rot);
   252→      m.compose(
   253→        new THREE.Vector3(px, 0.7 * scale, pz),
   254→        q,
   255→        new THREE.Vector3(scale * 0.9, scale, scale * 0.9)
   256→      );
   257→      trunkMesh.setMatrixAt(i, m);
   258→      m.compose(
   259→        new THREE.Vector3(px, (1.4 + 1.5) * scale, pz),
   260→        q,
   261→        new THREE.Vector3(scale, scale, scale)
   262→      );
   263→      canopyMesh.setMatrixAt(i, m);
   264→      canopyMesh.setColorAt(i, col.set(C.canopies[Math.floor(rand() * C.canopies.length)]));
   265→    }
   266→    trunkMesh.instanceMatrix.needsUpdate = true;
   267→    canopyMesh.instanceMatrix.needsUpdate = true;
   268→    if (canopyMesh.instanceColor) canopyMesh.instanceColor.needsUpdate = true;
   269→    const g = new THREE.Group();
   270→    g.add(trunkMesh, canopyMesh);
   271→    return g;
   272→  }, [road]);
   273→  return <primitive object={group} />;
   274→}

   307→// ------------------------------------------------------------------- road
   308→function RoadRibbon({ road }: { road: RoadData }) {
   309→  const asphalt = useMemo(() => {
   310→    const g = new THREE.BufferGeometry();
   311→    const n = ROAD_POINTS;
   312→    const pos = new Float32Array(n * 2 * 3);
   313→    const nor = new Float32Array(n * 2 * 3);
   314→    const idx: number[] = [];
   315→    for (let i = 0; i < n; i++) {
   316→      const theta = road.heading[i];
   317→      const nx = Math.cos(theta);
   318→      const nz = -Math.sin(theta);
   319→      const x = road.px[i];
   320→      const z = road.pz[i];
   321→      pos.set([x + nx * ROAD_HALF_WIDTH, 0.03, z + nz * ROAD_HALF_WIDTH], i * 6);
   322→      pos.set([x - nx * ROAD_HALF_WIDTH, 0.03, z - nz * ROAD_HALF_WIDTH], i * 6 + 3);
   323→      nor.set([0, 1, 0], i * 6);
   324→      nor.set([0, 1, 0], i * 6 + 3);
   325→      if (i < n - 1) {
   326→        const a = i * 2;
   327→        idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
   328→      }
   329→    }
   330→    g.setAttribute("position", new THREE.BufferAttribute(pos, 3));
   331→    g.setAttribute("normal", new THREE.BufferAttribute(nor, 3));
   332→    g.setIndex(idx);
   333→    return g;
   334→  }, [road]);
   335→
   336→  const dashes = useMemo(() => {
   337→    // warm dashed centre line: 1.5 m dash, 2.5 m gap
   338→    const pos: number[] = [];
   339→    const idx: number[] = [];
   340→    let v = 0;
   341→    for (let s = 2; s < 2395; s += 4) {
   342→      const i0 = Math.floor(s);
   343→      const i1 = Math.min(Math.floor(s + 1.5), ROAD_POINTS - 1);
   344→      if (i1 <= i0) continue;
   345→      for (const i of [i0, i1]) {
   346→        const theta = road.heading[i];
   347→        const nx = Math.cos(theta);
   348→        const nz = -Math.sin(theta);
   349→        pos.push(
   350→          road.px[i] + nx * 0.09, 0.045, road.pz[i] + nz * 0.09,
   351→          road.px[i] - nx * 0.09, 0.045, road.pz[i] - nz * 0.09
   352→        );
   353→      }
   354→      idx.push(v, v + 1, v + 2, v + 1, v + 3, v + 2);
   355→      v += 4;
   356→    }
   357→    const g = new THREE.BufferGeometry();
   358→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   359→    g.setIndex(idx);
   360→    g.computeVertexNormals();
   361→    return g;
   362→  }, [road]);
   363→
   364→  const edges = useMemo(() => {
   365→    // dim warm edge lines at ±2.32 m
   366→    const pos: number[] = [];
   367→    const idx: number[] = [];
   368→    let v = 0;
   369→    for (const side of [1, -1]) {
   370→      for (let i = 0; i < ROAD_POINTS - 1; i++) {
   371→        const theta = road.heading[i];
   372→        const nx = Math.cos(theta);
   373→        const nz = -Math.sin(theta);
   374→        const cx = road.px[i] + nx * 2.32 * side;
   375→        const cz = road.pz[i] + nz * 2.32 * side;
   376→        const theta2 = road.heading[i + 1];
   377→        const nx2 = Math.cos(theta2);
   378→        const nz2 = -Math.sin(theta2);
   379→        const cx2 = road.px[i + 1] + nx2 * 2.32 * side;
   380→        const cz2 = road.pz[i + 1] + nz2 * 2.32 * side;
   381→        pos.push(
   382→          cx + nx * 0.07, 0.04, cz + nz * 0.07,
   383→          cx - nx * 0.07, 0.04, cz - nz * 0.07,
   384→          cx2 + nx2 * 0.07, 0.04, cz2 + nz2 * 0.07,
   385→          cx2 - nx2 * 0.07, 0.04, cz2 - nz2 * 0.07
   386→        );
   387→        idx.push(v, v + 1, v + 2, v + 1, v + 3, v + 2);
   388→        v += 4;
   389→      }
   390→    }
   391→    const g = new THREE.BufferGeometry();
   392→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   393→    g.setIndex(idx);
   394→    g.computeVertexNormals();
   395→    return g;
   396→  }, [road]);
   397→
   398→  return (
   399→    <group>
   400→      <mesh geometry={asphalt}>
   401→        <meshStandardMaterial color={C.asphalt} roughness={0.94} metalness={0} />
   402→      </mesh>
   403→      <mesh geometry={dashes}>
   404→        <meshBasicMaterial color={C.dash} toneMapped={false} />
   405→      </mesh>
   406→      <mesh geometry={edges}>
   407→        <meshBasicMaterial color={C.edge} toneMapped={false} transparent opacity={0.75} />
   408→      </mesh>
   409→    </group>
   410→  );
   411→}

   413→function GlowPosts({ road }: { road: RoadData }) {
   414→  const group = useMemo(() => {
   415→    const m = new THREE.Matrix4();
   416→    const posts: { x: number; z: number }[] = [];
   417→    for (let s = POST_SPACING_M; s < 2395; s += POST_SPACING_M) {
   418→      const i = Math.floor(s);
   419→      const theta = road.heading[i];
   420→      const nx = Math.cos(theta);
   421→      const nz = -Math.sin(theta);
   422→      for (const side of [1, -1]) {
   423→        posts.push({
   424→          x: road.px[i] + nx * POST_OFFSET_M * side,
   425→          z: road.pz[i] + nz * POST_OFFSET_M * side,
   426→        });
   427→      }
   428→    }
   429→    const orbMesh = new THREE.InstancedMesh(
   430→      new THREE.SphereGeometry(0.11, 10, 8),
   431→      new THREE.MeshBasicMaterial({ color: C.post, toneMapped: false }),
   432→      posts.length
   433→    );
   434→    const stickMesh = new THREE.InstancedMesh(
   435→      new THREE.CylinderGeometry(0.025, 0.03, 0.62, 5),
   436→      new THREE.MeshStandardMaterial({ color: "#3a2530", roughness: 1 }),
   437→      posts.length
   438→    );
   439→    posts.forEach((p, i) => {
   440→      m.makeTranslation(p.x, 0.62, p.z);
   441→      orbMesh.setMatrixAt(i, m);
   442→      m.makeTranslation(p.x, 0.31, p.z);
   443→      stickMesh.setMatrixAt(i, m);
   444→    });
   445→    orbMesh.instanceMatrix.needsUpdate = true;
   446→    stickMesh.instanceMatrix.needsUpdate = true;
   447→    const g = new THREE.Group();
   448→    g.add(orbMesh, stickMesh);
   449→    return g;
   450→  }, [road]);
   451→  return <primitive object={group} />;
   452→}

   454→// -------------------------------------------------------------- fireflies
   455→function Fireflies({ road }: { road: RoadData }) {
   456→  const ref = useRef<THREE.Points>(null);
   457→  const COUNT = 900;
   458→  const { geometry, seeds } = useMemo(() => {
   459→    const rand = mulberry32(99117);
   460→    const positions = new Float32Array(COUNT * 3);
   461→    const colors = new Float32Array(COUNT * 3);
   462→    const seeds = new Float32Array(COUNT * 2);
   463→    const base = new THREE.Color(C.firefly);
   464→    for (let i = 0; i < COUNT; i++) {
   465→      const s = rand() * 2390;
   466→      const { x, z, theta } = road.sample(s);
   467→      const lat = (rand() < 0.5 ? -1 : 1) * (3.5 + rand() * 24);
   468→      positions[i * 3] = x - Math.sin(theta) * lat;
   469→      positions[i * 3 + 1] = 0.4 + rand() * 5;
   470→      positions[i * 3 + 2] = z + Math.cos(theta) * lat;
   471→      const warmth = 0.75 + rand() * 0.25;
   472→      colors[i * 3] = base.r * warmth;
   473→      colors[i * 3 + 1] = base.g * warmth * (0.85 + rand() * 0.15);
   474→      colors[i * 3 + 2] = base.b * warmth * 0.7;
   475→      seeds[i * 2] = rand() * Math.PI * 2; // phase
   476→      seeds[i * 2 + 1] = 0.3 + rand() * 0.8; // speed
   477→    }
   478→    const geometry = new THREE.BufferGeometry();
   479→    geometry.setAttribute("position", new THREE.BufferAttribute(positions, 3));
   480→    geometry.setAttribute("color", new THREE.BufferAttribute(colors, 3));
   481→    return { geometry, seeds };
   482→  }, [road]);
   483→
   484→  const basePos = useMemo(
   485→    () => Float32Array.from(geometry.attributes.position.array as Float32Array),
   486→    [geometry]
   487→  );
   488→
   489→  useFrame(({ clock }) => {
   490→    const t = clock.elapsedTime;
   491→    const attr = ref.current?.geometry.attributes.position as
   492→      | THREE.BufferAttribute
   493→      | undefined;
   494→    if (!attr) return;
   495→    const arr = attr.array as Float32Array;
   496→    for (let i = 0; i < COUNT; i++) {
   497→      const ph = seeds[i * 2];
   498→      const sp = seeds[i * 2 + 1];
   499→      arr[i * 3] = basePos[i * 3] + Math.sin(t * sp * 0.5 + ph) * 0.9;
   500→      arr[i * 3 + 1] = basePos[i * 3 + 1] + Math.sin(t * sp + ph * 2) * 0.55;
   501→      arr[i * 3 + 2] = basePos[i * 3 + 2] + Math.cos(t * sp * 0.4 + ph) * 0.9;
   502→    }
   503→    attr.needsUpdate = true;
   504→  });
   505→
   506→  return (
   507→    <points ref={ref} geometry={geometry} frustumCulled={false}>

   514→        depthWrite={false}
   515→        sizeAttenuation
   516→        toneMapped={false}
   517→      />
   518→    </points>
   519→  );

   564→const BIKE_R = 0.34; // wheel radius
   565→
   566→function BikeRig({
   567→  core,
   568→  riderIndex,
   569→}: {
   570→  core: BicycleTrainerCore;
   571→  riderIndex: number;
   572→}) {
   573→  const root = useRef<THREE.Group>(null);
   574→  const lean = useRef<THREE.Group>(null);
   575→  const wheelF = useRef<THREE.Group>(null);
   576→  const wheelR = useRef<THREE.Group>(null);
   577→  const crank = useRef<THREE.Group>(null);
   578→  const legL = useRef<THREE.Mesh>(null);
   579→  const legR = useRef<THREE.Mesh>(null);
   580→  const rim = useRef<THREE.Mesh>(null);
   581→  const lamp = useRef<THREE.PointLight>(null);
   582→  const wingL = useRef<THREE.Mesh>(null);
   583→  const wingR = useRef<THREE.Mesh>(null);
   584→
   585→  // every rig owns its own material set so ghosts / fades are per-rider
   586→  // (mutated imperatively each frame, outside React's data flow)
   587→  const mats = useMemo(() => makeRigMaterials(), []);
   588→
   589→  // bookkeeping for fall animation + episode resets
   590→  const wasAlive = useRef(true);
   591→  const crankAngle = useRef(0);
   592→  const wheelAngle = useRef(0);
   593→
   594→  useFrame((_, dt) => {
   595→    const r = core.riders[riderIndex];
   596→    const g = root.current;
   597→    const lg = lean.current;
   598→    if (!g || !lg) return;
   599→    if (!r) {
   600→      // population smaller than the rig pool — park it out of sight
   601→      g.visible = false;
   602→      return;
   603→    }
   604→    g.visible = true;
   605→
   606→    // --- transform from physics state ---
   607→    const w = core.riderWorld(riderIndex);
   608→    g.position.set(w.x, 0, w.z);
   609→    g.rotation.y = w.heading;
   610→
   611→    // --- ghost vs solid vs leader materials ---
   612→    const isLeader = riderIndex === core.leaderIdx;
   613→    const ghost = !isLeader && core.riders.length > 1;
   614→    const bodyMats = [mats.frame, mats.tire, mats.dark, mats.body];
   615→    if (r.st.alive) {
   616→      const target = ghost ? 0.28 : 1;
   617→      bodyMats.forEach((m) => {
   618→        m.transparent = ghost;
   619→        m.opacity += (target - m.opacity) * Math.min(1, dt * 6);
   620→        m.depthWrite = !ghost;
   621→      });
   622→      [mats.eye].forEach((m) => {
   623→        m.transparent = ghost;
   624→        m.opacity += ((ghost ? 0.45 : 1) - m.opacity) * Math.min(1, dt * 6);
   625→      });
   626→      [mats.wing].forEach((m) => {
   627→        m.opacity += ((ghost ? 0.12 : 0.32) - m.opacity) * Math.min(1, dt * 6);
   628→      });
   629→      lg.rotation.z = w.lean;
   630→    } else {
   631→      // tumble: tip over 0.5 s, then fade out over ~1.2 s
   632→      const tFall = core.realTime - r.doneRealT;
   633→      const tip = Math.min(1, tFall / 0.5);
   634→      const ease = tip * tip * (3 - 2 * tip);
   635→      lg.rotation.z = r.st.fallSign * (Math.PI / 2) * ease;
   636→      const fade = Math.max(0, 1 - Math.max(0, tFall - 0.45) / 1.2);
   637→      [...bodyMats, mats.eye].forEach((m) => {
   638→        m.transparent = true;
   639→        m.opacity = fade * (ghost ? 0.28 : 1);
   640→        m.depthWrite = fade > 0.5;
   641→      });
   642→      [mats.wing].forEach((m) => { m.opacity = fade * 0.3; });
   643→    }
   644→    if (r.st.alive && !wasAlive.current) {
   645→      // episode reset — restore upright pose and solid materials
   646→      lg.rotation.z = 0;
   647→      [mats.frame, mats.tire, mats.dark, mats.body, mats.eye].forEach((m) => {
   648→        m.opacity = 1;
   649→        m.transparent = false;
   650→        m.depthWrite = true;
   651→      });
   652→      [mats.wing].forEach((m) => { m.opacity = 0.32; });
   653→    }
   654→    wasAlive.current = r.st.alive;
   655→
   656→    // --- wheels & pedalling legs ---
   657→    const v = r.st.alive ? r.st.v : 0;
   658→    wheelAngle.current += (v / BIKE_R) * dt;
   659→    crankAngle.current += v * 1.15 * dt;
   660→    if (wheelF.current) wheelF.current.rotation.z = wheelAngle.current;
   661→    if (wheelR.current) wheelR.current.rotation.z = wheelAngle.current;
   662→    if (crank.current) crank.current.rotation.x = crankAngle.current;
   663→    const ph = crankAngle.current;
   664→    if (legL.current) legL.current.rotation.x = -0.35 + Math.sin(ph) * 0.55;
   665→    if (legR.current) legR.current.rotation.x = -0.35 + Math.sin(ph + Math.PI) * 0.55;
   666→    // wing flutter
   667→    const flut = Math.sin(core.realTime * 38 + riderIndex * 3) * 0.45;
   668→    if (wingL.current) wingL.current.rotation.y = 0.7 + flut;
   669→    if (wingR.current) wingR.current.rotation.y = -0.7 - flut;
   670→
   671→    // --- leader highlight: pulsing rose rim + warm lamp ---
   672→    if (rim.current) {
   673→      rim.current.visible = isLeader;
   674→      const s = 1 + Math.sin(core.realTime * 3.2) * 0.06;
   675→      rim.current.scale.set(s, s, 1);
   676→    }
   677→    if (lamp.current) lamp.current.intensity = isLeader ? 2.4 : 0;
   678→  });
   679→
   680→  const tube = (
   681→    from: [number, number, number],
   682→    to: [number, number, number],
   683→    r = 0.035,
   684→    mat: THREE.Material
   685→  ) => {
   686→    const dx = to[0] - from[0];
   687→    const dy = to[1] - from[1];
   688→    const dz = to[2] - from[2];
   689→    const len = Math.sqrt(dx * dx + dy * dy + dz * dz);
   690→    const mid: [number, number, number] = [
   691→      (from[0] + to[0]) / 2,
   692→      (from[1] + to[1]) / 2,
   693→      (from[2] + to[2]) / 2,
   694→    ];
   695→    const quat = new THREE.Quaternion().setFromUnitVectors(
   696→      new THREE.Vector3(0, 1, 0),
   697→      new THREE.Vector3(dx, dy, dz).normalize()
   698→    );
   699→    const e = new THREE.Euler().setFromQuaternion(quat);
   700→    return (
   701→      <mesh position={mid} rotation={[e.x, e.y, e.z]} material={mat}>
   702→        <cylinderGeometry args={[r, r, len, 6]} />
   703→      </mesh>
   704→    );
   705→  };
   706→
   707→  const rearHub: [number, number, number] = [0, BIKE_R, -0.55];
   708→  const frontHub: [number, number, number] = [0, BIKE_R, 0.55];
   709→  const bb: [number, number, number] = [0, 0.3, 0.02];
   710→  const seatTop: [number, number, number] = [0, 0.98, -0.22];
   711→  const headTop: [number, number, number] = [0, 0.95, 0.42];
   712→
   713→  return (
   714→    <group ref={root}>
   715→      <group ref={lean}>
   716→        {/* wheels */}
   717→        <group position={rearHub} rotation={[0, Math.PI / 2, 0]}>
   718→          <group ref={wheelR}>
   719→            <mesh material={mats.tire}>
   720→              <torusGeometry args={[BIKE_R, 0.045, 8, 20]} />
   721→            </mesh>
   722→            {[0, 1, 2].map((k) => (
   723→              <mesh key={k} rotation={[0, 0, (k * Math.PI) / 3]} material={mats.dark}>
   724→                <boxGeometry args={[0.016, BIKE_R * 2 - 0.06, 0.016]} />
   725→              </mesh>
   726→            ))}
   727→          </group>
   728→        </group>
   729→        <group position={frontHub} rotation={[0, Math.PI / 2, 0]}>
   730→          <group ref={wheelF}>
   731→            <mesh material={mats.tire}>
   732→              <torusGeometry args={[BIKE_R, 0.045, 8, 20]} />
   733→            </mesh>
   734→            {[0, 1, 2].map((k) => (
   735→              <mesh key={k} rotation={[0, 0, (k * Math.PI) / 3]} material={mats.dark}>
   736→                <boxGeometry args={[0.016, BIKE_R * 2 - 0.06, 0.016]} />
   737→              </mesh>
   738→            ))}
   739→          </group>
   740→        </group>
   741→
   742→        {/* frame */}
   743→        {tube(rearHub, seatTop, 0.035, mats.frame)}
   744→        {tube(seatTop, headTop, 0.032, mats.frame)}
   745→        {tube(rearHub, bb, 0.028, mats.dark)}
   746→        {tube(bb, seatTop, 0.032, mats.frame)}
   747→        {tube(bb, headTop, 0.03, mats.frame)}
   748→        {tube(headTop, frontHub, 0.026, mats.frame)}
   749→        {/* handlebar */}
   750→        <mesh position={[0, 1.02, 0.44]} rotation={[0, 0, Math.PI / 2]} material={mats.dark}>
   751→          <cylinderGeometry args={[0.022, 0.022, 0.52, 6]} />
   752→        </mesh>
   753→        {/* seat */}
   754→        <mesh position={[0, 1.0, -0.24]} material={mats.dark}>
   755→          <boxGeometry args={[0.09, 0.05, 0.24]} />
   756→        </mesh>
   757→
   758→        {/* cranks + pedals */}
   759→        <group ref={crank} position={[0, 0.3, 0.02]}>
   760→          <mesh position={[0.1, 0, 0]} material={mats.dark}>
   761→            <boxGeometry args={[0.16, 0.022, 0.03]} />
   762→          </mesh>
   763→          <mesh position={[-0.1, 0, 0]} material={mats.dark}>
   764→            <boxGeometry args={[0.16, 0.022, 0.03]} />
   765→          </mesh>
   766→        </group>
   767→
   768→        {/* the FLY rider */}
   769→        <group position={[0, 1.06, -0.18]}>
   770→          <mesh position={[0, 0.14, 0]} material={mats.body}>
   771→            <sphereGeometry args={[0.16, 12, 10]} />
   772→          </mesh>
   773→          {/* big red compound eyes */}
   774→          <mesh position={[0.1, 0.26, 0.05]} material={mats.eye}>
   775→            <sphereGeometry args={[0.078, 10, 8]} />
   776→          </mesh>
   777→          <mesh position={[-0.1, 0.26, 0.05]} material={mats.eye}>
   778→            <sphereGeometry args={[0.078, 10, 8]} />
   779→          </mesh>
   780→          {/* wings */}
   781→          <mesh ref={wingL} position={[0.06, 0.22, -0.1]} rotation={[0.5, 0.7, 0.2]} material={mats.wing}>
   782→            <planeGeometry args={[0.34, 0.13]} />
   783→          </mesh>
   784→          <mesh ref={wingR} position={[-0.06, 0.22, -0.1]} rotation={[0.5, -0.7, -0.2]} material={mats.wing}>
   785→            <planeGeometry args={[0.34, 0.13]} />
   786→          </mesh>
   787→          {/* antennae */}
   788→          <mesh position={[0.03, 0.36, 0.12]} rotation={[0.5, 0, 0.4]} material={mats.dark}>
   789→            <cylinderGeometry args={[0.006, 0.006, 0.16, 4]} />
   790→          </mesh>
   791→          <mesh position={[-0.03, 0.36, 0.12]} rotation={[0.5, 0, -0.4]} material={mats.dark}>
   792→            <cylinderGeometry args={[0.006, 0.006, 0.16, 4]} />
   793→          </mesh>
   794→          {/* arms to handlebar */}
   795→          <mesh position={[0.1, -0.04, 0.28]} rotation={[1.1, 0, 0.25]} material={mats.body}>
   796→            <cylinderGeometry args={[0.02, 0.02, 0.34, 5]} />
   797→          </mesh>
   798→          <mesh position={[-0.1, -0.04, 0.28]} rotation={[1.1, 0, -0.25]} material={mats.body}>
   799→            <cylinderGeometry args={[0.02, 0.02, 0.34, 5]} />
   800→          </mesh>
   801→          {/* pedalling legs */}
   802→          <mesh ref={legL} position={[0.07, -0.1, 0.06]} material={mats.body}>
   803→            <cylinderGeometry args={[0.018, 0.022, 0.4, 5]} />
   804→          </mesh>
   805→          <mesh ref={legR} position={[-0.07, -0.1, 0.06]} material={mats.body}>
   806→            <cylinderGeometry args={[0.018, 0.022, 0.4, 5]} />
   807→          </mesh>
   808→        </group>
   809→      </group>
   810→
   811→      {/* leader glow rim under the bike */}
   812→      <mesh ref={rim} position={[0, 0.06, 0]} rotation={[-Math.PI / 2, 0, 0]} visible={false}>
   813→        <ringGeometry args={[0.62, 0.72, 28]} />
   814→        <meshBasicMaterial color={C.rim} transparent opacity={0.85} toneMapped={false} side={THREE.DoubleSide} />

   817→    </group>
   818→  );

   821→// ----------------------------------------------------------- chase camera
   822→function ChaseCamera({ core }: { core: BicycleTrainerCore }) {
   823→  const { camera } = useThree();
   824→  const camPos = useRef(new THREE.Vector3(0, 2.8, -8));
   825→  const camTgt = useRef(new THREE.Vector3(0, 1, 3));
   826→  const desired = useRef(new THREE.Vector3());
   827→  const tgt = useRef(new THREE.Vector3());
   828→
   829→  useFrame((_, dt) => {
   830→    const lead = core.leader;
   831→    if (!lead) return;
   832→    const w = core.riderWorld(core.leaderIdx);
   833→    const fwdX = Math.sin(w.heading);
   834→    const fwdZ = Math.cos(w.heading);
   835→
   836→    desired.current.set(w.x - fwdX * 7.2, 2.6 + Math.max(0, w.lean) * 0, w.z - fwdZ * 7.2);
   837→    tgt.current.set(w.x + fwdX * 3.4, 1.05, w.z + fwdZ * 3.4);
   838→
   839→    const k = 1 - Math.exp(-dt * 2.6);
   840→    camPos.current.lerp(desired.current, k);
   841→    camTgt.current.lerp(tgt.current, 1 - Math.exp(-dt * 4));
   842→
   843→    // fall shake
   844→    let sx = 0;
   845→    let sy = 0;
   846→    if (core.shake > 0.005) {
   847→      sx = (Math.random() - 0.5) * core.shake;
   848→      sy = (Math.random() - 0.5) * core.shake * 0.7;
   849→    }
   850→    camera.position.set(camPos.current.x + sx, camPos.current.y + sy, camPos.current.z);
   851→    camera.up.set(0, 1, 0);
   852→    camera.lookAt(camTgt.current);
   853→    // gentle banking with the rider's lean
   854→    camera.rotateZ(-w.lean * 0.16);
   855→  });
   856→  return null;
   857→}

   859→// ------------------------------------------------------------- sim runner
   860→function SimRunner({
   861→  core,
   862→  retinaCanvas,
   863→}: {
   864→  core: BicycleTrainerCore;
   865→  retinaCanvas: React.RefObject<HTMLCanvasElement | null>;
   866→}) {
   867→  // pixel buffer for the retina mini-canvas — lazily created, mutated per frame
   868→  const bufRef = useRef<{
   869→    img: ImageData;
   870→    px: Uint8ClampedArray;
   871→  } | null>(null);
   872→
   873→  useFrame((_, dt) => {
   874→    core.advance(Math.min(dt, 0.1));
   875→
   876→    const cv = retinaCanvas.current;
   877→    const lead = core.leader;
   878→    if (cv && lead && typeof ImageData !== "undefined") {
   879→      const ctx = cv.getContext("2d");
   880→      if (ctx) {
   881→        if (!bufRef.current) {
   882→          bufRef.current = {
   883→            img: new ImageData(RETINA_COLS, RETINA_ROWS),
   884→            px: new Uint8ClampedArray(RETINA_COLS * RETINA_ROWS * 4),
   885→          };
   886→        }
   887→        const { img, px } = bufRef.current;
   888→        const r = lead.retina;
   889→        for (let i = 0; i < RETINA_COLS * RETINA_ROWS; i++) {
   890→          const [cr, cg, cb] = retinaColor(r[i]);
   891→          px[i * 4] = cr;
   892→          px[i * 4 + 1] = cg;
   893→          px[i * 4 + 2] = cb;
   894→          px[i * 4 + 3] = 255;
   895→        }
   896→        img.data.set(px);
   897→        ctx.putImageData(img, 0, 0);
   898→      }
   899→    }
   900→  });
   901→  return null;
   902→}
