   112→function sanitizeFileName(name: string): string {
   113→  const clean = name
   114→    .toLowerCase()
   115→    .replace(/[^a-z0-9-_]+/g, "-")
   116→    .replace(/^-+|-+$/g, "");
   117→  return clean || "brain";
   118→}

   141→export interface GenomeGap {
   142→  index: number;
   143→  label: string;
   144→  /** full annotation sentence, e.g. `M4 — 2.3× stronger in "Lab Demo Fly"` */
   145→  text: string;
   146→}

   148→/** Find the compartment with the largest |A lean − B lean| and describe it. */
   149→function biggestGap(sa: FingerprintStats, sb: FingerprintStats, nameA: string, nameB: string): GenomeGap | null {
   150→  const n = Math.min(sa.leans.length, sb.leans.length);
   151→  let idx = -1;
   152→  let best = 0;
   153→  for (let i = 0; i < n; i++) {
   154→    const d = Math.abs(sa.leans[i] - sb.leans[i]);
   155→    if (d > best) {
   156→      best = d;
   157→      idx = i;
   158→    }
   159→  }
   160→  if (idx < 0 || best < UNWRITTEN_EPS) return null; // identical fingerprints
   161→
   162→  const a = sa.leans[idx];
   163→  const b = sb.leans[idx];
   164→  const label = `M${idx + 1}`;
   165→  const magA = Math.abs(a);
   166→  const magB = Math.abs(b);
   167→  const aUnwritten = magA < UNWRITTEN_EPS;
   168→  const bUnwritten = magB < UNWRITTEN_EPS;
   169→
   170→  if (aUnwritten && bUnwritten) {
   171→    return { index: idx, label, text: `${label} — effectively unwritten in both brains` };
   172→  }
   173→  if (aUnwritten || bUnwritten) {
   174→    const writer = aUnwritten ? nameB : nameA;
   175→    const lean = aUnwritten ? b : a;
   176→    return {
   177→      index: idx,
   178→      label,
   179→      text: `${label} — written only by “${writer}” (${fmtLean(lean)}, ${
   180→        lean >= 0 ? "reward" : "punishment"
   181→      }-shaped)`,
   182→    };
   183→  }
   184→  if (a >= 0 === b >= 0) {
   185→    // same valence → magnitude ratio
   186→    const hi = Math.max(magA, magB);
   187→    const lo = Math.min(magA, magB);
   188→    const winner = magA >= magB ? nameA : nameB;
   189→    return {
   190→      index: idx,
   191→      label,
   192→      text: `${label} — ${(hi / lo).toFixed(1)}× stronger in “${winner}”`,
   193→    };
   194→  }
   195→  // opposite signs → the compartment flips valence between the two brains
   196→  return {
   197→    index: idx,
   198→    label,
   199→    text: `${label} — flips valence: “${nameA}” writes it ${a >= 0 ? "reward" : "punishment"}-shaped, “${nameB}” ${
   200→      b >= 0 ? "reward" : "punishment"
   201→    }-shaped`,
   202→  };
   203→}
   204→
   205→function buildVerdict(
   206→  rowA: BrainRow,
   207→  rowB: BrainRow,
   208→  sa: FingerprintStats,
   209→  sb: FingerprintStats,
   210→  gap: GenomeGap | null,
   211→): string {
   212→  const parts: string[] = [];
   213→  const dBal = sa.balance - sb.balance;
   214→  if (sa.kmCount === 0 && sb.kmCount === 0) {
   215→    parts.push("Neither brain carries learned Kenyon→MBON writes yet — there is nothing to diff.");
   216→  } else if (Math.abs(dBal) <= 0.01) {
   217→    parts.push(
   218→      `“${rowA.name}” and “${rowB.name}” are evenly matched on valence — neither leans harder toward reward or punishment (balance ${fmtLean(
   219→        sa.balance,
   220→      )} vs ${fmtLean(sb.balance)}).`,
   221→    );
   222→  } else {
   223→    // dBal > 0 → A leans reward harder (or punishment less hard) than B
   224→    const winner = dBal > 0 ? rowA : rowB;
   225→    const leanWord = (v: number) =>
   226→      v > 0.01
   227→        ? `leans reward ${fmtLean(v)}`
   228→        : v < -0.01
   229→          ? `leans punishment ${fmtLean(v)}`
   230→          : `stays nearly balanced ${fmtLean(v)}`;
   231→    parts.push(
   232→      `“${rowA.name}” ${leanWord(sa.balance)}, while “${rowB.name}” ${leanWord(sb.balance)} — “${
   233→        winner.name
   234→      }” is the more reward-shaped of the two (difference ${fmtLean(Math.abs(dBal))}).`,
   235→    );
   236→  }
   237→  if (gap) {
   238→    parts.push(`Their sharpest wiring difference is ${gap.text.replace(/^(\w+) — /, "$1: ")}.`);
   239→  }
   240→  return parts.join(" ");
   241→}

   243→// ---------------------------------------------------------------------------
   244→// plain-text diff (clipboard / fallback download)

   246→
   247→function buildDiffText(
   248→  rowA: BrainRow,
   249→  rowB: BrainRow,
   250→  snapA: BrainSnapshot,
   251→  snapB: BrainSnapshot,
   252→  sa: FingerprintStats,
   253→  sb: FingerprintStats,
   254→  gap: GenomeGap | null,
   255→): string {
   256→  const n = Math.min(sa.leans.length, sb.leans.length);
   257→  const balanceWord = (s: FingerprintStats) =>
   258→    Math.abs(s.balance) < 0.002 ? "balanced (naive)" : s.balance > 0 ? "leans reward" : "leans punishment";
   259→
   260→  const L: string[] = [];
   261→  L.push("EXPERT FRUIT FLY — GENOME DIFF");
   262→  L.push("=".repeat(40));
   263→  L.push("");
   264→  L.push("IDENTITY");
   265→  L.push(`  A  ${rowA.name} — ${rowA.task} · gen ${rowA.generation} · ${formatScore(rowA.task, rowA.score)} · ${formatBytes(rowA.snapshotBytes)} · ${relativeDate(rowA.createdAt)}`);
   266→  if (rowA.note) L.push(`     note: ${rowA.note}`);
   267→  L.push(`  B  ${rowB.name} — ${rowB.task} · gen ${rowB.generation} · ${formatScore(rowB.task, rowB.score)} · ${formatBytes(rowB.snapshotBytes)} · ${relativeDate(rowB.createdAt)}`);
   268→  if (rowB.note) L.push(`     note: ${rowB.note}`);
   269→  L.push("");
   270→  L.push("LEARNED WRITES BY MEMORY CHANNEL (mean Kenyon→MBON × valence)");
   271→  L.push("  Channel        A         B         Δ (A−B)");
   272→  for (let i = 0; i < n; i++) {
   273→    const a = sa.leans[i];
   274→    const b = sb.leans[i];
   275→    L.push(
   276→      `  M${String(i + 1).padEnd(11)} ${(a >= 0 ? "+" : "-") + Math.abs(a).toFixed(3).padStart(6)}  ${
   277→        (b >= 0 ? "+" : "-") + Math.abs(b).toFixed(3).padStart(6)
   278→      }  ${(a - b >= 0 ? "+" : "-") + Math.abs(a - b).toFixed(3).padStart(6)}`,
   279→    );
   280→  }
   281→  if (gap) L.push(`  Biggest gap  ${gap.text}`);
   282→  L.push("");
   283→  L.push("WIRING");
   284→  L.push(`  A  ${nums(sa.total)} total — plastic ${nums(sa.plastic)} (K→M ${nums(sa.kmCount)} · M→motor ${nums(sa.mmCount)} · giant-fiber ${nums(sa.gfCount)} · biases ${nums(sa.biasCount)}) / innate ${nums(sa.innate)}`);
   285→  L.push(`  B  ${nums(sb.total)} total — plastic ${nums(sb.plastic)} (K→M ${nums(sb.kmCount)} · M→motor ${nums(sb.mmCount)} · giant-fiber ${nums(sb.gfCount)} · biases ${nums(sb.biasCount)}) / innate ${nums(sb.innate)}`);
   286→  L.push(`  Training     A ${nums(snapA.meta?.steps ?? 0)} steps · ${snapA.meta?.rewards ?? 0} rewards / ${snapA.meta?.punishments ?? 0} punishments`);
   287→  L.push(`              B ${nums(snapB.meta?.steps ?? 0)} steps · ${snapB.meta?.rewards ?? 0} rewards / ${snapB.meta?.punishments ?? 0} punishments`);
   288→  L.push("");
   289→  L.push("VALENCE");
   290→  L.push(`  A  balance ${fmtLean(sa.balance)} — ${balanceWord(sa)} (reward mass ${sa.rewardMass.toFixed(3)} vs punishment ${sa.punishMass.toFixed(3)})`);
   291→  L.push(`  B  balance ${fmtLean(sb.balance)} — ${balanceWord(sb)} (reward mass ${sb.rewardMass.toFixed(3)} vs punishment ${sb.punishMass.toFixed(3)})`);
   292→  const dBal = sa.balance - sb.balance;
   293→  if (Math.abs(dBal) <= 0.01) {
   294→    L.push("     → evenly matched on valence");
   295→  } else {
   296→    L.push(`     → “${dBal > 0 ? rowA.name : rowB.name}” is the more reward-shaped brain`);
   297→  }
   298→  L.push("");
   299→  L.push("VERDICT");
   300→  L.push(`  ${buildVerdict(rowA, rowB, sa, sb, gap)}`);
   301→  L.push("");
   302→  L.push(`— Expert Fruit Fly genome diff · generated ${new Date().toISOString()}`);
   303→  return L.join("\n");
   304→}

   310→function DiffFingerprintChart({
   311→  nameA,
   312→  nameB,
   313→  sa,
   314→  sb,
   315→  gap,
   316→}: {
   317→  nameA: string;
   318→  nameB: string;
   319→  sa: FingerprintStats;
   320→  sb: FingerprintStats;
   321→  gap: GenomeGap | null;
   322→}) {
   323→  const n = Math.min(sa.leans.length, sb.leans.length);
   324→  const appHalf = Math.ceil(n / 2);
   325→  // ONE shared normalization scale across both brains, so cell intensities
   326→  // are directly comparable between the A and B strips.
   327→  const maxAbs = Math.max(UNWRITTEN_EPS, ...sa.leans.map(Math.abs), ...sb.leans.map(Math.abs));
   328→  const deltas = sa.leans.slice(0, n).map((v, i) => v - (sb.leans[i] ?? 0));
   329→  const maxDelta = Math.max(UNWRITTEN_EPS, ...deltas.map(Math.abs));
   330→
   331→  // geometry — shared by all three strips so columns align exactly
   332→  const x0 = 118;
   333→  const cellW = 34;
   334→  const cellGap = 2;
   335→  const step = cellW + cellGap;
   336→  const cellX = (i: number) => x0 + i * step;
   337→  const rowA_Y = 29;
   338→  const rowB_Y = 57;
   339→  const rowD_Y = 95;
   340→  const rowH = 22;
   341→
   342→  const opa = (mag: number, max: number) => 0.14 + 0.78 * Math.min(1, mag / max);
   343→
   344→  return (
   345→    <svg
   346→      viewBox={`0 0 ${SVG_W} ${SVG_H}`}
   347→      className="h-auto w-full"
   348→      role="img"
   349→      aria-label={`Genome diff fingerprint: rows A (${nameA}) and B (${nameB}) show each memory channel's learned lean — green is reward-shaped, red punishment-shaped. The third row is the per-channel difference: green means A is stronger, red means B is stronger.`}
   350→      fontFamily="system-ui, -apple-system, 'Segoe UI', sans-serif"
   351→    >
   352→      {/* panel */}
   353→      <rect x={0} y={0} width={SVG_W} height={SVG_H} rx={10} fill={INK} />
   354→
   355→      {/* compartment group headers + washes (reward-side left, punishment-side right) */}
   356→      <text x={x0 + (appHalf * step - cellGap) / 2} y={15} textAnchor="middle" fontSize={9} fontWeight={600} letterSpacing={0.6} fill={EM_SOFT} opacity={0.9}>
   357→        REWARD-SIDE (M1–M{appHalf})
   358→      </text>
   359→      <text x={cellX(appHalf) + (appHalf * step - cellGap) / 2} y={15} textAnchor="middle" fontSize={9} fontWeight={600} letterSpacing={0.6} fill={ROSE_SOFT} opacity={0.9}>
   360→        PUNISHMENT-SIDE (M{appHalf + 1}–M{n})
   361→      </text>
   362→      <rect x={x0 - 2} y={21} width={appHalf * step} height={58} rx={6} fill={EM} opacity={0.06} />
   363→      <rect x={cellX(appHalf) - 2} y={21} width={(n - appHalf) * step} height={58} rx={6} fill={ROSE} opacity={0.06} />
   364→
   365→      {/* ---- left gutter labels ---- */}
   366→      <rect x={4} y={rowA_Y + 3} width={14} height={14} rx={4} fill={EM} opacity={0.85} />
   367→      <text x={23} y={rowA_Y + 14.5} fontSize={9.5} fontWeight={600} fill="#d4d4d8">
   368→        {svgName(nameA)}
   369→      </text>
   370→      <rect x={4} y={rowB_Y + 3} width={14} height={14} rx={4} fill={ROSE} opacity={0.85} />
   371→      <text x={23} y={rowB_Y + 14.5} fontSize={9.5} fontWeight={600} fill="#d4d4d8">
   372→        {svgName(nameB)}
   373→      </text>
   374→      <text x={5} y={rowD_Y + 14.5} fontSize={11} fontWeight={700} fill={AMBER}>
   375→        Δ
   376→      </text>
   377→      <text x={21} y={rowD_Y + 14.5} fontSize={8.5} fill={TEXT_MUT}>
   378→        A − B
   379→      </text>
   380→
   381→      {/* ---- row A strip ---- */}
   382→      {sa.leans.slice(0, n).map((lean, i) => {
   383→        const unwritten = Math.abs(lean) < UNWRITTEN_EPS;
   384→        return (
   385→          <rect
   386→            key={`a-${i}`}
   387→            data-testid="diff-cell-a"
   388→            x={cellX(i)}
   389→            y={rowA_Y}
   390→            width={cellW}
   391→            height={rowH}
   392→            rx={4}
   393→            fill={unwritten ? UNWRITTEN_FILL : lean >= 0 ? EM : ROSE}
   394→            opacity={unwritten ? 1 : opa(Math.abs(lean), maxAbs)}
   395→            stroke={unwritten ? UNWRITTEN_EDGE : "none"}
   396→            strokeWidth={unwritten ? 0.75 : 0}
   397→          >
   398→            <title>{`A · M${i + 1} — ${
   399→              unwritten ? "unwritten (no learned lean)" : `lean ${fmtLean(lean)} (${lean >= 0 ? "reward" : "punishment"}-shaped)`
   400→            }`}</title>
   401→          </rect>
   402→        );
   403→      })}
   404→
   405→      {/* ---- row B strip ---- */}
   406→      {sb.leans.slice(0, n).map((lean, i) => {
   407→        const unwritten = Math.abs(lean) < UNWRITTEN_EPS;
   408→        return (
   409→          <rect
   410→            key={`b-${i}`}
   411→            data-testid="diff-cell-b"
   412→            x={cellX(i)}
   413→            y={rowB_Y}
   414→            width={cellW}
   415→            height={rowH}
   416→            rx={4}
   417→            fill={unwritten ? UNWRITTEN_FILL : lean >= 0 ? EM : ROSE}
   418→            opacity={unwritten ? 1 : opa(Math.abs(lean), maxAbs)}
   419→            stroke={unwritten ? UNWRITTEN_EDGE : "none"}
   420→            strokeWidth={unwritten ? 0.75 : 0}
   421→          >
   422→            <title>{`B · M${i + 1} — ${
   423→              unwritten ? "unwritten (no learned lean)" : `lean ${fmtLean(lean)} (${lean >= 0 ? "reward" : "punishment"}-shaped)`
   424→            }`}</title>
   425→          </rect>
   426→        );
   427→      })}
   428→
   429→      {/* ---- Δ strip (A − B) ---- */}
   430→      {deltas.map((d, i) => {
   431→        const unwritten = Math.abs(d) < UNWRITTEN_EPS;
   432→        const isGap = gap !== null && i === gap.index;
   433→        return (
   434→          <g key={`d-${i}`}>
   435→            <rect
   436→              data-testid="diff-cell-delta"
   437→              x={cellX(i)}
   438→              y={rowD_Y}
   439→              width={cellW}
   440→              height={rowH}
   441→              rx={4}
   442→              fill={unwritten ? UNWRITTEN_FILL : d >= 0 ? EM : ROSE}
   443→              opacity={unwritten ? 1 : opa(Math.abs(d), maxDelta)}
   444→              stroke={unwritten ? UNWRITTEN_EDGE : "none"}
   445→              strokeWidth={unwritten ? 0.75 : 0}
   446→            >
   447→              <title>{`Δ M${i + 1} — ${fmtDelta2(d)} ${
   448→                unwritten ? "(identical leans)" : d >= 0 ? "(A stronger)" : "(B stronger)"
   449→              } · A ${fmtLean(sa.leans[i] ?? 0)} vs B ${fmtLean(sb.leans[i] ?? 0)}`}</title>
   450→            </rect>
   451→            {isGap ? (
   452→              <rect
   453→                x={cellX(i) - 1.5}
   454→                y={rowD_Y - 1.5}
   455→                width={cellW + 3}
   456→                height={rowH + 3}
   457→                rx={5}
   458→                fill="none"
   459→                stroke={AMBER}
   460→                strokeWidth={1.25}
   461→              />
   462→            ) : null}
   463→          </g>
   464→        );
   465→      })}
   466→
   467→      {/* ---- shared M axis ---- */}
   468→      {Array.from({ length: n }, (_, i) => (
   469→        <text key={`m-${i}`} x={cellX(i) + cellW / 2} y={131} textAnchor="middle" fontSize={9} fill={TEXT_MUT}>
   470→          M{i + 1}
   471→        </text>
   472→      ))}
   473→
   474→      {/* ---- per-channel Δ values (readable on the ink background) ---- */}
   475→      {deltas.map((d, i) => (
   476→        <text key={`dv-${i}`} x={cellX(i) + cellW / 2} y={145} textAnchor="middle" fontSize={7.5} fill={TEXT_DIM}>
   477→          {fmtDelta2(d)}
   478→        </text>
   479→      ))}
   480→
   481→      {/* ---- legend ---- */}

   618→  const [exported, setExported] = useState<null | "text" | "png">(null);
   619→  const resetTimer = useRef<number | null>(null);
   620→  const fingerprintRef = useRef<HTMLDivElement | null>(null);
   621→
   622→  // sticky rows: keep the last non-null pair so the exit animation has
   623→  // content. React's render-phase "adjust state when a prop changes" pattern
   624→  // (see react.dev/learn/you-might-not-need-an-effect) — no effect needed.
   625→  const [shown, setShown] = useState<readonly [BrainRow, BrainRow] | null>(rows);
   626→  const [prevRows, setPrevRows] = useState<readonly [BrainRow, BrainRow] | null>(rows);
   627→  if (rows !== prevRows) {
   628→    setPrevRows(rows);
   629→    if (rows) setShown(rows);
   630→  }
   631→
   632→  // click when the dialog opens (fires on every false → true transition)
   633→  useEffect(() => {
   634→    if (open) playSound("click");
   635→  }, [open]);
   636→
   637→  // ---- fetch BOTH snapshots on open (state resets happen inside the async
   638→  // task so the effect body stays side-effect free until the microtask runs)
   639→  useEffect(() => {
   640→    if (!open || !shown) return;
   641→    const [rowA, rowB] = shown;
   642→    let cancelled = false;
   643→    (async () => {
   644→      setLoading(true);
   645→      setError(null);
   646→      setSnapshots(null);
   647→      setExported(null);
   648→      const [resA, resB] = await Promise.allSettled([fetchBrain(rowA.id), fetchBrain(rowB.id)]);
   649→      if (cancelled) return;
   650→      if (resA.status === "fulfilled" && resB.status === "fulfilled") {
   651→        setSnapshots({ a: resA.value.snapshot, b: resB.value.snapshot });
   652→      } else {
   653→        const failedA = resA.status === "rejected";
   654→        const name = failedA ? rowA.name : rowB.name;
   655→        const reason = failedA
   656→          ? (resA as PromiseRejectedResult).reason
   657→          : (resB as PromiseRejectedResult).reason;
   658→        const msg = reason instanceof Error ? reason.message : "fetch failed";
   659→        setError(`Couldn't load “${name}” — ${msg}`);
   660→      }
   661→      setLoading(false);
   662→    })();
   663→    return () => {
   664→      cancelled = true;
   665→    };
   666→  }, [open, shown, retryNonce]);
   667→
   668→  const stats = useMemo(() => {
   669→    if (!snapshots) return null;
   670→    return { a: deriveStats(snapshots.a), b: deriveStats(snapshots.b) };
   671→  }, [snapshots]);
   672→
   673→  const gap = useMemo(

   683→
   684→  // ---- inline feedback chip helper ---------------------------------------
   685→  const queueReset = useCallback(() => {
   686→    if (resetTimer.current) window.clearTimeout(resetTimer.current);
   687→    resetTimer.current = window.setTimeout(() => setExported(null), 2000);
   688→  }, []);
   689→  useEffect(() => {
   690→    return () => {
   691→      if (resetTimer.current) window.clearTimeout(resetTimer.current);
   692→    };
   693→  }, []);
   694→
   695→  // ---- export 1: copy diff (clipboard with file-download fallback) -------
   696→  const copyDiff = useCallback(async () => {
   697→    if (!shown || !snapshots || !stats) return;
   698→    const text = buildDiffText(
   699→      shown[0],
   700→      shown[1],
   701→      snapshots.a,
   702→      snapshots.b,
   703→      stats.a,
   704→      stats.b,
   705→      gap,
   706→    );
   707→    try {
   708→      await navigator.clipboard.writeText(text);
   709→    } catch {
   710→      // clipboard API unavailable (or denied) → fall back to a file download
   711→      const a = document.createElement("a");
   712→      a.href = URL.createObjectURL(new Blob([text], { type: "text/plain;charset=utf-8" }));
   713→      a.download = `fly-genomediff-${sanitizeFileName(shown[0].name)}-vs-${sanitizeFileName(shown[1].name)}.txt`;
   714→      a.click();
   715→      URL.revokeObjectURL(a.href);
   716→    }
   717→    setExported("text");
   718→    playSound("ding");
   719→    queueReset();
   720→  }, [shown, snapshots, stats, gap, queueReset]);
   721→
   722→  // ---- export 2: fingerprint PNG (SVG → 2× canvas raster → download) -----
   723→  const exportPng = useCallback(() => {
   724→    if (!shown) return;

   726→    if (!svg) return;
   727→    try {
   728→      const clone = svg.cloneNode(true) as SVGSVGElement;
   729→      clone.setAttribute("width", String(SVG_W));
   730→      clone.setAttribute("height", String(SVG_H));
   731→      clone.setAttribute("xmlns", "http://www.w3.org/2000/svg");
   732→      const xml = new XMLSerializer().serializeToString(clone);
   733→      const url = URL.createObjectURL(new Blob([xml], { type: "image/svg+xml;charset=utf-8" }));
   734→      const img = new Image();
   735→      img.onload = () => {
   736→        const scale = 2;
   737→        const canvas = document.createElement("canvas");
   738→        canvas.width = SVG_W * scale;
   739→        canvas.height = SVG_H * scale;
   740→        const c = canvas.getContext("2d");

   742→          c.fillStyle = INK;
   743→          c.fillRect(0, 0, canvas.width, canvas.height);
   744→          c.drawImage(img, 0, 0, canvas.width, canvas.height);

   746→            if (!blob) return;
   747→            const a = document.createElement("a");
   748→            a.href = URL.createObjectURL(blob);
   749→            a.download = `fly-genomediff-${sanitizeFileName(shown[0].name)}-vs-${sanitizeFileName(shown[1].name)}.png`;
   750→            a.click();
   751→            URL.revokeObjectURL(a.href);
   752→          });
   753→        }
   754→        URL.revokeObjectURL(url);
   755→        setExported("png");
   756→        playSound("ding");
   757→        queueReset();
   758→      };
   759→      img.onerror = () => URL.revokeObjectURL(url);
   760→      img.src = url;
   761→    } catch {
   762→      /* rasterization unsupported — ignore */
   763→    }
   764→  }, [shown, queueReset]);

   773→      : dBal > 0

   788→    let bothUnwritten = 0;

   790→      const a = stats.a.leans[i];
   791→      const b = stats.b.leans[i];
   792→      const au = Math.abs(a) < UNWRITTEN_EPS;
   793→      const bu = Math.abs(b) < UNWRITTEN_EPS;
   794→      if (au && bu) bothUnwritten++;
   795→      else if (au || bu) agree++; // one writes it — no sign conflict
   796→      else if (a >= 0 === b >= 0) agree++;
   797→      else flips++;
   798→    }
