     3→/**
     4→ * HowItWorksDiagrams — three hand-crafted, animated SVG illustrations for the
     5→ * How It Works tab. Each diagram is fully self-contained: its own scoped
     6→ * <style> block (class prefix `hiw-`), a desktop (horizontal) variant shown
     7→ * at md+, and a stacked variant for narrow viewports, both viewBox-scaled.
     8→ *
     9→ * Region colors mirror REGION_META in BrainVisualizer3D.tsx (duplicated as
    10→ * literals so the docs tab never pulls three.js into its bundle). Light mode
    11→ * uses slightly deeper strokes of the same hues for contrast on white; text
    12→ * uses the theme's --foreground / --muted-foreground via CSS classes so both
    13→ * modes stay correct. Animations are pure CSS keyframes and are switched off
    14→ * under prefers-reduced-motion.
    15→ */

    35→/** Tiny arrowhead marker (points +x; rotate as needed). */
    36→function Arrow({
    37→  x,
    38→  y,
    39→  a = 0,
    40→  s = 1,
    41→  className,
    42→  fill,
    43→}: {
    44→  x: number;
    45→  y: number;
    46→  a?: number;
    47→  s?: number;
    48→  className?: string;
    49→  fill?: string;
    50→}) {
    51→  return (
    52→    <path
    53→      d="M -6.5 -5 L 6.5 0 L -6.5 5 Z"
    54→      transform={`translate(${x} ${y}) rotate(${a}) scale(${s})`}
    55→      className={className}
    56→      fill={fill}
    57→      aria-hidden="true"
    58→    />
    59→  );
    60→}
