    85→export interface FingerprintStats {
    86→  /** per-MBON learned lean: mean Kenyon→MBON weight × compartment valence.
    87→   *  > 0 = reward-shaped (approach-leaning), < 0 = punishment-shaped. */
    88→  leans: number[];
    89→  /** per-MBON net motor effect: mean MBON→motor weight over motors */
    90→  drives: number[];
    91→  /** +1 for reward-side (appetitive) MBONs, −1 for punishment-side (aversive) */
    92→  valences: number[];
    93→  strongest: { label: string; index: number; lean: number; valenceLabel: string } | null;
    94→  rewardMass: number;
    95→  punishMass: number;
    96→  balance: number;
    97→  plastic: number;
    98→  innate: number;
    99→  total: number;
   100→  kmCount: number;
   101→  mmCount: number;
   102→  gfCount: number;
   103→  biasCount: number;
   104→  neurons: number;
   105→  seed: number;
   106→  arch: BrainArchitecture;
   107→}
   108→
   109→/** MBON compartment rule, read from the engine: the first ceil(mbon/2) MBONs
   110→ *  are appetitive (+1), the rest aversive (−1) — like real mushroom-body
   111→ *  compartments. kenyonToMbon is MBON-major: [m * kenyonCount + k].
   112→ *  mbonToMotor is motor-major: [motor * mbonCount + m]. */
   113→function deriveStats(snapshot: BrainSnapshot): FingerprintStats {
   114→  const arch = snapshot.arch;
   115→  const mbonCount = arch.mbonCount;
   116→  const appHalf = Math.ceil(mbonCount / 2);
   117→  const km = snapshot.weights.kenyonToMbon ?? [];
   118→  const mm = snapshot.weights.mbonToMotor ?? [];
   119→  // the giant-fiber (lobula→motor) escape reflex is present in every
   120→  // snapshot the engine serializes, but older type declarations omit it
   121→  const gf = (snapshot.weights as { lobulaToMotor?: number[] }).lobulaToMotor ?? [];
   122→  const mbonBias = snapshot.weights.mbonBias ?? [];
   123→  const motorBias = snapshot.weights.motorBias ?? [];
   124→
   125→  const leans: number[] = [];
   126→  const drives: number[] = [];
   127→  const valences: number[] = [];
   128→  for (let m = 0; m < mbonCount; m++) {
   129→    const valence = m < appHalf ? 1 : -1;
   130→    valences.push(valence);
   131→    // mean Kenyon→MBON weight for this memory channel
   132→    const start = m * arch.kenyonCount;
   133→    let sum = 0;
   134→    let n = 0;
   135→    for (let k = start; k < start + arch.kenyonCount && k < km.length; k++) {
   136→      sum += km[k] ?? 0;
   137→      n++;
   138→    }
   139→    const meanKm = n > 0 ? sum / n : 0;
   140→    // sign by compartment: + = reward wrote this channel, − = punishment did
   141→    leans.push(meanKm * valence);
   142→    // net effect of this channel on the motors (mean over motor outputs)
   143→    let dsum = 0;
   144→    let dn = 0;
   145→    for (let o = 0; o < arch.motorCount; o++) {
   146→      const idx = o * mbonCount + m;
   147→      if (idx < mm.length) {
   148→        dsum += mm[idx] ?? 0;
   149→        dn++;
   150→      }
   151→    }
   152→    drives.push(dn > 0 ? dsum / dn : 0);
   153→  }
   154→
   155→  let strongest: FingerprintStats["strongest"] = null;
   156→  for (let m = 0; m < leans.length; m++) {
   157→    if (strongest === null || Math.abs(leans[m]) > Math.abs(leans[strongest.index])) {
   158→      strongest = {
   159→        label: `M${m + 1}`,
   160→        index: m,
   161→        lean: leans[m],
   162→        valenceLabel: valences[m] > 0 ? "reward-side" : "punishment-side",
   163→      };
   164→    }
   165→  }
   166→
   167→  const rewardMass = leans.reduce((a, v) => a + Math.max(v, 0), 0);
   168→  const punishMass = leans.reduce((a, v) => a + Math.max(-v, 0), 0);
   169→
   170→  const plastic = km.length + mm.length + gf.length + mbonBias.length + motorBias.length;
   171→  const innate = countInnateSynapses(arch);
   172→  const neurons =
   173→    arch.retinaCols * arch.retinaRows * 2 + // retina + lamina (1:1)
   174→    arch.medullaChannels * arch.medullaPools +
   175→    arch.lobulaCount +
   176→    arch.kenyonCount +
   177→    arch.mbonCount +
   178→    arch.motorCount;
   179→
   180→  return {
   181→    leans,
   182→    drives,
   183→    valences,
   184→    strongest,
   185→    rewardMass,
   186→    punishMass,
   187→    balance: rewardMass - punishMass,
   188→    plastic,
   189→    innate,
   190→    total: plastic + innate,
   191→    kmCount: km.length,
   192→    mmCount: mm.length,
   193→    gfCount: gf.length,
   194→    biasCount: mbonBias.length + motorBias.length,
   195→    neurons,
   196→    seed: arch.seed,
   197→    arch,
   198→  };
   199→}
   200→
   201→/** Count the brain's innate (hard-wired) synapses from the architecture alone.
   202→ *  Replicates FlyBrain.buildFixedWiring()'s wiring sizes read-only — the
   203→ *  engine itself is locked; these formulas are deterministic on the arch:
   204→ *    retina→lamina (1:1 graded), lamina→medulla (nonzero kernel cells),
   205→ *    medulla→lobula (lobulaCount × min(10, medulla cells)),
   206→ *    lobula→kenyon (kenyonCount × min(6, lobulaCount)). */
   207→function countInnateSynapses(arch: BrainArchitecture): number {
   208→  const { retinaCols: C, retinaRows: R, medullaChannels: CH, medullaPools: P, lobulaCount, kenyonCount } = arch;
   209→  const retina = C * R;
   210→  const poolCols = P === 24 ? 8 : P === 12 ? 4 : Math.ceil(Math.sqrt(P));
   211→  const poolRows = Math.max(1, Math.round(P / poolCols));
   212→  const cellW = C / poolCols;
   213→  const cellH = R / poolRows;
   214→
   215→  let medullaNZ = 0;
   216→  for (let ch = 0; ch < CH; ch++) {
   217→    for (let p = 0; p < P; p++) {
   218→      const pc = p % poolCols;
   219→      const pr = Math.floor(p / poolCols);
   220→      for (let r = 0; r < R; r++) {
   221→        if (r < pr * cellH || r >= (pr + 1) * cellH) continue;
   222→        for (let c = 0; c < C; c++) {
   223→          if (c < pc * cellW || c >= (pc + 1) * cellW) continue;
   224→          const lx = ((c - pc * cellW) / cellW) * 2 - 1;
   225→          const ly = ((r - pr * cellH) / cellH) * 2 - 1;
   226→          let w = 0;
   227→          switch (ch) {
   228→            case 0: w = (1 - Math.abs(lx) * 1.2) * (1 - Math.abs(ly) * 1.2) * 2 - 0.5; break;
   229→            case 1: w = 0.5 - (1 - Math.abs(lx) * 1.2) * (1 - Math.abs(ly) * 1.2) * 2; break;
   230→            case 2: w = -lx * 0.9; break;
   231→            case 3: w = lx * 0.9; break;
   232→            case 4: w = -ly * 0.9; break;
   233→            case 5: w = ly * 0.9; break;
   234→            case 6: w = Math.abs(lx) * Math.abs(ly) * 0.8 + 0.1; break;
   235→            default: w = 0.35; break;
   236→          }
   237→          if (Math.abs(w) > 0.05) medullaNZ++;
   238→        }
   239→      }
   240→    }
   241→  }
   242→
   243→  const lobulaFanIn = Math.min(10, CH * P);
   244→  const kenyonFanIn = Math.min(6, lobulaCount);
   245→  return retina + medullaNZ + lobulaCount * lobulaFanIn + kenyonCount * kenyonFanIn;
   246→}

   252→function sanitizeFileName(name: string): string {
   253→  const clean = name
   254→    .toLowerCase()
   255→    .replace(/[^a-z0-9-_]+/g, "-")
   256→    .replace(/^-+|-+$/g, "");
   257→  return clean || "brain";
   258→}

   260→function fmtLean(v: number): string {
   261→  return `${v >= 0 ? "+" : "-"}${Math.abs(v).toFixed(3)}`;
   262→}

   264→function buildSummaryText(row: BrainRow, snapshot: BrainSnapshot, stats: FingerprintStats): string {
   265→  const taskLabel = row.task.charAt(0).toUpperCase() + row.task.slice(1);
   266→  const dateStr = (() => {
   267→    try {
   268→      return new Date(row.createdAt).toLocaleString("en-US", { timeZone: "UTC", hour12: false });
   269→    } catch {
   270→      return row.createdAt;
   271→    }
   272→  })();
   273→  const meta = snapshot.meta;
   274→  const strongestLine = stats.strongest
   275→    ? `${stats.strongest.label} · ${stats.strongest.valenceLabel} channel · lean ${fmtLean(stats.strongest.lean)} (${stats.strongest.lean >= 0 ? "reward" : "punishment"}-shaped)`
   276→    : "none — an untrained brain";
   277→  const balanceWord =
   278→    Math.abs(stats.balance) < 0.002
   279→      ? "balanced (naive)"
   280→      : stats.balance > 0
   281→        ? "leans reward"
   282→        : "leans punishment";
   283→  const nums = (n: number) => n.toLocaleString("en-US");
   284→
   285→  const L: string[] = [];
   286→  L.push("EXPERT FRUIT FLY — BRAIN REPORT CARD");
   287→  L.push("=".repeat(40));
   288→  L.push("");
   289→  L.push("IDENTITY");
   290→  L.push(`  Name          ${row.name}`);
   291→  L.push(`  Task          ${taskLabel}`);
   292→  L.push(`  Generation    ${row.generation}`);
   293→  L.push(`  Score         ${formatScore(row.task, row.score)}`);
   294→  L.push(`  Saved         ${dateStr} (${relativeDate(row.createdAt)})`);
   295→  L.push(`  Snapshot      ${formatBytes(row.snapshotBytes)}`);
   296→  L.push(
   297→    `  Training      ${nums(meta?.steps ?? 0)} brain-steps · ${meta?.rewards ?? 0} rewards · ${meta?.punishments ?? 0} punishments`,
   298→  );
   299→  if (row.note) L.push(`  Note          ${row.note}`);
   300→  L.push("");
   301→  L.push("PERSONALITY FINGERPRINT");
   302→  L.push(`  How this brain's ${stats.leans.length} memory channels lean — learned from`);
   303→  L.push("  its training. + = reward-shaped (approach), - = punishment-shaped");
   304→  L.push("  (avoid). M1–M6 are reward-side channels, the rest punishment-side.");
   305→  const per = 4;
   306→  for (let i = 0; i < stats.leans.length; i += per) {
   307→    const cells = stats.leans
   308→      .slice(i, i + per)
   309→      .map((v, j) => `M${i + j + 1} ${fmtLean(v)}`.padEnd(14));
   310→    L.push(`  ${cells.join("")}`.trimEnd());
   311→  }
   312→  L.push(`  Strongest channel  ${strongestLine}`);
   313→  L.push(
   314→    `  Valence balance    ${fmtLean(stats.balance)} — ${balanceWord} (reward mass ${stats.rewardMass.toFixed(3)} vs punishment mass ${stats.punishMass.toFixed(3)})`,
   315→  );
   316→  L.push("");
   317→  L.push("WIRING");
   318→  L.push(`  Total synapses     ${nums(stats.total)}`);
   319→  L.push(
   320→    `  Plastic (learned)  ${nums(stats.plastic)} — Kenyon→MBON ${nums(stats.kmCount)} · MBON→motor ${nums(stats.mmCount)} · giant-fiber ${nums(stats.gfCount)} · biases ${nums(stats.biasCount)}`,
   321→  );
   322→  L.push(`  Innate (fixed)     ${nums(stats.innate)} — rebuilt identically from seed ${stats.seed}`);
   323→  L.push(
   324→    `  Brain              ${nums(stats.neurons)} neurons · ${nums(stats.arch.kenyonCount)} Kenyon cells · ${stats.leans.length} memory channels · ${stats.arch.motorCount} motors · retina ${stats.arch.retinaCols}×${stats.arch.retinaRows}`,
   325→  );
   326→  L.push("");
   327→  L.push(`— Expert Fruit Fly report card · generated ${new Date().toISOString()}`);
   328→  return L.join("\n");
   329→}

   336→const INK = "#0d1512"; // panel background (also the PNG canvas fill)
   337→const EM = "#34d399"; // emerald-400 — appetitive / reward-shaped
   338→const EM_SOFT = "#6ee7b7";
   339→const ROSE = "#fb7185"; // rose-400 — aversive / punishment-shaped
   340→const ROSE_SOFT = "#fda4af";
   341→const TEXT_MUT = "#a1a1aa";
   342→const TEXT_DIM = "#71717a";
   343→const GRID = "#9ca3af";

   345→const SVG_W = 560;
   346→const SVG_H = 244;

   348→function FingerprintChart({ stats }: { stats: FingerprintStats }) {
   349→  const n = stats.leans.length;
   350→  const appHalf = Math.ceil(n / 2);
   351→  // wash zones
   352→  const washY = 30;
   353→  const washH = 152;
   354→  const zeroY = washY + washH / 2; // 106
   355→  const maxBar = 70;
   356→  const maxAbsLean = Math.max(0.01, ...stats.leans.map((v) => Math.abs(v)));
   357→  const maxAbsDrive = Math.max(0.01, ...stats.drives.map((v) => Math.abs(v)));
   358→
   359→  // column centers: reward group in the left wash, punishment group right
   360→  const centers: number[] = [];
   361→  const groupCenters = (x0: number, w: number, count: number) => {
   362→    const out: number[] = [];
   363→    for (let i = 0; i < count; i++) out.push(x0 + (w * (i + 0.5)) / count);
   364→    return out;
   365→  };
   366→  centers.push(...groupCenters(12, 266, appHalf), ...groupCenters(282, 266, n - appHalf));
   367→  const spacing = n > 0 ? SVG_W / n : SVG_W;
   368→  const barW = Math.max(10, Math.min(26, spacing * 0.58));
   369→  const driveW = Math.max(14, Math.min(34, spacing * 0.76));
   370→  const strongestIdx = stats.strongest?.index ?? -1;
   371→
   372→  return (
   373→    <svg
   374→      viewBox={`0 0 ${SVG_W} ${SVG_H}`}
   375→      className="h-auto w-full"
   376→      role="img"
   377→      aria-label={`Personality fingerprint: ${n} memory channels — reward-side channels 1–${appHalf}, punishment-side channels ${appHalf + 1}–${n}. Bar direction shows each channel's learned lean: up is reward-shaped, down is punishment-shaped.`}
   378→      fontFamily="system-ui, -apple-system, 'Segoe UI', sans-serif"
   379→    >
   380→      {/* panel */}
   381→      <rect x={0} y={0} width={SVG_W} height={SVG_H} rx={10} fill={INK} />
   382→
   383→      {/* compartment washes + group headers */}
   384→      <rect x={12} y={washY} width={266} height={washH} rx={8} fill={EM} opacity={0.07} />
   385→      <rect x={282} y={washY} width={266} height={washH} rx={8} fill={ROSE} opacity={0.07} />
   386→      <text
   387→        x={145}
   388→        y={21}
   389→        textAnchor="middle"
   390→        fontSize={10}
   391→        fontWeight={600}
   392→        letterSpacing={0.8}
   393→        fill={EM_SOFT}
   394→        opacity={0.9}
   395→      >
   396→        REWARD-SIDE CHANNELS
   397→      </text>
   398→      <text
   399→        x={415}
   400→        y={21}
   401→        textAnchor="middle"
   402→        fontSize={10}
   403→        fontWeight={600}
   404→        letterSpacing={0.8}
   405→        fill={ROSE_SOFT}
   406→        opacity={0.9}
   407→      >
   408→        PUNISHMENT-SIDE CHANNELS
   409→      </text>
   410→
   411→      {/* zero line */}
   412→      <line x1={16} y1={zeroY} x2={544} y2={zeroY} stroke={GRID} strokeWidth={1} strokeDasharray="4 4" opacity={0.55} />
   413→
   414→      {/* diverging lean bars */}
   415→      {stats.leans.map((lean, i) => {
   416→        const h = Math.max(1.5, (Math.abs(lean) / maxAbsLean) * maxBar);
   417→        const up = lean >= 0;
   418→        const color = up ? EM : ROSE;
   419→        return (
   420→          <rect
   421→            key={`bar-${i}`}
   422→            data-testid="fingerprint-bar"
   423→            x={centers[i] - barW / 2}
   424→            y={up ? zeroY - h : zeroY}
   425→            width={barW}
   426→            height={h}
   427→            rx={3}
   428→            fill={color}
   429→            opacity={0.92}
   430→            stroke={i === strongestIdx ? "#f4f4f5" : "none"}
   431→            strokeWidth={i === strongestIdx ? 1.25 : 0}
   432→          >
   433→            <title>{`M${i + 1} · lean ${fmtLean(lean)} — ${up ? "reward" : "punishment"}-shaped memory (${stats.valences[i] > 0 ? "reward-side" : "punishment-side"} channel${i === strongestIdx ? " · strongest" : ""})`}</title>
   434→          </rect>
   435→        );
   436→      })}
   437→
   438→      {/* column labels */}
   439→      {stats.leans.map((_, i) => (
   440→        <text key={`label-${i}`} x={centers[i]} y={196} textAnchor="middle" fontSize={9.5} fill={TEXT_MUT}>
   441→          M{i + 1}
   442→        </text>
   443→      ))}
   444→
   445→      {/* net motor drive strip */}
   446→      {stats.drives.map((drive, i) => {
   447→        const opacity = 0.25 + 0.65 * (Math.abs(drive) / maxAbsDrive);
   448→        const color = drive >= 0 ? EM : ROSE;
   449→        return (
   450→          <rect
   451→            key={`drive-${i}`}
   452→            data-testid="fingerprint-drive"
   453→            x={centers[i] - driveW / 2}
   454→            y={204}
   455→            width={driveW}
   456→            height={16}
   457→            rx={4}
   458→            fill={color}
   459→            opacity={opacity}
   460→          >
   461→            <title>{`M${i + 1} · net motor drive ${fmtLean(drive)} (${drive >= 0 ? "pushes the motors" : "holds the motors back"})`}</title>
   462→          </rect>
   463→        );
   464→      })}
   465→
   466→      {/* legend */}
   467→      <rect x={16} y={227} width={8} height={8} rx={2} fill={EM} />
   468→      <text x={28} y={235} fontSize={9} fill={TEXT_MUT}>
   469→        reward-shaped (approach)
   470→      </text>
   471→      <rect x={392} y={227} width={8} height={8} rx={2} fill={ROSE} />
   472→      <text x={404} y={235} fontSize={9} fill={TEXT_MUT}>
   473→        punishment-shaped (avoid)
   474→      </text>
   475→    </svg>
   476→  );
   477→}

   505→        <Icon className="h-3.5 w-3.5 shrink-0" aria-hidden />
   506→        {label}
   507→      </span>
   508→      <span className="truncate text-right text-sm font-medium">{value}</span>
   509→    </div>
   510→  );
   511→}
   512→
   513→function StatTile({
   514→  icon: Icon,
   515→  label,
   516→  value,
   517→  valueClass,
   518→  sub,
   519→  children,
   520→}: {
   521→  icon: typeof Network;
   522→  label: string;
   523→  value: ReactNode;
   524→  valueClass?: string;
   525→  sub?: ReactNode;
   526→  children?: ReactNode;
   527→}) {
   528→  return (
   529→    <div className="rounded-xl border border-border/60 bg-muted/20 p-3">
   530→      <p className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   531→        <Icon className="h-3.5 w-3.5 shrink-0 text-teal-500" aria-hidden />
   532→        {label}
   533→      </p>
   534→      <p className={cn("mt-1 text-sm font-semibold", valueClass)}>{value}</p>
   535→      {sub ? <p className="mt-0.5 text-[11px] leading-snug text-muted-foreground">{sub}</p> : null}
   536→      {children}
   537→    </div>
   538→  );
   539→}

   541→// ---------------------------------------------------------------------------
   542→// main component
   543→// ---------------------------------------------------------------------------
   544→
   545→export function BrainReportCard({ row, open, onOpenChange }: BrainReportCardProps) {
   546→  const [snapshot, setSnapshot] = useState<BrainSnapshot | null>(null);
   547→  const [loading, setLoading] = useState(false);
   548→  const [error, setError] = useState<string | null>(null);
   549→  const [retryNonce, setRetryNonce] = useState(0);
   550→  /** transient inline feedback: "Copied ✓" / "Saved ✓" */
   551→  const [exported, setExported] = useState<null | "text" | "png">(null);
   552→  const resetTimer = useRef<number | null>(null);
   553→  const fingerprintRef = useRef<HTMLDivElement | null>(null);
   554→
   555→  // sticky row: keep the last non-null row so the exit animation has content
   556→  const [shown, setShown] = useState<BrainRow | null>(row);
   557→  useEffect(() => {
   558→    if (row) setShown(row);
   559→  }, [row]);
   560→
   561→  const rowId = row?.id ?? null;
   562→
   563→  // ---- fetch the snapshot on open (existing fetchBrain helper) -----------
   564→  useEffect(() => {
   565→    if (!open || !rowId) return;
   566→    let cancelled = false;
   567→    setLoading(true);
   568→    setError(null);
   569→    setSnapshot(null);
   570→    setExported(null);
   571→    (async () => {
   572→      try {
   573→        const { snapshot: snap } = await fetchBrain(rowId);
   574→        if (!cancelled) setSnapshot(snap);
   575→      } catch (err) {
   576→        if (!cancelled) setError(err instanceof Error ? err.message : "Could not load this brain");
   577→      } finally {
   578→        if (!cancelled) setLoading(false);
   579→      }
   580→    })();
   581→    return () => {
   582→      cancelled = true;
   583→    };
   584→  }, [open, rowId, retryNonce]);
   585→
   586→  const stats = useMemo(() => (snapshot ? deriveStats(snapshot) : null), [snapshot]);
   587→
   588→  // ---- inline feedback chip helper ---------------------------------------
   589→  const queueReset = useCallback(() => {
   590→    if (resetTimer.current) window.clearTimeout(resetTimer.current);
   591→    resetTimer.current = window.setTimeout(() => setExported(null), 2000);
   592→  }, []);
   593→  useEffect(() => {
   594→    return () => {
   595→      if (resetTimer.current) window.clearTimeout(resetTimer.current);
   596→    };
   597→  }, []);
   598→
   599→  // ---- export 1: copy summary (clipboard with file-download fallback) ---
   600→  const copySummary = useCallback(async () => {
   601→    if (!shown || !snapshot || !stats) return;
   602→    const text = buildSummaryText(shown, snapshot, stats);
   603→    try {
   604→      await navigator.clipboard.writeText(text);
   605→    } catch {
   606→      // clipboard API unavailable (or denied) → fall back to a file download
   607→      const a = document.createElement("a");
   608→      a.href = URL.createObjectURL(new Blob([text], { type: "text/plain;charset=utf-8" }));
   609→      a.download = `fly-reportcard-${sanitizeFileName(shown.name)}.txt`;
   610→      a.click();
   611→      URL.revokeObjectURL(a.href);
   612→    }
   613→    setExported("text");
   614→    playSound("ding");
   615→    queueReset();
   616→  }, [shown, snapshot, stats, queueReset]);
   617→
   618→  // ---- export 2: fingerprint PNG (SVG → 2× canvas raster → download) ----
   619→  const exportPng = useCallback(() => {
   620→    if (!shown || !stats) return;
   621→    const svg = fingerprintRef.current?.querySelector("svg");
   622→    if (!svg) return;
   623→    try {
   624→      const clone = svg.cloneNode(true) as SVGSVGElement;
   625→      clone.setAttribute("width", String(SVG_W));
   626→      clone.setAttribute("height", String(SVG_H));
   627→      clone.setAttribute("xmlns", "http://www.w3.org/2000/svg");
   628→      const xml = new XMLSerializer().serializeToString(clone);
   629→      const url = URL.createObjectURL(new Blob([xml], { type: "image/svg+xml;charset=utf-8" }));
   630→      const img = new Image();
   631→      img.onload = () => {
   632→        const scale = 2;
   633→        const canvas = document.createElement("canvas");
   634→        canvas.width = SVG_W * scale;
   635→        canvas.height = SVG_H * scale;
   636→        const c = canvas.getContext("2d");
   637→        if (c) {
   638→          c.fillStyle = INK;
   639→          c.fillRect(0, 0, canvas.width, canvas.height);
   640→          c.drawImage(img, 0, 0, canvas.width, canvas.height);
   641→          canvas.toBlob((blob) => {
   642→            if (!blob) return;
   643→            const a = document.createElement("a");
   644→            a.href = URL.createObjectURL(blob);
   645→            a.download = `fly-reportcard-${sanitizeFileName(shown.name)}.png`;
   646→            a.click();
   647→            URL.revokeObjectURL(a.href);
   648→          });
   649→        }
   650→        URL.revokeObjectURL(url);
   651→        setExported("png");
   652→        playSound("ding");
   653→        queueReset();
   654→      };
   655→      img.onerror = () => URL.revokeObjectURL(url);
   656→      img.src = url;
   657→    } catch {
   658→      /* rasterization unsupported — ignore */
   659→    }
   660→  }, [shown, stats, queueReset]);
   661→
   662→  const meta = snapshot?.meta;
   663→  const balance =
   664→    stats && Math.abs(stats.balance) < 0.002
   665→      ? { word: "balanced (naive)", cls: "text-muted-foreground" }
   666→      : stats && stats.balance > 0
   667→        ? { word: "leans reward", cls: "text-emerald-500 dark:text-emerald-400" }
   668→        : { word: "leans punishment", cls: "text-rose-500 dark:text-rose-400" };
   669→  const plasticShare = stats && stats.total > 0 ? stats.plastic / stats.total : 0;
   670→  const nums = (v: number) => v.toLocaleString("en-US");
   671→
   672→  return (
   673→    <Dialog open={open} onOpenChange={onOpenChange}>
   674→      <DialogContent
   675→        data-testid="report-card"
   676→        className={cn(
   677→          "max-h-[85vh] overflow-y-auto p-4 sm:max-w-lg sm:p-6",
   678→          SCROLLBAR,
   679→        )}
   680→      >
   681→        <DialogHeader>
   682→          <DialogTitle className="flex items-center gap-2 text-left">
   683→            <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg border border-teal-500/30 bg-teal-500/10">
   684→              <GraduationCap className="h-4 w-4 text-teal-500 dark:text-teal-400" aria-hidden />
   685→            </span>
   686→            Brain report card
   687→          </DialogTitle>
   688→          <DialogDescription>
   689→            {shown ? (
   690→              <>
   691→                Identity, learned personality and wiring stats for &ldquo;{shown.name}&rdquo; — share it
   692→                as text or an image.
   693→              </>
   694→            ) : (
   695→              "Loading this brain's report card…"
   696→            )}
   697→          </DialogDescription>
   698→        </DialogHeader>
   699→
   700→        {loading ? (
   701→          <div className="flex flex-col items-center gap-3 py-10 text-center" data-testid="report-loading">
   702→            <Loader2 className="h-6 w-6 animate-spin text-teal-500" aria-hidden />
   703→            <p className="text-sm text-muted-foreground">Fetching the connectome…</p>
   704→          </div>
   705→        ) : error ? (
   706→          <div className="flex flex-col items-center gap-3 rounded-xl border border-rose-500/30 bg-rose-500/5 px-4 py-8 text-center">
   707→            <p className="text-sm text-rose-500 dark:text-rose-400">Couldn&apos;t load this brain: {error}</p>
   708→            <Button
   709→              variant="outline"
   710→              className="h-11"
   711→              onClick={() => setRetryNonce((v) => v + 1)}
   712→            >
   713→              <RefreshCw className="h-4 w-4" /> Try again
   714→            </Button>
   715→          </div>
   716→        ) : shown && snapshot && stats ? (
   717→          <motion.div
   718→            initial={{ opacity: 0, y: 10 }}
   719→            animate={{ opacity: 1, y: 0 }}
   720→            transition={{ duration: 0.3, ease: "easeOut" }}
   721→            className="space-y-4"
   722→          >
   723→            {/* 1 — identity block */}
   724→            <section aria-label="Brain identity" data-testid="report-identity">
   725→              <div className="flex flex-wrap items-center gap-2">
   726→                <span className="text-base font-semibold leading-tight">{shown.name}</span>
   727→                <TaskBadge task={shown.task} />
   728→                <Badge variant="outline" className="border-teal-500/30 bg-teal-500/10 text-teal-600 dark:text-teal-300">
   729→                  GEN {shown.generation}
   730→                </Badge>
   731→              </div>
   732→              <div className="mt-3 grid grid-cols-1 gap-2 sm:grid-cols-2">
   733→                <InfoRow icon={Trophy} label="Score" value={formatScore(shown.task, shown.score)} />
   734→                <InfoRow
   735→                  icon={Calendar}
   736→                  label="Saved"
   737→                  value={
   738→                    <span title={new Date(shown.createdAt).toLocaleString()}>
   739→                      {relativeDate(shown.createdAt)}
   740→                    </span>
   741→                  }
   742→                />
   743→                <InfoRow icon={Dna} label="Snapshot" value={formatBytes(shown.snapshotBytes)} />
   744→                <InfoRow
   745→                  icon={Activity}
   746→                  label="Training"
   747→                  value={`${nums(meta?.steps ?? 0)} steps · ${meta?.rewards ?? 0}✓ / ${meta?.punishments ?? 0}✕`}
   748→                  className="sm:col-span-2"
   749→                />
   750→              </div>
   751→              {shown.note ? (
   752→                <p
   753→                  className="mt-2 flex items-start gap-1.5 rounded-lg border border-border/60 bg-muted/30 px-3 py-2 text-xs text-muted-foreground"
   754→                  data-testid="report-note"
   755→                >
   756→                  <StickyNote className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   757→                  <span className="italic">{shown.note}</span>
   758→                </p>
   759→              ) : null}
   760→            </section>
   761→
   762→            {/* 2 — personality fingerprint */}
   763→            <section aria-label="Personality fingerprint">
   764→              <h4 className="flex items-center gap-1.5 text-sm font-semibold">
   765→                <Sparkles className="h-4 w-4 text-teal-500 dark:text-teal-400" aria-hidden />
   766→                Personality fingerprint
   767→              </h4>
   768→              <div
   769→                ref={fingerprintRef}
   770→                data-testid="fingerprint"
   771→                className="mt-2 overflow-hidden rounded-xl border border-border/60"
   772→              >
   773→                <FingerprintChart stats={stats} />
   774→              </div>
   775→              <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   776→                How this brain&rsquo;s {stats.leans.length} memory channels lean — learned from its
   777→                training. <span className="text-emerald-500 dark:text-emerald-400">Green</span> bars
   778→                lean reward (approach), <span className="text-rose-500 dark:text-rose-400">red</span>{" "}
   779→                lean punishment (avoid); height = how strongly the channel was written. The small
   780→                cells below each column show that channel&rsquo;s net push on the motors. Hover any
   781→                bar for exact numbers.
   782→              </p>
   783→            </section>
   784→
   785→            {/* 3 — stats row */}
   786→            <section
   787→              aria-label="Wiring statistics"
   788→              data-testid="report-stats"
   789→              className="grid grid-cols-1 gap-2 sm:grid-cols-2"
   790→            >
   791→              <StatTile
   792→                icon={Network}
   793→                label="Total synapses"
   794→                value={nums(stats.total)}
   795→                sub={`${nums(stats.neurons)} neurons · ${stats.leans.length} memory channels`}
   796→              />
   797→              <StatTile
   798→                icon={Zap}
   799→                label="Strongest channel"
   800→                value={
   801→                  <span className={stats.strongest && stats.strongest.lean >= 0 ? "text-emerald-500 dark:text-emerald-400" : "text-rose-500 dark:text-rose-400"}>
   802→                    {stats.strongest
   803→                      ? `${stats.strongest.label} · ${fmtLean(stats.strongest.lean)}`
   804→                      : "none yet"}
   805→                  </span>
   806→                }
   807→                sub={
   808→                  stats.strongest
   809→                    ? `${stats.strongest.valenceLabel} channel, ${stats.strongest.lean >= 0 ? "reward" : "punishment"}-shaped (mean Kenyon→MBON)`
   810→                    : undefined
   811→                }
   812→              />
   813→              <StatTile
   814→                icon={Scale}
   815→                label="Valence balance"
   816→                value={<span className={balance?.cls}>{`${fmtLean(stats.balance)} · ${balance?.word}`}</span>}
   817→                sub={`reward mass ${stats.rewardMass.toFixed(3)} vs punishment ${stats.punishMass.toFixed(3)}`}
   818→              />
   819→              <StatTile
   820→                icon={Dna}
   821→                label="Plastic vs innate"
   822→                value={`${nums(stats.plastic)} / ${nums(stats.innate)}`}
   823→                sub="learned vs hard-wired (innate rebuilds from seed)"
   824→              >
   825→                <div
   826→                  className="mt-2 h-1.5 w-full overflow-hidden rounded-full bg-muted"
   827→                  role="progressbar"
   828→                  aria-label="Plastic share of all synapses"
   829→                  aria-valuenow={Math.round(plasticShare * 100)}
   830→                  aria-valuemin={0}
   831→                  aria-valuemax={100}
   832→                >
   833→                  <div
   834→                    data-testid="plastic-share-bar"
   835→                    className="h-full rounded-full bg-teal-500/70"
   836→                    style={{ width: `${Math.round(plasticShare * 100)}%` }}
   837→                  />
   838→                </div>
   839→              </StatTile>
   840→            </section>
   841→
   842→            {/* 4 — export actions (inline feedback chips, no toasts) */}
   843→            <div className="flex flex-col gap-2 sm:flex-row">
   844→              <Button
   845→                variant="outline"
   846→                size="sm"
   847→                data-testid="report-copy"
   848→                className="h-11 flex-1 gap-1.5 px-4"
   849→                onClick={() => void copySummary()}
   850→              >
   851→                <ClipboardCopy className="h-4 w-4" />
   852→                {exported === "text" ? "Copied ✓" : "Copy summary"}
   853→              </Button>
   854→              <Button
   855→                variant="outline"
   856→                size="sm"
   857→                data-testid="report-png"
   858→                className="h-11 flex-1 gap-1.5 px-4"
   859→                onClick={exportPng}
   860→              >
   861→                <Download className="h-4 w-4" />
   862→                {exported === "png" ? "Saved ✓" : "Download PNG"}
   863→              </Button>
   864→            </div>
   865→            <p className="text-[11px] leading-relaxed text-muted-foreground">
   866→              Summary copies as a plain-text report card (falls back to a .txt download if the
   867→              clipboard is blocked). The PNG is a 2× raster of the fingerprint above.
   868→            </p>
   869→          </motion.div>
   870→        ) : null}
   871→      </DialogContent>
   872→    </Dialog>
   873→  );
   874→}
