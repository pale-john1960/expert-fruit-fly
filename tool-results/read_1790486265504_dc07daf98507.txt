    65→function eventIcon(kind: TrainerEvent["kind"]) {
    66→  switch (kind) {
    67→    case "fall":
    68→      return <Zap className="h-3.5 w-3.5 text-rose-400" />;
    69→    case "offroad":
    70→      return <TriangleAlert className="h-3.5 w-3.5 text-amber-400" />;
    71→    case "milestone":
    72→      return <Sparkles className="h-3.5 w-3.5 text-amber-300" />;
    73→    case "gen":
    74→      return <Flag className="h-3.5 w-3.5 text-rose-300" />;
    75→    case "best":
    76→      return <Trophy className="h-3.5 w-3.5 text-emerald-400" />;
    77→    case "load":
    78→      return <Eye className="h-3.5 w-3.5 text-emerald-400" />;
    79→    default:
    80→      return <Info className="h-3.5 w-3.5 text-muted-foreground" />;
    81→  }
    82→}

    84→function TurboControl({
    85→  turbo,
    86→  onChange,
    87→  size = "sm",
    88→}: {
    89→  turbo: number;
    90→  onChange: (t: number) => void;
    91→  size?: "sm" | "md";
    92→}) {
    93→  return (
    94→    <div
    95→      className="flex items-center gap-0.5 rounded-lg border border-white/10 bg-black/45 p-0.5 backdrop-blur-md"
    96→      role="group"
    97→      aria-label="Simulation speed"
    98→    >
    99→      {[1, 3, 10].map((t) => (
   100→        <button
   101→          key={t}
   102→          onClick={() => onChange(t)}
   103→          aria-pressed={turbo === t}
   104→          className={`flex items-center gap-1 rounded-md font-medium transition-colors ${
   105→            size === "md" ? "h-10 px-3 text-sm" : "h-8 px-2.5 text-xs"
   106→          } ${
   107→            turbo === t
   108→              ? "bg-amber-500/90 text-black"
   109→              : "text-amber-200/80 hover:bg-white/10 hover:text-amber-100"
   110→          }`}
   111→        >
   112→          {t === 1 ? null : <FastForward className={size === "md" ? "h-3.5 w-3.5" : "h-3 w-3"} />}
   113→          ×{t}
   114→        </button>
   115→      ))}
   116→    </div>
   117→  );
   118→}

   137→export function BicycleTrainer() {
   138→  const [core] = useState(() => new BicycleTrainerCore(5));
   139→  const retinaCanvas = useRef<HTMLCanvasElement | null>(null);
   140→  const [hud, setHud] = useState<HudSnapshot | null>(null);
   141→  const [leaderBrain, setLeaderBrain] = useState<FlyBrain | null>(null);
   142→  const [running, setRunning] = useState(true);
   143→  const [turbo, setTurbo] = useState(1);
   144→  const [preset, setPresetState] = useState<"steady" | "standard" | "frisky">("standard");
   145→  const [popQueued, setPopQueued] = useState(5);
   146→  const [mutation, setMutation] = useState(0.3);
   147→  const [watchBest, setWatchBest] = useState(false);
   148→  const [brainName, setBrainName] = useState("Dusk Rider");
   149→  const [saving, setSaving] = useState(false);
   150→  const [session, setSession] = useState<{
   151→    generation: number;
   152→    bestDistance: number;
   153→    history: { gen: number; best: number; avg: number }[];
   154→    snapshot: BrainSnapshot;
   155→  } | null>(null);
   156→
   157→  // ---- HUD polling (keeps React renders at ~6 Hz, the scene stays 60 fps)
   158→  useEffect(() => {
   159→    const iv = setInterval(() => {
   160→      setHud(core.hudSnapshot());
   161→      const lead = core.leader;
   162→      if (lead) {
   163→        setLeaderBrain((prev) => (prev === lead.brain ? prev : lead.brain));
   164→      }
   165→    }, 160);
   166→    return () => clearInterval(iv);
   167→  }, [core]);
   168→
   169→  // ---- brain-library load integration + session resume banner
   170→  useEffect(() => {
   171→    const tryConsume = () => {
   172→      const snap = useBrainStore.getState().pending.bicycle;
   173→      if (snap) {
   174→        useBrainStore.getState().consumeLoad("bicycle");
   175→        core.adoptSnapshot(snap);
   176→        setSession(null);
   177→        toast.success(`Loaded "${snap.name}"`, {
   178→          description: `Gen ${snap.generation} champion (${snap.score.toFixed(0)} m) now seeds the whole population.`,
   179→        });
   180→      }
   181→    };
   182→    tryConsume();
   183→    const unsub = useBrainStore.subscribe(tryConsume);
   184→    const sess = BicycleTrainerCore.loadSession();
   185→    if (sess && sess.bestDistance > 1) setSession(sess);
   186→    return () => unsub();
   187→  }, [core]);
   188→
   189→  // ---- keep the R3F subtree out of the HUD re-render loop
   190→  const canvasElement = useMemo(
   191→    () => (
   192→      <Canvas
   193→        dpr={[1, 1.75]}
   194→        camera={{ fov: 55, near: 0.1, far: 2200, position: [0, 2.8, -8] }}
   195→        gl={{ antialias: true, powerPreference: "high-performance" }}
   196→        style={{ position: "absolute", inset: 0 }}
   197→      >
   198→        <DuskScene core={core} retinaCanvas={retinaCanvas} maxRiders={MAX_RIDERS} />
   199→      </Canvas>
   200→    ),
   201→    [core]
   202→  );
   203→
   204→  // ------------------------------------------------------------- controls
   205→  const toggleRun = () => {
   206→    const next = !running;
   207→    setRunning(next);
   208→    core.setRunning(next);
   209→    playSound("click");
   210→  };
   211→
   212→  const doReset = () => {
   213→    core.reset();
   214→    setWatchBest(false);
   215→    setHud(core.hudSnapshot());
   216→    toast.info("Fresh start", { description: "New random brains — Generation 1 rolls out." });
   217→  };
   218→
   219→  const changeTurbo = (t: number) => {
   220→    setTurbo(t);
   221→    core.setTurbo(t);
   222→  };
   223→
   224→  const changePreset = (p: "steady" | "standard" | "frisky") => {
   225→    if (p === preset) return;
   226→    setPresetState(p);
   227→    core.setPreset(p);
   228→    playSound("click");
   229→  };
   230→
   231→  const changePopulation = (v: number[]) => {
   232→    const n = v[0];
   233→    setPopQueued(n);
   234→    core.setQueuedPopulation(n);
   235→  };
   236→
   237→  const changeMutation = (v: number[]) => {
   238→    const s = v[0];
   239→    setMutation(s);
   240→    core.setMutationStrength(s);
   241→  };
   242→
   243→  const toggleWatchBest = (on: boolean) => {
   244→    setWatchBest(on);
   245→    core.setWatchBest(on);
   246→  };
   247→
   248→  const resumeSession = () => {
   249→    if (!session) return;
   250→    playSound("click");
   251→    core.adoptSnapshot(session.snapshot);
   252→    core.history = session.history;
   253→    core.generation = session.generation;
   254→    setSession(null);
   255→    setHud(core.hudSnapshot());
   256→    toast.success("Session resumed", {
   257→      description: `Continuing from generation ${session.generation} (best ${session.bestDistance.toFixed(0)} m).`,
   258→    });
   259→  };
   260→
   261→  const saveBest = async () => {
   262→    const brain = core.bestBrain ?? core.leader?.brain;
   263→    if (!brain) {
   264→      toast.error("Nothing to save yet", { description: "Let at least one generation finish." });
   265→      return;
   266→    }
   267→    const name = brainName.trim() || "Dusk Rider";
   268→    const score = core.bestEverDistance;
   269→    setSaving(true);
   270→    try {
   271→      const snapshot = brain.toJSON("bicycle", name, core.generation, score);
   272→      const res = await fetch("/api/brains", {
   273→        method: "POST",
   274→        headers: { "Content-Type": "application/json" },
   275→        body: JSON.stringify({
   276→          name,
   277→          task: "bicycle",
   278→          generation: core.generation,
   279→          score,
   280→          snapshot,
   281→        }),
   282→      });
   283→      if (!res.ok) {
   284→        const detail = await res.text().catch(() => "");
   285→        throw new Error(detail || `HTTP ${res.status}`);
   286→      }
   287→      toast.success(`Saved "${name}"`, {
   288→        description: `Generation ${core.generation} champion · ${score.toFixed(0)} m — find it in the Brain Library.`,
   289→      });
   290→      playSound("ding");
   291→    } catch (err) {
   292→      const msg = err instanceof Error ? err.message : String(err);
   293→      toast.error("Save failed", {
   294→        description: msg.includes("404")
   295→          ? "The brain library API is not available yet — try again in a moment."
   296→          : msg,
   297→      });
   298→    } finally {
   299→      setSaving(false);
   300→    }
   301→  };
   302→
   303→  // ----------------------------------------------------------------- HUD
   304→  const alive = hud?.alive ?? 0;
   305→  const total = hud?.total ?? 5;
   306→  const gen = hud?.gen ?? 1;
   307→  const bestEver = hud?.bestEver ?? 0;
   308→  const leaderS = hud?.leaderS ?? 0;
   309→  const leaderV = hud?.leaderV ?? 0;
   310→  const leaderPhi = hud?.leaderPhi ?? 0;
   311→  const dop = hud?.leaderDop ?? 0;
   312→  const dopPct = Math.min(1, Math.abs(dop));
   313→  const events = hud?.events ?? [];
   314→  const history = hud?.history ?? [];
   315→  const popApplied = (hud?.popSize ?? popQueued) === popQueued;
   316→
   317→  return (
   318→    <section className="flex flex-col gap-4">
   319→      <Toaster theme="dark" position="bottom-right" closeButton />
   320→
   321→      {/* ------------------------------------------------------- header */}
   322→      <Card className="border-border/70 bg-gradient-to-br from-[#2a1224] via-card to-[#241a20]">
   323→        <CardHeader className="pb-3">
   324→          <div className="flex flex-wrap items-center gap-3">
   325→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-rose-500 to-amber-500 text-white shadow-lg shadow-rose-900/40">
   326→              <Bike className="h-5 w-5" />
   327→            </div>
   328→            <div className="min-w-0 flex-1">
   329→              <CardTitle className="text-lg">Bicycle Training</CardTitle>
   330→              <CardDescription className="mt-0.5 text-sm">
   331→                Each fly rides the same dusk road. Distance = sugar. Falling or
   332→                leaving the road = shock. The best riders breed, generation by
   333→                generation.
   334→              </CardDescription>
   335→            </div>
   336→            <div className="flex items-center gap-2">
   337→              <Badge variant="secondary" className="gap-1.5 border-amber-500/30 bg-amber-500/10 text-amber-300">
   338→                <Sparkles className="h-3 w-3" /> reward +0.03 / m
   339→              </Badge>
   340→              <Badge variant="secondary" className="gap-1.5 border-rose-500/30 bg-rose-500/10 text-rose-300">
   341→                <Zap className="h-3 w-3" /> shock −0.5 on fall
   342→              </Badge>
   343→            </div>
   344→          </div>
   345→        </CardHeader>
   346→      </Card>
   347→
   348→      {/* ------------------------------------------------ resume banner */}
   349→      {session && (
   350→        <div className="flex flex-wrap items-center gap-3 rounded-xl border border-amber-500/40 bg-amber-500/10 px-4 py-3 text-sm text-amber-200">
   351→          <History className="h-4 w-4 shrink-0" />
   352→          <span className="min-w-0 flex-1">
   353→            Previous dusk ride found — generation {session.generation}, best{" "}
   354→            {session.bestDistance.toFixed(0)} m. Continue breeding from that champion?
   355→          </span>
   356→          <Button
   357→            size="sm"
   358→            onClick={resumeSession}
   359→            className="h-9 bg-amber-500 text-black hover:bg-amber-400"
   360→          >
   361→            Resume last session
   362→          </Button>
   363→          <button
   364→            onClick={() => setSession(null)}
   365→            className="rounded-md p-1.5 opacity-60 hover:opacity-100"
   366→            aria-label="Dismiss session banner"
   367→          >
   368→            <X className="h-4 w-4" />
   369→          </button>
   370→        </div>
   371→      )}
   372→
   373→      {/* ------------------------------------------------- main + side */}
   374→      <div className="flex flex-col gap-4 lg:flex-row">
   375→        {/* 3D canvas card */}
   376→        <Card className="min-w-0 flex-1 overflow-hidden p-0">
   377→          <div className="relative aspect-video min-h-[420px] w-full bg-[#150a1e]">
   378→            {canvasElement}
   379→
   380→            {/* HUD overlay */}
   381→            <div className="pointer-events-none absolute inset-0 select-none">
   382→              <div className="absolute left-3 top-3 flex flex-wrap items-center gap-2">
   383→                <StatChip
   384→                  label="Generation"
   385→                  value={`#${gen}`}
   386→                  accent="text-rose-300"
   387→                />
   388→                <StatChip
   389→                  label="Alive"
   390→                  value={`${alive} / ${total}`}
   391→                  accent={alive > 0 ? "text-emerald-300" : "text-rose-300"}
   392→                />
   393→                {watchBest && (
   394→                  <StatChip label="Mode" value="watch best" accent="text-amber-300" />
   395→                )}
   396→              </div>
   397→              <div className="absolute right-3 top-3 flex flex-col items-end gap-2">
   398→                <StatChip
   399→                  label="Best ever"
   400→                  value={`${bestEver.toFixed(0)} m`}
   401→                  accent="text-amber-300"
   402→                />
   403→                <StatChip label="Leader" value={`${leaderS.toFixed(0)} m`} />
   404→                <StatChip label="Speed" value={`${leaderV.toFixed(1)} m/s`} />
   405→              </div>
   406→
   407→              {/* lean + dopamine bottom-left */}
   408→              <div className="absolute bottom-3 left-3 flex items-end gap-3">
   409→                <div className="rounded-lg border border-white/10 bg-black/40 px-2.5 py-2 backdrop-blur-md">
   410→                  <div className="mb-1 text-[10px] uppercase tracking-wide text-white/50">
   411→                    Lean
   412→                  </div>
   413→                  <div className="flex h-12 w-16 items-center justify-center">
   414→                    <div
   415→                      className="h-1.5 w-14 rounded-full bg-gradient-to-r from-rose-400 via-white/80 to-amber-300 shadow"
   416→                      style={{
   417→                        transform: `rotate(${(leaderPhi * (180 / Math.PI)).toFixed(1)}deg)`,
   418→                      }}
   419→                    />
   420→                  </div>
   421→                </div>
   422→                <div className="w-36 rounded-lg border border-white/10 bg-black/40 px-2.5 py-2 backdrop-blur-md">
   423→                  <div className="mb-1 flex items-center justify-between text-[10px] uppercase tracking-wide text-white/50">
   424→                    <span>Dopamine</span>
   425→                    <span className={dop >= 0 ? "text-emerald-300" : "text-rose-300"}>
   426→                      {dop >= 0 ? "+" : ""}
   427→                      {dop.toFixed(2)}
   428→                    </span>
   429→                  </div>
   430→                  <div className="relative h-2 w-full overflow-hidden rounded-full bg-white/10">
   431→                    <div className="absolute left-1/2 top-0 h-full w-px bg-white/40" />
   432→                    <div
   433→                      className={`absolute top-0 h-full ${
   434→                        dop >= 0 ? "bg-emerald-400" : "bg-rose-400"
   435→                      }`}
   436→                      style={
   437→                        dop >= 0
   438→                          ? { left: "50%", width: `${dopPct * 50}%` }
   439→                          : { right: "50%", width: `${dopPct * 50}%` }
   440→                      }
   441→                    />
   442→                  </div>
   443→                </div>
   444→              </div>
   445→
   446→              {/* turbo bottom-right */}
   447→              <div className="pointer-events-auto absolute bottom-3 right-3">
   448→                <TurboControl turbo={turbo} onChange={changeTurbo} />
   449→              </div>
   450→
   451→              {/* paused veil */}
   452→              {!running && (
   453→                <div className="absolute inset-0 flex items-center justify-center bg-black/35 backdrop-blur-[2px]">
   454→                  <div className="flex items-center gap-2 rounded-full border border-white/15 bg-black/60 px-5 py-2.5 text-sm font-medium text-white/90">
   455→                    <Pause className="h-4 w-4" /> Paused — the flies are resting
   456→                  </div>
   457→                </div>
   458→              )}
   459→            </div>
   460→          </div>
   461→        </Card>
   462→
   463→        {/* right column */}
   464→        <div className="flex w-full flex-col gap-4 lg:w-96">
   465→          {/* what the fly sees */}
   466→          <Card>
   467→            <CardHeader className="pb-2">
   468→              <CardTitle className="flex items-center gap-2 text-sm">
   469→                <Eye className="h-4 w-4 text-rose-300" />
   470→                What the fly sees
   471→              </CardTitle>
   472→              <CardDescription className="text-xs">
   473→                The 24×9 retina — road stripe, edge lines, glow posts. Lean
   474→                shifts the whole view.
   475→              </CardDescription>
   476→            </CardHeader>
   477→            <CardContent>
   478→              <div className="overflow-hidden rounded-lg border border-white/10 bg-[#0d0610] p-1.5">
   479→                <canvas
   480→                  ref={retinaCanvas}
   481→                  width={24}
   482→                  height={9}
   483→                  className="block w-full"
   484→                  style={{ imageRendering: "pixelated" }}
   485→                  aria-label="Live view of the fly's 24 by 9 retina"
   486→                  role="img"
   487→                />
   488→              </div>
   489→              <div className="mt-1.5 flex justify-between text-[10px] uppercase tracking-wide text-muted-foreground">
   490→                <span>28 m ahead</span>
   491→                <span>4 m ahead</span>
   492→              </div>
   493→            </CardContent>
   494→          </Card>
   495→
   496→          {/* brain activity */}

   500→              <CardDescription className="text-xs">

   509→          {/* event feed */}

   527→                {events.map((e) => (

   538→              </div>

   542→          {/* controls */}
   543→          <Card>
   544→            <CardHeader className="pb-3">
   545→              <CardTitle className="text-sm">Training controls</CardTitle>
   546→            </CardHeader>
   547→            <CardContent className="flex flex-col gap-4">
   548→              <div className="flex gap-2">
   549→                <Button
   550→                  onClick={toggleRun}
