   106→function fmtInt(n: number): string {
   107→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   108→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k`;
   109→  return `${Math.round(n)}`;
   110→}
   111→
   112→function signed(n: number): string {
   113→  return `${n >= 0 ? "+" : "−"}${Math.abs(n).toFixed(2)}`;
   114→}

   146→
   147→type DemoPhase = "stimulus" | "reward" | "pause";
   148→type DemoLetter = "A" | "B";
   149→
   150→/** One measured trial of the conditioning demo.
   151→ *
   152→ *  index    = MEASUREMENT (plotted): mean motor approach drive, i.e. the
   153→ *             average of all 4 smoothed motorRates sampled across the final
   154→ *             1 s of the 3 s stimulus. Validated against the real engine
   155→ *             (see worklog 7-b): it is the sensitive readout — reward-paired
   156→ *             A holds approach high while shock-paired B suppresses it, and
   157→ *             the effect compounds on repeated runs. valence = secondary
   158→ *             index: mean(appetitive MBON rates) − mean(aversive MBON rates)
   159→ *             (region indices via brain.range("mbon") + brain.mbonValence);
   160→ *             at only 12 pairings it stays within ±0.02 noise, so it is kept
   161→ *             for the tooltip/record but not plotted. */
   162→interface DemoTrialPoint {
   163→  trial: number; // 1-based
   164→  letter: DemoLetter;
   165→  index: number;
   166→  valence: number;
   167→}
   168→
   169→interface DemoResult {
   170→  points: DemoTrialPoint[];
   171→  /** first-4 → last-4 average change per stimulus (the "verdict" numbers) */
   172→  dA: number;
   173→  dB: number;
   174→  gapEarly: number;
   175→  gapLate: number;
   176→  baseline: number;
   177→  run: number;
   178→}
   179→
   180→interface DemoHud {
   181→  status: "idle" | "running" | "done";
   182→  trial: number; // 0-based
   183→  phase: DemoPhase;
   184→  letter: DemoLetter | null;
   185→  progress: number; // 0..1
   186→}
   187→
   188→interface DemoRuntime {
   189→  running: boolean;
   190→  trial: number; // 0-based
   191→  t: number; // demo-time seconds inside the current trial
   192→  delivered: boolean;
   193→  valSum: number;
   194→  apprSum: number;
   195→  n: number;
   196→  points: DemoTrialPoint[];
   197→}
   198→
   199→const DEMO_HUD_IDLE: DemoHud = {
   200→  status: "idle",
   201→  trial: 0,
   202→  phase: "stimulus",
   203→  letter: null,
   204→  progress: 0,
   205→};

   207→export function BrainLab(_props: BrainLabProps = {}) {
   208→  // ---- React-visible state (low-frequency sync from the sim loop) ----
   209→  // brain created with a lazy initializer — FlyBrain construction is pure,
   210→  // deterministic and DOM-free, so it is safe during render (SSR included).
   211→  const [brain, setBrain] = useState<FlyBrain>(() => new FlyBrain(DEFAULT_ARCH_LAB));
   212→  const [running, setRunning] = useState(true);
   213→  const [pattern, setPattern] = useState<Pattern>("sweep");
   214→  const [speed, setSpeed] = useState(1);
   215→  const [autoRotate, setAutoRotate] = useState(true);
   216→  const [showSynapses, setShowSynapses] = useState(true);
   217→  const [ui, setUi] = useState({
   218→    ticks: 0,
   219→    spikes: 0,
   220→    da: 0,
   221→    rewards: 0,
   222→    punishments: 0,
   223→  });
   224→  const [motors, setMotors] = useState<number[]>([0, 0, 0, 0]);
   225→  const [demoHud, setDemoHud] = useState<DemoHud>(DEMO_HUD_IDLE);
   226→  const [demoResult, setDemoResult] = useState<DemoResult | null>(null);
   227→
   228→  // ---- sim state that must NOT trigger re-renders (refs) ----
   229→  const brainRef = useRef<FlyBrain>(brain);
   230→  const runningRef = useRef(true);
   231→  const patternRef = useRef<Pattern>("sweep");
   232→  const speedRef = useRef(1);
   233→  const pendingDaRef = useRef(0);
   234→  const phaseRef = useRef(0);
   235→  const sparkleRef = useRef(new Float32Array(RET_COLS * RET_ROWS));
   236→  const drumRef = useRef(0); // optomotor drum scroll offset (px)
   237→  const spikeEmaRef = useRef(0);
   238→  const retCanvasRef = useRef<HTMLCanvasElement | null>(null);
   239→
   240→  // ---- demo wizard runtime (refs; HUD syncs at the existing ~11 Hz) ----
   241→  const demoRef = useRef<DemoRuntime>({
   242→    running: false,
   243→    trial: 0,
   244→    t: 0,
   245→    delivered: false,
   246→    valSum: 0,
   247→    apprSum: 0,
   248→    n: 0,
   249→    points: [],
   250→  });
   251→  /** the user's stimulus selection before the wizard took over the projector */
   252→  const demoPrevPatternRef = useRef<Pattern | null>(null);
   253→  const demoRunsRef = useRef(0);
   254→
   255→  // ---- demo wizard control ----
   256→  const startDemo = useCallback(() => {
   257→    const d = demoRef.current;
   258→    if (d.running) return;
   259→    d.running = true;
   260→    d.trial = 0;
   261→    d.t = 0;
   262→    d.delivered = false;
   263→    d.valSum = 0;
   264→    d.apprSum = 0;
   265→    d.n = 0;
   266→    d.points = [];
   267→    demoPrevPatternRef.current = patternRef.current;
   268→    setDemoResult(null);
   269→    setDemoHud({ status: "running", trial: 0, phase: "stimulus", letter: "A", progress: 0 });
   270→    // a paused brain would stall the wizard — bring the sim back with it
   271→    if (!runningRef.current) {
   272→      runningRef.current = true;
   273→      setRunning(true);
   274→    }
   275→    playSound("click", 0.6);
   276→  }, []);
   277→
   278→  const abortDemo = useCallback(() => {
   279→    const d = demoRef.current;
   280→    if (!d.running) return;
   281→    d.running = false;
   282→    const prev = demoPrevPatternRef.current;
   283→    if (prev) {
   284→      patternRef.current = prev;
   285→      setPattern(prev);
   286→      demoPrevPatternRef.current = null;
   287→    }
   288→    // brain state is about to change — old measurements are stale
   289→    setDemoResult(null);
   290→    setDemoHud(DEMO_HUD_IDLE);
   291→  }, []);
   292→
   293→  // ---- brain lifecycle (reset button / brain swap) ----
   294→  const bootBrain = useCallback(() => {
   295→    abortDemo(); // "Reset brain" during a demo aborts the wizard cleanly
   296→    const b = new FlyBrain(DEFAULT_ARCH_LAB);
   297→    brainRef.current = b;
   298→    setBrain(b);
   299→    phaseRef.current = 0;
   300→    drumRef.current = 0;
   301→    sparkleRef.current.fill(0);
   302→    pendingDaRef.current = 0;
   303→    spikeEmaRef.current = 0;
   304→    demoRunsRef.current = 0;
   305→    setDemoHud(DEMO_HUD_IDLE);
   306→    setUi({ ticks: 0, spikes: 0, da: 0, rewards: 0, punishments: 0 });
   307→    setMotors(new Array(b.sizes.motor).fill(0));
   308→  }, [abortDemo]);
   309→
   310→  // ---- controls ----
   311→  const toggleRunning = useCallback(() => {
   312→    const next = !runningRef.current;
   313→    runningRef.current = next;
   314→    setRunning(next);
   315→  }, []);
   316→
   317→  const onPatternChange = useCallback((v: string) => {
   318→    const p = (Object.keys(PATTERN_LABELS) as Pattern[]).includes(v as Pattern)
   319→      ? (v as Pattern)
   320→      : "sweep";
   321→    patternRef.current = p;
   322→    setPattern(p);
   323→    if (p === "sparkle") sparkleRef.current.fill(0);
   324→  }, []);
   325→
   326→  const onSpeedChange = useCallback((v: number[]) => {
   327→    const s = v[0] ?? 1;
   328→    speedRef.current = s;
   329→    setSpeed(s);
   330→  }, []);
   331→
   332→  const giveSugar = useCallback(() => {
   333→    pendingDaRef.current = 1;
   334→    playSound("sugar");
   335→  }, []);
   336→
   337→  const giveShock = useCallback(() => {
   338→    pendingDaRef.current = -1;
   339→    playSound("shock");
   340→  }, []);
   341→
   342→  // ---- simulation loop: rAF accumulator at TICK_RATE ----
   343→  useEffect(() => {
   344→    if (!brain) return;
   345→    const COLS = brain.arch.retinaCols;
   346→    const ROWS = brain.arch.retinaRows;
   347→    const input = new Float32Array(COLS * ROWS);
   348→    const STEP_MS = 1000 / TICK_RATE;
   349→    let raf = 0;
   350→    let last = performance.now();
   351→    let acc = 0;
   352→    let lastUi = 0;
   353→    let lastDraw = 0;
   354→
   355→    // MBON compartment indices (global neuron ids) for the valence index
   356→    const mb = brain.range("mbon");
   357→    const mbonApp: number[] = [];
   358→    const mbonAve: number[] = [];
   359→    for (let m = 0; m < brain.sizes.mbon; m++) {
   360→      (brain.mbonValence[m] === 1 ? mbonApp : mbonAve).push(mb.start + m);
   361→    }
   362→
   363→    const paintBar = (cx: number, w: number) => {
   364→      for (let c = cx; c < cx + w; c++) {
   365→        if (c < 0 || c >= COLS) continue;
   366→        for (let r = 0; r < ROWS; r++) input[r * COLS + c] = 1;
   367→      }
   368→    };
   369→
   370→    const computeInput = (dtSec: number) => {
   371→      input.fill(0);
   372→      let pat = patternRef.current;
   373→      const d = demoRef.current;
   374→      if (d.running) {
   375→        // the wizard owns the projector while it runs: the trial's
   376→        // conditioning stimulus, or darkness during the inter-trial pause
   377→        pat =
   378→          d.t < DEMO_STIM_S
   379→            ? d.trial % 2 === 0
   380→              ? "barLeft"
   381→              : "barRight"
   382→            : "dark";
   383→      }
   384→      if (pat === "sweep" || pat === "two") {
   385→        phaseRef.current = (phaseRef.current + dtSec * 0.3 * speedRef.current) % 1;
   386→        paintBar(Math.floor(phaseRef.current * COLS), 2);
   387→        if (pat === "two") {
   388→          const p2 = COLS - 1 - Math.floor(((phaseRef.current + 0.5) % 1) * COLS);
   389→          paintBar(p2 - 1, 2);
   390→        }
   391→        // sparse random dots (visual "noise" the fly also sees)
   392→        for (let d = 0; d < 3; d++) {
   393→          const i = Math.floor(Math.random() * input.length);
   394→          input[i] = Math.max(input[i], 0.35 + Math.random() * 0.3);
   395→        }
   396→      } else if (pat === "sparkle") {
   397→        const sp = sparkleRef.current;
   398→        for (let i = 0; i < sp.length; i++) sp[i] *= 0.85;
   399→        const n = 2 + Math.floor(Math.random() * 4);
   400→        for (let d = 0; d < n; d++) {
   401→          sp[Math.floor(Math.random() * sp.length)] = 0.75 + Math.random() * 0.25;
   402→        }
   403→        for (let i = 0; i < input.length; i++) input[i] = sp[i];
   404→      } else if (pat === "barLeft" || pat === "barRight") {
   405→        // conditioning stimuli A / B: a steady bright bar filling one
   406→        // third of the retina (left or right)
   407→        const third = Math.max(1, Math.round(COLS / 3));
   408→        paintBar(pat === "barLeft" ? 0 : COLS - third, third);
   409→      } else if (pat === "loom") {
   410→        // looming shadow (classic fly escape stimulus): a bright field with
   411→        // an expanding dark disc. Radius eases 0 → ~80 % coverage over
   412→        // LOOM_PERIOD_S, then snaps back — the sweep + reset produce heavy
   413→        // ON/OFF transients that light up the optic lobes. A thin bright
   414→        // leading rim keeps the expanding edge visible at 24×9 pixels.
   415→        phaseRef.current += (dtSec * speedRef.current) / LOOM_PERIOD_S;
   416→        if (phaseRef.current >= 1) phaseRef.current -= 1;
   417→        const R = LOOM_MAX_R * (1 - Math.pow(1 - phaseRef.current, 2));
   418→        input.fill(0.85);
   419→        const cx = (COLS - 1) / 2;
   420→        const cy = (ROWS - 1) / 2;
   421→        for (let r = 0; r < ROWS; r++) {
   422→          const dy = r - cy;
   423→          for (let c = 0; c < COLS; c++) {
   424→            const dx = c - cx;
   425→            const dist = Math.sqrt(dx * dx + dy * dy);
   426→            if (dist <= R) input[r * COLS + c] = 0;
   427→            else if (dist <= R + 1.3) input[r * COLS + c] = 1;
   428→          }
   429→        }
   430→      } else if (pat === "drum") {
   431→        // optomotor drum: alternating vertical stripes (~6 px period)
   432→        // scrolling horizontally; speed follows the pattern-speed slider
   433→        drumRef.current =
   434→          (drumRef.current + dtSec * speedRef.current * DRUM_PX_S) % DRUM_PERIOD;
   435→        const off = drumRef.current;
   436→        for (let c = 0; c < COLS; c++) {
   437→          if ((c + off) % DRUM_PERIOD < DRUM_PERIOD / 2) paintBar(c, 1);
   438→        }
   439→      }
   440→      // "dark" → all zeros
   441→    };
   442→
   443→    const drawRetina = () => {
   444→      const cv = retCanvasRef.current;
   445→      if (!cv) return;
   446→      const ctx = cv.getContext("2d");
   447→      if (!ctx) return;
   448→      ctx.fillStyle = "#07060a";
   449→      ctx.fillRect(0, 0, COLS, ROWS);
   450→      for (let r = 0; r < ROWS; r++) {
   451→        for (let c = 0; c < COLS; c++) {
   452→          const v = input[r * COLS + c];
   453→          if (v <= 0.03) continue;
   454→          const rr = Math.round(18 + 233 * v);
   455→          const gg = Math.round(10 + 181 * v);
   456→          const bb = Math.round(6 + 30 * v);
   457→          ctx.fillStyle = `rgb(${rr},${gg},${bb})`;
   458→          ctx.fillRect(c, r, 1, 1);
   459→        }
   460→      }
   461→    };
   462→
   463→    // ---- demo wizard: clock / reward delivery (before brain.step) ----
   464→    const finishDemo = () => {
   465→      const d = demoRef.current;
   466→      d.running = false;
   467→      // hand the projector back to the user's previous stimulus choice
   468→      const prev = demoPrevPatternRef.current;
   469→      if (prev) {
   470→        patternRef.current = prev;
   471→        setPattern(prev);
   472→        demoPrevPatternRef.current = null;
   473→      }
   474→      // verdict: first-4 vs last-4 trial averages per stimulus
   475→      const pts = d.points;
   476→      const meanIdx = (xs: DemoTrialPoint[]) =>
   477→        xs.length ? xs.reduce((a, p) => a + p.index, 0) / xs.length : 0;
   478→      const aPts = pts.filter((p) => p.letter === "A");
   479→      const bPts = pts.filter((p) => p.letter === "B");
   480→      setDemoResult({
   481→        points: pts.slice(),
   482→        dA: meanIdx(aPts.slice(-4)) - meanIdx(aPts.slice(0, 4)),
   483→        dB: meanIdx(bPts.slice(-4)) - meanIdx(bPts.slice(0, 4)),
   484→        gapEarly: meanIdx(aPts.slice(0, 4)) - meanIdx(bPts.slice(0, 4)),
   485→        gapLate: meanIdx(aPts.slice(-4)) - meanIdx(bPts.slice(-4)),
   486→        baseline: meanIdx(pts.slice(0, 4)),
   487→        run: (demoRunsRef.current += 1),
   488→      });
   489→      setDemoHud({ status: "done", trial: DEMO_TRIALS, phase: "pause", letter: null, progress: 1 });
   490→      playSound("milestone", 0.7);
   491→    };
   492→
   493→    const demoPre = (dtSec: number) => {
   494→      const d = demoRef.current;
   495→      if (!d.running) return;
   496→      // the demo clock rides the same pattern-speed slider
   497→      d.t += dtSec * speedRef.current;
   498→      const letter: DemoLetter = d.trial % 2 === 0 ? "A" : "B";
   499→      if (!d.delivered && d.t >= DEMO_REWARD_AT_S && d.t < DEMO_STIM_S) {
   500→        d.delivered = true;
   501→        // same mechanism as the Sugar / Shock buttons: pending dopamine
   502→        pendingDaRef.current = letter === "A" ? 1 : -1;
   503→        playSound(letter === "A" ? "sugar" : "shock", 0.8);
   504→      }
   505→      if (d.t >= DEMO_STIM_S + DEMO_PAUSE_S) {
   506→        // trial over: freeze the measurement and move on
   507→        const n = Math.max(1, d.n);
   508→        d.points.push({
   509→          trial: d.trial + 1,
   510→          letter,
   511→          index: d.apprSum / n,
   512→          valence: d.valSum / n,
   513→        });
   514→        d.trial += 1;
   515→        d.t = 0;
   516→        d.delivered = false;
   517→        d.valSum = 0;
   518→        d.apprSum = 0;
   519→        d.n = 0;
   520→        if (d.trial >= DEMO_TRIALS) finishDemo();
   521→      }
   522→    };
   523→
   524→    // ---- demo wizard: measurement sampling (after brain.step) ----
   525→    const demoPost = () => {
   526→      const d = demoRef.current;
   527→      if (!d.running) return;
   528→      if (d.t < DEMO_SAMPLE_FROM_S || d.t >= DEMO_STIM_S) return;
   529→      let ms = 0;
   530→      for (let i = 0; i < brain.motorRates.length; i++) ms += brain.motorRates[i];
   531→      d.apprSum += ms / brain.motorRates.length;
